# TAC exports begin with provenance comment rows.
read_tac <- function(path) {
  lines <- readLines(path, n = 25L, warn = FALSE)
  at <- which(startsWith(lines, "ID\t"))
  if (!length(at)) stop("No ID header in TAC export: ", path)
  utils::read.delim(path, skip = at[1L] - 1L, check.names = FALSE,
                    stringsAsFactors = FALSE, quote = "", na.strings = c("", "NA"))
}

# Transcript clusters occasionally have more than one gene annotation.
# Preserve the original ID for traceability and expand valid symbols only
# for gene-level membership; do not duplicate biological sample measurements.
expand_symbols <- function(x) {
  x <- ifelse(is.na(x), "", as.character(x))
  lapply(x, function(s) {
    a <- trimws(unlist(strsplit(gsub("///|//|;|,", "|", s), "|", fixed = TRUE)))
    unique(a[nzchar(a) & !a %in% c("---", "NA", "-")])
  })
}

# ------------------ Database-supported GO classification ----------------
# All qualifying TAC genes and all KEA3 predictions enter annotation. This
# avoids both a manually predetermined candidate set and keyword-only screening.
# The 554-gene reference screen is used only as a coverage
# comparator as it cannot force an enzyme assignment.
# ---------------------- GO functional classification ----------------------
# GO molecular-function terms define kinase, phosphatase and regulatory classes.
# GO biological-process terms provide the corresponding pathway group labels.
# These definitions use org.Hs.eg.db and GO.db annotations and preserve
# annotation evidence and database-version details in the exported tables.

            "GO:0016791", "GO:0019207", "GO:0019208"),
  Biological_role = c("Protein kinase activity", "Lipid kinase activity",
                      "Other kinase activity", "Protein phosphatase activity",
                      "Other phosphatase activity", "Kinase regulator activity",
                      "Phosphatase regulator activity"),
  Figure_Group = c(rep("True kinase enzymes", 3L),
                   rep("True phosphatases", 2L),
                   rep("Regulators / adjacent", 2L)),
  Order = seq_len(7L), stringsAsFactors = FALSE)

  GO_ID = c("GO:0000165", "GO:0043491", "GO:0007049", "GO:0006629",
            "GO:0009117", "GO:0005975", "GO:0006914", "GO:0006950",

# Database-backed mapping: current symbols are preferred. An alias maps only
# when it uniquely identifies one NCBI Entrez gene. Ambiguous aliases are NOT
# assigned annotations from an arbitrarily selected gene.
automatic_go_annotations <- function(gene_symbols) {
  needed <- c("AnnotationDbi", "org.Hs.eg.db", "GO.db")
  for (pkg in needed) {
    if (!requireNamespace(pkg, quietly = TRUE)) {
      if (!requireNamespace("BiocManager", quietly = TRUE))
        install.packages("BiocManager", repos = "https://cloud.r-project.org")
      BiocManager::install(pkg, ask = FALSE, update = FALSE)
    }
    if (!requireNamespace(pkg, quietly = TRUE))
      stop("Required Gene Ontology annotation package unavailable: ", pkg)
  }
  org <- org.Hs.eg.db::org.Hs.eg.db
  genes <- sort(unique(trimws(as.character(gene_symbols))))
  genes <- genes[!is.na(genes) & nzchar(genes)]
  message("Mapping ", length(genes), " distinct symbols with org.Hs.eg.db ...")
  valid <- intersect(genes, AnnotationDbi::keys(org, keytype = "SYMBOL"))
  direct <- if (length(valid))
    suppressMessages(AnnotationDbi::select(org, keys = valid,
      columns = c("ENTREZID", "GENENAME"), keytype = "SYMBOL")) else
      data.frame(SYMBOL=character(), ENTREZID=character(), GENENAME=character())
  direct$Query_Gene <- direct$SYMBOL
  direct$Mapping_Method <- "Official symbol"
  unmatched <- setdiff(genes, unique(direct$Query_Gene))
  valid_alias <- intersect(unmatched, AnnotationDbi::keys(org, keytype = "ALIAS"))
  alias <- if (length(valid_alias)) suppressMessages(AnnotationDbi::select(
    org, keys=valid_alias, columns=c("ENTREZID", "SYMBOL", "GENENAME"),
    keytype="ALIAS")) else data.frame(ALIAS=character(), ENTREZID=character(),
                                      SYMBOL=character(), GENENAME=character())
  if (nrow(alias)) {
    alias$Query_Gene <- alias$ALIAS
    alias$Mapping_Method <- "Unambiguous alias"
    alias <- alias[,c("SYMBOL","ENTREZID","GENENAME","Query_Gene","Mapping_Method")]
  }
  direct <- direct[,c("SYMBOL","ENTREZID","GENENAME","Query_Gene","Mapping_Method")]
  fullmap <- unique(rbind(direct,alias))
  fullmap <- fullmap[!is.na(fullmap$ENTREZID) & nzchar(fullmap$ENTREZID),,drop=FALSE]
  ids_by_gene <- split(fullmap$ENTREZID, fullmap$Query_Gene)
  accepted <- vapply(genes, function(g) {
    ids <- unique(ids_by_gene[[g]])
    if (length(ids) == 1L) as.character(ids[1L]) else NA_character_
  }, character(1))
  names(accepted) <- genes
  ent <- sort(unique(accepted[!is.na(accepted)]))
  message("Retrieving propagated GO annotations for ", length(ent), " unique Entrez IDs ...")
  go <- if (length(ent)) suppressMessages(AnnotationDbi::select(org, keys = ent,
    columns = c("GOALL", "ONTOLOGYALL", "EVIDENCEALL"),
    keytype = "ENTREZID")) else data.frame(ENTREZID=character(),
      GOALL=character(), ONTOLOGYALL=character(), EVIDENCEALL=character())
  go <- go[!is.na(go$GOALL) & nzchar(go$GOALL) &
           go$ONTOLOGYALL %in% c("MF","BP") &
           (is.na(go$EVIDENCEALL) | go$EVIDENCEALL != "ND"),,drop=FALSE]
  go <- unique(go[c("ENTREZID","GOALL","ONTOLOGYALL","EVIDENCEALL")])
  # Keep GO records with valid Entrez identifiers.
  go <- go[!is.na(go$ENTREZID) & nzchar(go$ENTREZID),,drop=FALSE]
  by_id <- split(go, go$ENTREZID)
  role_codes <- GO_MF_CLASSES$GO_ID
  bp_codes <- GO_BP_AXES$GO_ID
  role_pairs <- GO_MF_CLASSES[,c("GO_ID","Biological_role","Figure_Group","Order")]
  bp_pairs <- GO_BP_AXES
  # The rarest process among those documented for this gene is used for the
  # SINGLE row-block label. The multi-label process track is also exported.
  bp_freq <- vapply(bp_codes, function(code) length(unique(go$ENTREZID[
    go$ONTOLOGYALL=="BP" & go$GOALL==code])), numeric(1))
  names(bp_freq) <- bp_codes
  classify_one <- function(g) {
    ids <- unique(ids_by_gene[[g]])
    mapping_status <- if (!length(ids)) "No Entrez mapping" else
       if (length(ids) > 1L) "Ambiguous symbol / alias" else "Mapped"
    id <- accepted[[g]]
    z <- if (!is.na(id)) by_id[[id]] else NULL
    if (is.null(z)) z <- go[0L,,drop=FALSE]
    mf <- z[z$ONTOLOGYALL == "MF",,drop=FALSE]
    bp <- z[z$ONTOLOGYALL == "BP",,drop=FALSE]
    roles <- role_pairs[role_pairs$GO_ID %in% unique(mf$GOALL),,drop=FALSE]
    roles <- roles[order(roles$Order),,drop=FALSE]
    if (nrow(roles)) {
      role <- as.character(roles$Biological_role[1L])
      grp <- as.character(roles$Figure_Group[1L])
      role_id <- as.character(roles$GO_ID[1L])
      evidence <- unique(mf$EVIDENCEALL[mf$GOALL == role_id])
      evidence <- sort(evidence[!is.na(evidence) & nzchar(evidence)])
    } else {
      role <- "No documented kinase/phosphatase molecular function"
      grp <- "Annotation review"
      role_id <- ""
      evidence <- character()
    }
    all_roles <- paste(roles$Biological_role, collapse="; ")
    all_role_ids <- paste(roles$GO_ID, collapse="; ")
    pathways <- bp_pairs[bp_pairs$GO_ID %in% unique(bp$GOALL),,drop=FALSE]
    if (nrow(pathways)) {
      # A less prevalent annotated biological process is more informative.
      # Alphabetical GO ID breaks ties consistently, without manual selection.
      pathways$N_annotated_genes <- bp_freq[pathways$GO_ID]
      pathways <- pathways[order(pathways$N_annotated_genes, pathways$GO_ID),,drop=FALSE]
      axis <- as.character(pathways$Biological_axis[1L])
      axis_id <- as.character(pathways$GO_ID[1L])
      process_ids <- paste(pathways$GO_ID, collapse="; ")
      process_names <- paste(pathways$Biological_axis, collapse="; ")
    } else {
      axis <- if (nrow(roles)) role else "No selected GO biological process"
      axis_id <- ""
      process_ids <- ""
      process_names <- ""
    }
    e <- if (length(evidence)) paste(evidence, collapse="; ") else ""
    ev_status <- if (!length(evidence)) "No supporting GO MF evidence" else
      if (any(!evidence %in% c("IEA","ISS","ISO","ISA","ISM","RCA")))
        "Curator-supported or experimental GO annotation" else
        "Sequence-inferred or electronic GO annotation"
    data.frame(Gene=g, Entrez_ID=ifelse(is.na(id),"",id),
      Symbol_mapping=mapping_status,
      Official_symbol=if (!is.na(id)) {
        syms <- unique(fullmap$SYMBOL[fullmap$Query_Gene==g & fullmap$ENTREZID==id])
        if (length(syms)) as.character(syms[1]) else ""
      } else "",
      Figure_Group=grp, Biological_Group=axis,
      GO_MF_Primary_Role=role, GO_MF_Primary_ID=role_id,
      GO_MF_All_Roles=all_roles, GO_MF_All_Role_IDs=all_role_ids,
      GO_MF_Evidence_Codes=e, GO_MF_Evidence_Status=ev_status,
      GO_BP_Primary_ID=axis_id, GO_BP_All_Axes=process_names,
      GO_BP_All_Axis_IDs=process_ids,
      N_GO_MF_terms=nrow(mf), N_GO_BP_terms=nrow(bp),
      Annotation_status=if (grp=="Annotation review") "Unresolved by functional GO" else
         "GO molecular function assigned",
      stringsAsFactors=FALSE)
  }
  result <- do.call(rbind, lapply(genes, classify_one))
  rownames(result) <- NULL
  list(annotations=result, GO_records=go, symbol_map=fullmap,
       molecular_function_terms=GO_MF_CLASSES,
       biological_process_terms=transform(GO_BP_AXES,
           N_annotated_genes=as.numeric(bp_freq[GO_BP_AXES$GO_ID])),
       database_versions=data.frame(
        Database=c("org.Hs.eg.db","GO.db","AnnotationDbi","Bioconductor","R"),
        Version=c(as.character(utils::packageVersion("org.Hs.eg.db")),
                  as.character(utils::packageVersion("GO.db")),
                  as.character(utils::packageVersion("AnnotationDbi")),
                  if (requireNamespace("BiocManager",quietly=TRUE))
                    as.character(BiocManager::version()) else "not installed",
                  R.version.string),
        Retrieved=as.character(Sys.Date()),stringsAsFactors=FALSE))
}

summarize_gene <- function(g) {
  z <- canon[canon$Gene == g, , drop = FALSE]
  v <- data.frame(Gene = g, N_DEG_comparisons = length(unique(z$Comparison)),
                  Max_abs_FC = if (nrow(z)) max(z$Abs_FC) else NA_real_,
                  Min_P_value = if (nrow(z)) min(z$P_value) else NA_real_,
                  Min_FDR_P_value = if (nrow(z)) min(z$FDR_P_value, na.rm = TRUE) else NA_real_,
                  UP_comparisons = paste(z$Comparison[z$Direction == "UP"], collapse = "; "),
                  DOWN_comparisons = paste(z$Comparison[z$Direction == "DOWN"], collapse = "; "),
                  stringsAsFactors = FALSE)
  if (!is.finite(v$Min_FDR_P_value)) v$Min_FDR_P_value <- NA_real_
  v$Direction_summary <- if (!nrow(z)) "Not_DEG" else
    if (all(z$Direction == "UP")) "UP_only_across_comparisons" else
      if (all(z$Direction == "DOWN")) "DOWN_only_across_comparisons" else "UP_and_DOWN_across_comparisons"
  v$Representative_ID <- choose_feature(g)
  # Obtain the gene description from the matched TAC feature.
  ix <- match(v$Representative_ID, ex$ID)
  v$Description <- if (is.na(ix)) "" else as.character(ex$Description[ix])
  v
}

# ---------------------------- Plot Figure S8 ----------------------------
# Figure S8 uses the same heatmap layout conventions as Figure S9:
#   gene | numeric green DEG hits / 6 | one biological-group label per block |
#   slim class-color strip | 6 comparison directions | 6 log2 FC | 4 group Z.
# DEG membership, fold changes and expression summaries come from the supplied
# expression and comparison data, with review flags kept for uncertain classes.
if (MAKE_FIGURES) {
  make_s8 <- function(category, label, prefix) {
    sub <- s8_kept[s8_kept$Figure_Group == category, , drop = FALSE]
    if (!nrow(sub)) return(invisible(NULL))

    # Biological groups form contiguous blocks; within each block, genes
    # are ordered by the number of DEG comparisons.
    sub <- sub[order(sub$Biological_Group, -sub$N_DEG_comparisons, sub$Gene),
               , drop = FALSE]
    row.names(sub) <- NULL
    genes <- sub$Gene
    comps <- comparison_definitions$Comparison
    biological_groups <- as.character(sub$Biological_Group)
    group_levels <- unique(biological_groups)
    row_blocks <- factor(biological_groups, levels = group_levels)

    directions <- matrix(0L, length(genes), length(comps),
                         dimnames = list(genes, comps))
    fcs <- matrix(0, length(genes), length(comps),
                  dimnames = list(genes, comps))
    for (i in seq_len(nrow(canon))) {
      a <- canon[i, ]
      if (a$Gene %in% genes) {
        directions[a$Gene, a$Comparison] <- if (a$Direction == "UP") 1L else -1L
        fcs[a$Gene, a$Comparison] <- a$Signed_log2FC
      }
    }

    # Standardize each representative feature across the 12 samples.
    groups <- c("NC", "Control", "FreeISO", "EXOISO")
    z <- matrix(NA_real_, length(genes), length(groups),
                dimnames = list(genes, groups))
    for (i in seq_along(genes)) {
      id <- sub$Representative_ID[i]
      if (is.na(id) || !id %in% rownames(expr)) next
      a <- as.numeric(expr[id, ])
      if (!is.finite(stats::sd(a, na.rm = TRUE)) ||
          stats::sd(a, na.rm = TRUE) <= 0) next
      a <- as.numeric(scale(a))
      z[i, ] <- c(mean(a[grepl("^NC ", arr)], na.rm = TRUE),
                  mean(a[grepl("^Ctr ", arr)], na.rm = TRUE),
                  mean(a[grepl("^Free ISO ", arr)], na.rm = TRUE),
                  mean(a[grepl("^EXO ISO ", arr)], na.rm = TRUE))
    }

    # S9-style numeric hits: show REAL recurrence counts on colored cells.
    # One distinct DEG comparison is one hit; there are 6 contrasts (0-6).
    hits <- matrix(as.integer(sub$N_DEG_comparisons), ncol = 1L,
                   dimnames = list(genes, "DEG hits / 6"))
    hit_fun <- circlize::colorRamp2(c(0, 3, 6),
                                    c("#f0faed", "#63b86a", "#00551a"))
    h_hits <- ComplexHeatmap::Heatmap(
      hits, name = "DEG hits", col = hit_fun,
      cluster_rows = FALSE, cluster_columns = FALSE,
      row_split = row_blocks, row_gap = grid::unit(1.7, "mm"),
      row_title_side = "left", row_title_rot = 90,
      row_title_gp = grid::gpar(fontsize = 10, fontface = "bold"),
      show_row_names = TRUE, row_names_side = "left",
      row_names_gp = grid::gpar(fontsize = 8, fontface = "bold"),
      column_names_gp = grid::gpar(fontsize = 9),
      width = grid::unit(25, "mm"),
      cell_fun = function(j, i, x, y, width, height, fill) {
        v <- as.integer(hits[i, j])
        grid::grid.text(as.character(v), x, y,
          gp = grid::gpar(fontsize = 8, fontface = "bold",
                          col = if (v >= 5L) "white" else "#182f22"))
      })

    # Flat, light-grey biological-group band EXACTLY as in S9. Only one label
    # at the central gene of each contiguous block, never at every row.
    group_midpoint <- vapply(group_levels, function(g) {
      these <- which(biological_groups == g)
      these[ceiling(length(these) / 2)]
    }, integer(1))
    group_band <- matrix(biological_groups, ncol = 1L,
                         dimnames = list(genes, "Biological group"))
    h_bio <- ComplexHeatmap::Heatmap(
      group_band, name = "Biological group",
      col = stats::setNames(rep("#f5f7f7", length(group_levels)), group_levels),
      show_row_names = FALSE, cluster_rows = FALSE, cluster_columns = FALSE,
      show_heatmap_legend = FALSE, width = grid::unit(64, "mm"),
      column_names_gp = grid::gpar(fontsize = 9),
      cell_fun = function(j, i, x, y, width, height, fill) {
        if (i %in% group_midpoint) {
          grid::grid.text(biological_groups[i], x, y,
            gp = grid::gpar(fontsize = 8, fontface = "bold", col = "#24323b"))
        }
      })

    # A slim colored subclass strip is analogous to S9's narrow Tier track:
    # no duplicated long labels or overly wide colored text columns.
    strip_index <- match(biological_groups, group_levels)
    class_strip <- matrix(as.character(strip_index), ncol = 1L,
                          dimnames = list(genes, "Class"))
    class_colors <- grDevices::hcl.colors(
      length(group_levels), palette = "Set 2", alpha = 1)
    h_class <- ComplexHeatmap::Heatmap(
      class_strip, name = "Class", col = stats::setNames(
        class_colors, as.character(seq_along(group_levels))),
      show_row_names = FALSE, cluster_rows = FALSE, cluster_columns = FALSE,
      show_heatmap_legend = FALSE, width = grid::unit(7, "mm"),
      column_names_gp = grid::gpar(fontsize = 8))

    fc_range <- max(1.5, abs(fcs), na.rm = TRUE)
    fc_fun <- circlize::colorRamp2(c(-fc_range, 0, fc_range),
                                   c("#344aa1", "#ffffff", "#d96a57"))
    z_fun <- circlize::colorRamp2(c(-2, 0, 2),
                                  c("#373399", "#ffffff", "#b12225"))
    h_direction <- ComplexHeatmap::Heatmap(
      directions, name = "Direction",
      col = c("-1" = "#0e247d", "0" = "#ffffff", "1" = "#af272a"),
      show_row_names = FALSE, cluster_rows = FALSE, cluster_columns = FALSE,
      column_names_rot = 90, column_names_gp = grid::gpar(fontsize = 8),
      width = grid::unit(57, "mm"))
    h_fc <- ComplexHeatmap::Heatmap(
      fcs, name = "log2 FC", col = fc_fun,
      show_row_names = FALSE, cluster_rows = FALSE, cluster_columns = FALSE,
      column_names_rot = 90, column_names_gp = grid::gpar(fontsize = 8),
      width = grid::unit(57, "mm"))
    h_z <- ComplexHeatmap::Heatmap(
      z, name = "Group Z", col = z_fun, na_col = "#dddddd",
      show_row_names = FALSE, cluster_rows = FALSE, cluster_columns = FALSE,
      column_names_rot = 90, column_names_gp = grid::gpar(fontsize = 8),
      width = grid::unit(39, "mm"))

    ht <- h_hits + h_bio + h_class + h_direction + h_fc + h_z
    # Scale figure height with the gene count to keep labels readable.
    # The accompanying CSV retains every gene without truncation.
    fig_height <- max(8.0, 0.285 * length(genes) + 3.0)
    fig_width <- 18.6
    for (dev in c("pdf", "png")) {
      filepath <- file.path(plots_dir, paste0(prefix, ".", dev))
      if (dev == "pdf") {
        grDevices::pdf(filepath, width = fig_width, height = fig_height,
                       onefile = FALSE)
      } else {
        grDevices::png(filepath, width = fig_width, height = fig_height,
                        units = "in", res = 125)
      }
      tryCatch({
        ComplexHeatmap::draw(
          ht, merge_legend = TRUE, heatmap_legend_side = "right",
          column_title = paste0(label, " (", length(genes), " genes; ",
            sum(sub$Figure_Group == "Annotation review"), " without documented kinase/phosphatase molecular function)"),
          column_title_gp = grid::gpar(fontsize = 13, fontface = "bold"))
      }, finally = grDevices::dev.off())
    }
    invisible(TRUE)
  }

  make_s8("True kinase enzymes", "GO-annotated kinase enzymes",
          "S8a_kinase_enzymes")
  make_s8("True phosphatases", "GO-annotated phosphatases",
          "S8b_phosphatases")
  make_s8("Regulators / adjacent", "GO-annotated kinase/phosphatase regulators",
          "S8c_regulators_adjacent")
  make_s8("Annotation review", "Annotations without kinase/phosphatase molecular-function support",
          "S8d_annotation_review")
}

# S9 uses the directional KEA3 rankings generated by script 02.
integrated_path <- kea3_integrated_file
libraries_path <- kea3_libraries_file
KEA3_READY <- all(file.exists(c(integrated_path, libraries_path)))
if (!KEA3_READY) {
  message("FC ", FC_THRESHOLD, ": S8 and all 12 directional KEA3 query lists are ready.")
  message("S9 requires the KEA3 results generated by script 02:")
  message(paste(c(integrated_path, libraries_path)[!file.exists(c(integrated_path, libraries_path))],
                collapse = "\n"))
  message("Run script 02 to generate the required 12 directional KEA3 results.")
} else {
  message("Recomputing S9 from KEA3 responses matched to ", run_tag, " ...")
  k <- utils::read.csv(integrated_path, stringsAsFactors = FALSE, check.names = FALSE)
  libs <- utils::read.csv(libraries_path, stringsAsFactors = FALSE, check.names = FALSE)
  if (abs(FC_THRESHOLD - 1.50) > 1e-8) {
    # A matched filename is not enough: require the new query names too.
    expected_names <- unlist(lapply(comparison_definitions$Comparison, function(comp)
      paste0(comp, "_", c("UP", "DOWN"), "_", run_tag)), use.names = FALSE)
    observed_names <- unique(c(as.character(k$Query_Name), as.character(libs$Query_Name)))
    if (!length(observed_names) || any(!observed_names %in% expected_names))
      stop("FC-specific KEA3 query names do not match the current run: ",
           paste(setdiff(observed_names, expected_names), collapse = ", "))
  }
  k$Rank_num <- suppressWarnings(as.numeric(k$Rank))
  libs$Rank_num <- suppressWarnings(as.numeric(libs$Rank))
  methods <- c("MeanRank", "TopRank_integrated_scaled")
  k <- k[!is.na(k$Kinase_symbol) & nzchar(k$Kinase_symbol) &
           k$Integrated_Result_Type %in% methods & is.finite(k$Rank_num), ]
  libs <- libs[!is.na(libs$Kinase_symbol) & nzchar(libs$Kinase_symbol) &
                 is.finite(libs$Rank_num), ]
  # Min rank per query, kinase and integrated ranking method removes any
  # repeated API labels without counting one kinase twice in a signature.
  kr <- aggregate(Rank_num ~ Query_Name + Kinase_symbol + Integrated_Result_Type,
                  data = k, FUN = min)
  libs <- libs[!is.na(libs$Library_clean) & nzchar(libs$Library_clean), ]
  # Saved KEA3 tables use the first 50 records within each ranking
  # method, NOT numeric Rank <= 50. Some integrated lists start at Rank 2
  # or contain ties; their 50th record can consequently have Rank 51.
  # Preserving the original API table order reproduces the exact top-50
  # query membership instead of accidentally adding/removing border genes.
  k$Group_key <- paste(k$Query_Name, k$Integrated_Result_Type, sep = "||")
  top50 <- do.call(rbind, lapply(split(k, k$Group_key), function(z)
    utils::head(z, 50L)))
  # Many KEA3 aliases resolve to one canonical gene (e.g. PKCD and PRKCD).
  # Library support was counted separately for each API kinase label and
  # then maximized over labels/queries. Pooling libraries across aliases
  # would artificially increase support for the canonical gene.
  unique_lib <- unique(libs[c("Query_Name", "Kinase_symbol", "Kinase_API",
                              "Library_clean", "Rank_num")])
  count_lib <- function(threshold) {
    v <- unique_lib[unique_lib$Rank_num <= threshold,
                    c("Query_Name", "Kinase_symbol", "Kinase_API", "Library_clean")]
    v <- unique(v)
    # A canonical symbol can represent more than one KEA3 API label. The
    # The score counts library support only for those exact API
    # kinase labels that themselves occurred in a top-50 integrated result.
    selected_api <- unique(paste(top50$Query_Name, top50$Kinase_symbol,
                                 top50$Kinase_API, sep = "||"))
    v <- v[paste(v$Query_Name, v$Kinase_symbol, v$Kinase_API,
                 sep = "||") %in% selected_api, , drop = FALSE]
    if (!nrow(v)) return(data.frame(Query_Name=character(), Kinase_symbol=character(),
                                    Kinase_API=character(), N=integer()))
    agg <- aggregate(Library_clean ~ Query_Name + Kinase_symbol + Kinase_API,
                     data = v, FUN = length)
    names(agg)[4L] <- "N"
    agg
  }
  lib20 <- count_lib(20)
  lib50 <- count_lib(50)
  # Candidate universe: union of all KEA3 top-50 predictions and every
  # DEG-eligible GO-classified (or explicitly unresolved) signaling candidate.
  candidates <- sort(unique(c(top50$Kinase_symbol, s8_kept$Gene)))
  stopifnot(all(s8_kept$Gene %in% candidates))
  gene_q <- split(canon, canon$Gene)
  per_gene <- split(kr, kr$Kinase_symbol)
  per_top50 <- split(top50, top50$Kinase_symbol)
  per20 <- split(lib20, lib20$Kinase_symbol)
  per50 <- split(lib50, lib50$Kinase_symbol)
  exo_names <- grep("^EXOISO_", comparison_definitions$Comparison, value = TRUE)
  min_finite <- function(x) if (length(x) && any(is.finite(x))) min(x[is.finite(x)]) else NA_real_
  max_finite <- function(x) if (length(x) && any(is.finite(x))) max(x[is.finite(x)]) else 0
  component_top <- function(v) if (is.na(v)) 0 else if (v <= 5) 20 else
    if (v <= 10) 16 else if (v <= 25) 10 else if (v <= 50) 5 else 0
  component_mean <- function(v) if (is.na(v)) 0 else if (v <= 5) 15 else
    if (v <= 10) 12 else if (v <= 25) 8 else if (v <= 50) 4 else 0
  component_deg <- function(group, n) {
    base <- switch(group, "True kinase enzymes" = 15,
                   "True phosphatases" = 10, "Regulators / adjacent" = 8, 0)
    if (base == 0 || n == 0) 0 else base + min(n, 4)
  }
  build_row <- function(g) {
    x <- per_gene[[g]]
    if (is.null(x)) x <- kr[0, ]
    h <- per_top50[[g]]
    if (is.null(h)) h <- top50[0, ]
    qq <- unique(h$Query_Name)
    qe <- grep("^EXOISO_", qq, value = TRUE)
    tr <- x[x$Integrated_Result_Type == "TopRank_integrated_scaled", ]
    mr <- x[x$Integrated_Result_Type == "MeanRank", ]
    # Best-rank evidence is defined WITHIN selected top-50 hits;
    # an unselected rank 63 must not receive a TopRank component.
    bestTR <- min_finite(h$Rank_num[h$Integrated_Result_Type == "TopRank_integrated_scaled"])
    bestMR <- min_finite(h$Rank_num[h$Integrated_Result_Type == "MeanRank"])
    best <- min_finite(c(bestTR, bestMR))
    ll20 <- per20[[g]]
    ll50 <- per50[[g]]
    ml20 <- if (is.null(ll20)) 0L else max_finite(ll20$N)
    ml50 <- if (is.null(ll50)) 0L else max_finite(ll50$N)
    s8info <- s8_kept[s8_kept$Gene == g, ]
    is_screened <- nrow(s8info) > 0L
    # Provisional functional assignments contribute to the exploratory
    # priority score and remain explicitly flagged. Final reported tiers
    # should be interpreted after review of those assignments.
    isdeg <- is_screened &&
      s8info$Figure_Group[1L] %in% c("True kinase enzymes",
        "True phosphatases", "Regulators / adjacent")
    group <- if (is_screened) s8info$Figure_Group[1L] else ""
    ndegs <- if (is_screened) s8info$N_DEG_comparisons[1L] else 0L
    qcontrol <- any(grepl("^EXOISO_vs_Control_", qe))
    qfree <- any(grepl("^EXOISO_vs_FreeISO_", qe))
    qnc <- any(grepl("^EXOISO_vs_NC_", qe))
    mainScore <- if (qcontrol) 10 else if (qfree) 8 else if (qnc) 6 else 0
    recScore <- min(30, 2 * length(qq) + 3 * length(qe))
    scoreTR <- component_top(bestTR)
    scoreMR <- component_mean(bestMR)
    libScore <- min(15, 2 * ml20 + 0.5 * ml50)
    degScore <- component_deg(group, ndegs)
    score <- recScore + scoreTR + scoreMR + libScore + degScore + mainScore
    # Tier 1 requires recurrence, a strong integrated rank and GO-supported
    # DEG evidence. Unresolved annotations receive no enzyme/regulator score.
    tier <- if (isdeg && length(qe) >= 2 && !is.na(best) && best <= 25) 1L else
      if (isdeg && length(qe) >= 1) 2L else
        if (!isdeg && length(qe) >= 2 && !is.na(best) && best <= 10) 3L else
          if (isdeg && length(qq) == 0) 4L else 5L
    tier_label <- c("Tier 1: recurrent KEA3 + DEG kinase/regulator",
                    "Tier 2: KEA3 + DEG kinase/regulator",
                    "Tier 3: strong KEA3-only predicted kinase",
                    "Tier 4: DEG kinase/phosphatase only",
                    "Tier 5: lower-priority supporting candidate")[tier]
    e <- gene_q[[g]]
    e <- if (is.null(e)) canon[0,] else e
    exo_dir <- e[grepl("^EXOISO_", e$Comparison), ]
    pred_UP <- sum(grepl("_UP_", qe))
    pred_DOWN <- sum(grepl("_DOWN_", qe))
    predpattern <- if (!length(qe)) "No EXOISO KEA3 signature" else
      if (pred_UP && pred_DOWN) "Predicted from both EXOISO UP and DOWN signatures" else
        if (pred_UP) "Predicted mainly from EXOISO UP signatures" else
          "Predicted mainly from EXOISO DOWN signatures"
    degpattern <- if (!nrow(exo_dir)) "No EXOISO DEG direction available" else
      if (all(exo_dir$Direction == "UP")) "DEG mainly UP in EXOISO comparisons" else
        if (all(exo_dir$Direction == "DOWN")) "DEG mainly DOWN in EXOISO comparisons" else
          "DEG from both directions in EXOISO comparisons"
    data.frame(Kinase_symbol=g, Priority_tier=tier_label, Tier=tier,
      Priority_score=score, Biological_axis="", Figure_Group=group,
      Is_DEG_kinase_or_regulator=isdeg,
      Is_screened_signaling_DEG=is_screened,
      DEG_annotation_status=if (is_screened) s8info$Review_Status[1L] else "Not screened DEG",
      DEG_support_is_provisional=isdeg &&
        s8info$Figure_Group[1L] == "Annotation review",
      N_queries_top50=length(qq), N_EXOISO_queries_top50=length(qe),
      N_TopRank_queries_top50=length(unique(h$Query_Name[h$Integrated_Result_Type == "TopRank_integrated_scaled"])),
      N_MeanRank_queries_top50=length(unique(h$Query_Name[h$Integrated_Result_Type == "MeanRank"])),
      Best_TopRank_rank=bestTR, Best_MeanRank_rank=bestMR,
      Max_N_libraries_top20=ml20, Max_N_libraries_top50=ml50,
      N_DEG_comparisons=ndegs,
      Max_abs_FC=if (is_screened) s8info$Max_abs_FC[1L] else NA_real_,
      KEA3_recurrence_score=recScore, TopRank_score_component=scoreTR,
      MeanRank_score_component=scoreMR, Library_support_score=libScore,
      DEG_score_component=degScore, Main_EXOISO_score=mainScore,
      KEA3_direction_pattern=predpattern, DEG_EXOISO_direction_pattern=degpattern,
      Queries_EXOISO=paste(sort(qe), collapse="; "),
      Queries_all=paste(sort(qq), collapse="; "),
      EXOISO_vs_Control_state=if (any(grepl("^EXOISO_vs_Control_UP",qe)) &&
             any(grepl("^EXOISO_vs_Control_DOWN",qe))) 2L else
        if (any(grepl("^EXOISO_vs_Control_UP",qe))) 1L else
          if (any(grepl("^EXOISO_vs_Control_DOWN",qe))) -1L else 0L,
      EXOISO_vs_FreeISO_state=if (any(grepl("^EXOISO_vs_FreeISO_UP",qe)) &&
             any(grepl("^EXOISO_vs_FreeISO_DOWN",qe))) 2L else
        if (any(grepl("^EXOISO_vs_FreeISO_UP",qe))) 1L else
          if (any(grepl("^EXOISO_vs_FreeISO_DOWN",qe))) -1L else 0L,
      EXOISO_vs_NC_state=if (any(grepl("^EXOISO_vs_NC_UP",qe)) &&
             any(grepl("^EXOISO_vs_NC_DOWN",qe))) 2L else
        if (any(grepl("^EXOISO_vs_NC_UP",qe))) 1L else
          if (any(grepl("^EXOISO_vs_NC_DOWN",qe))) -1L else 0L,
      stringsAsFactors = FALSE)
  }
  ranked <- do.call(rbind, lapply(candidates, build_row))
  # Link ranked KEA3 candidates to GO biological-process annotations.
  ranked$Biological_axis <- go_ann$Biological_Group[
    match(ranked$Kinase_symbol,go_ann$Gene)]
  ranked$Biological_axis[is.na(ranked$Biological_axis) |
    !nzchar(ranked$Biological_axis)] <- "No selected GO biological process"
  ranked$GO_MF_role <- go_ann$GO_MF_Primary_Role[
    match(ranked$Kinase_symbol,go_ann$Gene)]
  ranked$GO_MF_ID <- go_ann$GO_MF_Primary_ID[
    match(ranked$Kinase_symbol,go_ann$Gene)]
  ranked$GO_BP_ID <- go_ann$GO_BP_Primary_ID[
    match(ranked$Kinase_symbol,go_ann$Gene)]
  ranked$Biological_Group <- ranked$Biological_axis
  ranked <- ranked[order(-ranked$Priority_score, ranked$Kinase_symbol), ]
  row.names(ranked) <- NULL
  write_csv(ranked, "S9_recomputed_all_kinase_priority_scores.csv")
  write_csv(ranked[ranked$Is_screened_signaling_DEG &
                     ranked$Figure_Group == "Annotation review", ],
            "S9_unresolved_GO_annotations.csv")
  if (!all(s8_kept$Gene %in% ranked$Kinase_symbol)) {
    stop("S9 lost one or more screened signaling candidates")
  }
  write_csv(kr, "S9_integrated_KEA3_per_query_ranks.csv")
  write_csv(lib20, "S9_kinase_library_support_top20_per_query.csv")
  write_csv(lib50, "S9_kinase_library_support_top50_per_query.csv")
  write_csv(ranked[seq_len(min(30L,nrow(ranked))), ], "S9_top30_overall_priority.csv")
  write_csv(ranked[ranked$Tier == 1, ], "S9_tier1_recurrent_KEA3_and_DEG.csv")
  main <- ranked[ranked$N_EXOISO_queries_top50 > 0L, ]
  write_csv(head(main, 50L), "S9_EXOISO_focused_top50.csv")

      split_rows <- factor(d$Biological_axis, levels=unique(d$Biological_axis))
      hscore <- ComplexHeatmap::Heatmap(score, name="Priority score", col =
         circlize::colorRamp2(c(0,50,100), c("#f0faed","#55ab4e","#00551a")),
         cluster_rows=FALSE, cluster_columns=FALSE, row_split=split_rows,
         row_gap=grid::unit(1.5,"mm"),
         show_row_names=TRUE, row_names_side="left",
         row_names_gp=grid::gpar(fontsize=8,fontface="bold"),
         width=grid::unit(26,"mm"),
         cell_fun=function(j,i,x,y,width,height,fill) {
           v <- score[i,j]
           grid::grid.text(sprintf("%.1f",v),x,y,
             gp=grid::gpar(fontsize=8,fontface="bold",
                  col=if (v >= 75) "white" else "#1b3527"))
         })
      # KEA3 top-50 hits: each direction-specific query counted once even if
      # a kinase appears in BOTH MeanRank and TopRank (max 12 total, 6 EXO).
      hit_counts <- matrix(d$N_queries_top50,ncol=1L,
                           dimnames=list(gn,"KEA3 hits / 12"))
      hit_exo <- matrix(d$N_EXOISO_queries_top50,ncol=1L,
                        dimnames=list(gn,"EXO hits / 6"))
      deg_counts <- matrix(d$N_DEG_comparisons,ncol=1L,
                           dimnames=list(gn,"DEG hits / 6"))
      numeric_hit_heatmap <- function(mat, title, upper) {
        ComplexHeatmap::Heatmap(mat,name=title,
          col=circlize::colorRamp2(c(0,upper/2,upper),
                                   c("#f0faed","#63b86a","#00551a")),
          show_row_names=FALSE,cluster_rows=FALSE,cluster_columns=FALSE,
          width=grid::unit(20,"mm"),column_names_gp=grid::gpar(fontsize=8),
          cell_fun=function(j,i,x,y,width,height,fill) {
            v <- as.integer(mat[i,j])
            grid::grid.text(as.character(v),x,y,
              gp=grid::gpar(fontsize=8,fontface="bold",
                col=if(v >= upper*0.72) "white" else "#182f22"))
          })
      }
      h_hits <- numeric_hit_heatmap(hit_counts,"KEA3 hits",12)
      h_exo <- numeric_hit_heatmap(hit_exo,"EXO KEA3 hits",6)
      h_deg_hits <- numeric_hit_heatmap(deg_counts,"DEG hits",6)
      htier <- ComplexHeatmap::Heatmap(tier, name="Tier", col=
         c("1"="#6b1d99","2"="#8549b8","3"="#1594d7","4"="#4dac54","5"="#aaaaaa"),
         cluster_rows=FALSE, cluster_columns=FALSE, show_row_names=FALSE,
         width=grid::unit(7,"mm"))
      hcomp <- ComplexHeatmap::Heatmap(ee, name="Relative evidence", col=
         circlize::colorRamp2(c(0,0.5,1),c("#e9f8e7","#6abf6d","#005719")),
         cluster_rows=FALSE, cluster_columns=FALSE, show_row_names=FALSE,
         column_names_rot=90, column_names_gp=grid::gpar(fontsize=8),
         width=grid::unit(61,"mm"))
      hdir <- ComplexHeatmap::Heatmap(sig, name="KEA3 EXOISO", col=
         c("-1"="#001875", "0"="#ffffff", "1"="#b61f24", "2"="#642d87"),
         cluster_rows=FALSE, cluster_columns=FALSE, show_row_names=FALSE,
         column_names_rot=90, column_names_gp=grid::gpar(fontsize=8),
         width=grid::unit(38,"mm"))
      hdeg <- ComplexHeatmap::Heatmap(de, name="DEG EXOISO", col=
         c("-1"="#001875", "0"="#ffffff", "1"="#b61f24", "2"="#642d87"),
         cluster_rows=FALSE, cluster_columns=FALSE, show_row_names=FALSE,
         column_names_rot=90, width=grid::unit(13,"mm"))
      obj <- hscore + h_hits + h_exo + h_deg_hits + hbio + htier + hcomp + hdir + hdeg
      h <- max(8.0, 0.34*nrow(d)+3.0)
      for (kind in c("pdf", "png")) {
        file <- file.path(plots_dir, paste0(prefix,".",kind))
        if (kind=="pdf") grDevices::pdf(file,width=18.6,height=h,onefile=FALSE)
        else grDevices::png(file,width=18.6,height=h,units="in",res=160)
        ComplexHeatmap::draw(obj, merge_legend=TRUE, heatmap_legend_side="right",
          column_title=heading,
          column_title_gp=grid::gpar(fontsize=13,fontface="bold"))
        grDevices::dev.off()
      }
    }
    make_s9(head(ranked,30L), "S9a_top30_overall", "Top 30 overall kinase-priority candidates")
    make_s9(ranked[ranked$Tier==1, ], "S9b_tier1", "Tier 1: recurrent KEA3 and DEG evidence")
    make_s9(head(main,50L), "S9c_EXOISO_focused_top50", "EXO-ISO-focused priority candidates")
  }
}
