# ============================================================================
# AK-FAMILY EXPRESSION AND KEA3 CONVERGENCE — FIGURE 6a–d
# ============================================================================
# This script combines the 12-sample AK-family expression score, the directional
# KEA3 results from Supplementary Figure S9, and the 6 AK-neighbourhood KEA3
# queries generated from the AK-score correlation analysis.
# It also performs the specified limma treatment comparisons for Figure 6d.
#
# HOW TO RUN THIS CODE?
# In R or RStudio, enter: source(file.choose(), encoding = "UTF-8")
# Select this TXT, then choose All Expressions.txt from Input_Source_Files.
# Please run scripts 01, 02, 03 and 04 beforehand. The 8 additional inputs are read
# from the same directory without further file-selection windows.
#
# REQUIRED INPUTS
# All Expressions.txt, KEA3_FC1p30_integrated_all.csv,
# S9_recomputed_all_kinase_priority_scores.csv, and the 6 AKscore KEA3 JSONs.
#
# OUTPUTS
# Figure 6a–d panels, expression/correlation tables, convergence tables,
# statistical contrasts, file checksums and the R session record.
# ============================================================================

run_figure6_fc1p30 <- function(expression_file = NULL,
                               integrated_file = NULL,
                               priority_file = NULL,
                               ak_json_files = NULL,
                               output_dir = NULL) {
  required_packages <- c("jsonlite", "dplyr", "ggplot2", "patchwork", "openxlsx",
                         "ggrepel", "AnnotationDbi", "org.Hs.eg.db", "limma")
  missing_packages <- required_packages[!vapply(required_packages,
                                                 requireNamespace, logical(1), quietly = TRUE)]
  if (length(missing_packages)) {
    stop("Required R packages missing: ", paste(missing_packages, collapse = ", "),
         ". Install CRAN packages with install.packages() and Bioconductor ",
         "packages with BiocManager::install().")
  }
  suppressPackageStartupMessages({
    library(dplyr)
    library(ggplot2)
    library(patchwork)
  })

  # All 9 files are located relative to All Expressions.txt.
  if (is.null(expression_file)) {
    message("Choose All Expressions.txt from Input_Source_Files.")
    expression_file <- normalizePath(file.choose(), winslash="/", mustWork=TRUE)
  }
  expression_file <- normalizePath(expression_file,winslash="/",mustWork=TRUE)
  if (basename(expression_file)!="All Expressions.txt") stop("Choose All Expressions.txt.")
  input_dir <- dirname(expression_file)
  if (is.null(integrated_file))
    integrated_file <- file.path(input_dir,"KEA3_FC1p30_integrated_all.csv")
  if (is.null(priority_file))
    priority_file <- file.path(input_dir,"S9_recomputed_all_kinase_priority_scores.csv")
  ak_queries <- unlist(lapply(c("POS","NEG"), function(d)
    paste0("AKscore_",d,"_top",c(100,150,200))),use.names=FALSE)
  if (is.null(ak_json_files))
    ak_json_files <- setNames(file.path(input_dir,paste0(ak_queries,".json")),ak_queries)
  required_files <- c(expression_file,integrated_file,priority_file,unname(ak_json_files))
  missing_files <- required_files[!file.exists(required_files)]
  if (length(missing_files)) stop("Run scripts 01, 02, 03 and 04 before Figure 6. Missing input files:\n",
                                  paste(missing_files,collapse="\n"))
  integrated_file <- normalizePath(integrated_file,winslash="/",mustWork=TRUE)
  priority_file <- normalizePath(priority_file,winslash="/",mustWork=TRUE)
  if (basename(integrated_file)!="KEA3_FC1p30_integrated_all.csv")
    stop("Expected KEA3_FC1p30_integrated_all.csv")
  if (basename(priority_file)!="S9_recomputed_all_kinase_priority_scores.csv")
    stop("Expected S9_recomputed_all_kinase_priority_scores.csv")
  if (is.null(names(ak_json_files)) ||
      !setequal(names(ak_json_files), ak_queries) ||
      length(ak_json_files) != length(ak_queries)) {
    stop("AK-neighbourhood JSON selections must contain exactly these six named queries: ",
         paste(ak_queries, collapse = ", "))
  }
  ak_json_files <- vapply(ak_json_files[ak_queries], normalizePath,
                          character(1), winslash = "/", mustWork = TRUE)
  if (anyDuplicated(ak_json_files)) stop("Each of the six AK JSON files must be distinct")
  # Validate each KEA3 response using the Query Name stored in its payload.
  for (q in ak_queries) {
    raw <- tryCatch(jsonlite::fromJSON(ak_json_files[[q]], simplifyVector = TRUE),
                    error = function(e) stop("Unreadable AK KEA3 JSON for ", q,
                                              ": ", conditionMessage(e)))
    for (key in c("Integrated--meanRank", "Integrated--topRank")) {
      tab <- raw[[key]]
      if (!is.data.frame(tab) || nrow(tab) < 50L ||
          !"Query Name" %in% names(tab) ||
          !identical(unique(as.character(tab[["Query Name"]])), q)) {
        stop("AK JSON content does NOT match the requested ", q,
             ".\nChosen file: ", ak_json_files[[q]],
             "\nCheck the internal 'Query Name' and select the correct file.")
      }
    }
  }
  message("All 9 required Figure 6 input files have been checked.")

  if (is.null(output_dir))
    output_dir <- file.path(dirname(priority_file), "Figure6_FC1p30_Results")
  dir.create(output_dir, recursive = TRUE, showWarnings = FALSE)
  output_dir <- normalizePath(output_dir, winslash = "/", mustWork = TRUE)
  message("Results directory: ", output_dir)

  save_csv <- function(x, name) {
    utils::write.csv(x, file.path(output_dir, name), row.names = FALSE, na = "")
  }
  get_num <- function(x) suppressWarnings(as.numeric(as.character(x)))
  blank_zero <- function(x) { x[is.na(x)] <- 0L; x }

  # ---------- Twelve-array expression matrix ---------------------
  header_lines <- readLines(expression_file, n = 15L, warn = FALSE, encoding = "UTF-8")
  header_line <- which(grepl("^ID\\t", header_lines))[1]
  if (is.na(header_line)) stop("Cannot find the TAC expression header beginning with ID.")
  tac <- utils::read.delim(expression_file, skip = header_line - 1L,
                           sep = "\t", header = TRUE, quote = "", comment.char = "",
                           check.names = FALSE, stringsAsFactors = FALSE,
                           fileEncoding = "UTF-8")
  if (!"Gene Symbol" %in% names(tac)) stop("TAC file is missing Gene Symbol.")

  samples <- unlist(lapply(list(
    NC = "NC", Ctrl = "Ctr", FreeISO = "Free ISO", EXOISO = "EXO ISO"),
    function(prefix) sprintf("%s %s Avg (log2)", prefix, 1:3)), use.names = FALSE)
  if (!all(samples %in% names(tac))) {
    stop("Missing TAC sample columns: ", paste(setdiff(samples, names(tac)), collapse = ", "))
  }
  expr_features <- as.matrix(tac[, samples, drop = FALSE])
  storage.mode(expr_features) <- "numeric"
  if (any(!is.finite(expr_features))) stop("Non-finite measurements found in TAC expression matrix.")
  colnames(expr_features) <- c(paste0("NC_", 1:3), paste0("Ctrl_", 1:3),
                                paste0("FreeISO_", 1:3), paste0("EXOISO_", 1:3))
  sample_names <- colnames(expr_features)
  group_names <- rep(c("NC", "Ctrl", "FreeISO", "EXOISO"), each = 3L)

  # If a symbol occurs on several TAC features, average its log2 expression
  # values within each sample. Keep only valid annotated gene symbols.
  feature_sums <- new.env(hash = TRUE, parent = emptyenv())
  feature_counts <- new.env(hash = TRUE, parent = emptyenv())
  for (i in seq_len(nrow(tac))) {
    genes <- unique(toupper(trimws(strsplit(as.character(tac[["Gene Symbol"]][i]),
                                          "///|//|;|,")[[1]])))
    genes <- genes[!is.na(genes) & nzchar(genes) & !genes %in% c("---", "NA", "-")]
    for (gene in genes) {
      if (exists(gene, envir = feature_sums, inherits = FALSE)) {
        assign(gene, get(gene, envir = feature_sums, inherits = FALSE) +
               expr_features[i, ], envir = feature_sums)
        assign(gene, get(gene, envir = feature_counts, inherits = FALSE) + 1L,
               envir = feature_counts)
      } else {
        assign(gene, expr_features[i, ], envir = feature_sums)
        assign(gene, 1L, envir = feature_counts)
      }
    }
  }
  all_symbols <- sort(ls(feature_sums))
  expr_gene <- t(vapply(all_symbols, function(g) {
    get(g, envir = feature_sums) / get(g, envir = feature_counts)
  }, numeric(12)))
  rownames(expr_gene) <- all_symbols
  colnames(expr_gene) <- sample_names
  expr_gene <- expr_gene[apply(expr_gene, 1, stats::sd) > 0, , drop = FALSE]

  # 8 AK-family members annotated to GO:0004017 and present on the array.
  ak_genes <- c("AK1", "AK2", "AK3", "AK4", "AK5", "AK7", "AK8", "AK9")
  if (!all(ak_genes %in% rownames(expr_gene))) {
    stop("Missing AK-family genes: ", paste(setdiff(ak_genes, rownames(expr_gene)), collapse = ", "))
  }
  ak_z <- t(scale(t(expr_gene[ak_genes, , drop = FALSE]), center = TRUE, scale = TRUE))
  if (any(!is.finite(ak_z))) stop("Invalid AK-gene Z scores.")
  ak_score <- colMeans(ak_z)
  ak_samples <- data.frame(Sample = sample_names, Group = group_names,
                           AK_GO0004017_score = as.numeric(ak_score))
  ak_means <- ak_samples %>% group_by(Group) %>%
    summarise(N_biological = n(), Mean_AK_family_score = mean(AK_GO0004017_score),
              .groups = "drop")
  ak_means$Group <- factor(ak_means$Group,
                           levels = c("NC", "Ctrl", "FreeISO", "EXOISO"))
  save_csv(ak_samples, "Figure6a_AK_scores_per_biological_sample.csv")
  save_csv(ak_means, "Figure6a_AK_group_means_FC1p30.csv")

  # ---------- Kinase transcript correlations -------------------------------
  # Spearman p uses the standard t approximation with 12 independent arrays,
  # consistent with cor.test(method = "spearman", exact = FALSE). There are
  # 3 biological samples per treatment group. No technical wells enter.
  score_resid <- ak_score - ave(ak_score, group_names)
  all_correlations <- lapply(seq_len(nrow(expr_gene)), function(i) {
    x <- as.numeric(expr_gene[i, ])
    rho <- suppressWarnings(stats::cor(x, ak_score, method = "spearman"))
    p <- if (is.finite(rho)) {
      if (abs(rho) == 1) 0 else {
        t_value <- rho * sqrt(10 / (1 - rho^2))
        2 * stats::pt(-abs(t_value), df = 10)
      }
    } else NA_real_
    group_resid <- x - ave(x, group_names)
    group_r <- suppressWarnings(stats::cor(group_resid, score_resid))
    group_p <- if (is.finite(group_r)) {
      if (abs(group_r) == 1) 0 else {
        2 * stats::pt(-abs(group_r * sqrt(10 / (1 - group_r^2))), df = 10)
      }
    } else NA_real_
    data.frame(Kinase_symbol = rownames(expr_gene)[i],
               Spearman_rho = rho, Spearman_p = p,
               Pearson_r = suppressWarnings(stats::cor(x, ak_score)),
               Group_adjusted_r = group_r, Group_adjusted_p = group_p)
  })
  cor_table <- dplyr::bind_rows(all_correlations)
  cor_table$Spearman_FDR_all_tested_genes <- stats::p.adjust(cor_table$Spearman_p, method = "BH")
  save_csv(cor_table, "Figure6c_all_gene_correlations.csv")

  # Generate the 6 gene lists used for the AK-neighbourhood
  # sensitivity analysis. The saved KEA3 responses are fixed reference data.
  # Tthe lists document the correlation-based gene selection.
  query_output_dir <- file.path(output_dir, "AK_neighbourhood_query_lists")
  dir.create(query_output_dir, recursive = TRUE, showWarnings = FALSE)
  AK_ranked <- cor_table %>%
    filter(is.finite(Spearman_rho),
           !Kinase_symbol %in% c(ak_genes, "CMPK2"))
  save_csv(AK_ranked %>% arrange(desc(Spearman_rho), Kinase_symbol),
           "Figure6_AK_neighbourhood_gene_correlations.csv")
  for (n in c(100L, 150L, 200L)) {
    pos <- AK_ranked %>% arrange(desc(Spearman_rho), Kinase_symbol) %>%
      slice_head(n = n) %>% pull(Kinase_symbol)
    neg <- AK_ranked %>% arrange(Spearman_rho, Kinase_symbol) %>%
      slice_head(n = n) %>% pull(Kinase_symbol)
    writeLines(pos, file.path(query_output_dir, paste0("AKscore_POS_top", n, ".txt")))
    writeLines(neg, file.path(query_output_dir, paste0("AKscore_NEG_top", n, ".txt")))
  }

  # ---------- Read KEA3 integrated rankings -------------------------------
  # KEA3 provides 2 integrated summaries. A kinase enters the top 50 if
  # either method ranks it in its first 50 rows. Each query counts only once.
  # The positional top-50 rule reproduces the S8–S9 integration convention.
  read_integrated <- function(json_file, query_name, canonicalize = FALSE) {
    if (!file.exists(json_file)) {
      stop("Missing KEA3 response: ", json_file)
    }
    if (file.info(json_file)$size < 1000) {
      stop("KEA3 response appears empty or truncated: ", json_file)
    }
    obj <- tryCatch(
      jsonlite::fromJSON(paste(readLines(json_file, warn = FALSE, encoding = "UTF-8"),
                                collapse = "\n"), simplifyVector = TRUE),
      error = function(e) stop("Cannot read KEA3 JSON ", basename(json_file), ": ", conditionMessage(e))
    )
    out <- list()
    for (method in c("Integrated--meanRank", "Integrated--topRank")) {
      x <- obj[[method]]
      if (is.null(x) || !is.data.frame(x) || nrow(x) < 50L) {
        stop("Missing complete KEA3 integrated table in: ", json_file, " / ", method)
      }
      if (!all(c("TF", "Rank") %in% names(x))) stop("Invalid KEA3 fields: ", json_file)
      if ("Query Name" %in% names(x) &&
          any(as.character(x[["Query Name"]]) != query_name)) {
        stop("KEA3 query identity mismatch in ", json_file)
      }
      genes <- toupper(trimws(as.character(x$TF)))
      if (canonicalize) {
        alias <- c(PKCA="PRKCA", PKCB="PRKCB", PKCD="PRKCD", PKCE="PRKCE",
                   PKCG="PRKCG", PKCH="PRKCH", PKCI="PRKCI", PKCQ="PRKCQ", PKCZ="PRKCZ")
        matches <- genes %in% names(alias)
        genes[matches] <- unname(alias[genes[matches]])
      }
      tab <- data.frame(Query_Name = query_name,
                        Integrated_Result_Type = if (method == "Integrated--meanRank")
                          "MeanRank" else "TopRank_integrated_scaled",
                        Kinase_symbol = genes, Kinase_API = as.character(x$TF),
                        Rank = get_num(x$Rank), Position = seq_len(nrow(x)))
      if (any(!is.finite(tab$Rank)) || any(!nzchar(tab$Kinase_symbol))) {
        stop("Invalid ranks/symbols in KEA3 response: ", json_file)
      }
      out[[method]] <- tab
    }
    dplyr::bind_rows(out)
  }

  # AK neighbourhood: 6 query-specific KEA3 responses, validated by query name.
  ak_ranks <- dplyr::bind_rows(lapply(ak_queries, function(q) {
    read_integrated(ak_json_files[[q]], q)
  }))
  # Use first 50 rows in each integrated method (the same rule used in S9).
  ak_ranks$In_top50 <- ak_ranks$Position <= 50L
  boundary_audit <- ak_ranks %>% group_by(Query_Name, Integrated_Result_Type) %>%
    summarise(N_records = n(), N_position_top50 = sum(Position <= 50L),
              N_numeric_rank_le_50 = sum(Rank <= 50), .groups = "drop")
  save_csv(boundary_audit, "Figure6b_AK_KEA3_top50_boundary_audit.csv")
  ak_hit_rows <- ak_ranks[ak_ranks$In_top50, c("Kinase_symbol", "Query_Name"), drop = FALSE]
  ak_hit_rows <- unique(ak_hit_rows)
  ak_hit_rows$Direction <- ifelse(grepl("_POS_", ak_hit_rows$Query_Name), "POS", "NEG")
  ak_counts <- ak_hit_rows %>% group_by(Kinase_symbol) %>% summarise(
    Positive_stability = n_distinct(Query_Name[Direction == "POS"]),
    Negative_stability = n_distinct(Query_Name[Direction == "NEG"]),
    AK_queries = paste(sort(unique(Query_Name)), collapse = "; "),
    .groups = "drop") %>%
    mutate(AK_stability = pmax(Positive_stability, Negative_stability))
  save_csv(ak_ranks, "Figure6b_AK_neighbourhood_all_integrated_ranks.csv")
  save_csv(ak_counts, "Figure6b_AK_neighbourhood_stability.csv")

  # ----------- Read GO-based S8-S9 GLOBAL KEA3 results ---------
  # The S8-S9 workflow supplies 12 FC1p30 KEA3 query results.
  comparisons <- c("Control_vs_NC", "FreeISO_vs_Control", "EXOISO_vs_Control",
                   "EXOISO_vs_FreeISO", "EXOISO_vs_NC", "FreeISO_vs_NC")
  global_queries <- unlist(lapply(comparisons, function(comp) {
    paste0(comp, "_", c("UP", "DOWN"), "_FC1p30")
  }), use.names = FALSE)
  global_ranks <- utils::read.csv(integrated_file, check.names = FALSE,
                                   stringsAsFactors = FALSE)
  expected_columns <- c("Query_Name", "Kinase_symbol", "Kinase_API",
                        "Rank", "Integrated_Result_Type")
  if (!all(expected_columns %in% names(global_ranks))) {
    stop("FC1p30 integrated KEA3 CSV lacks: ",
         paste(setdiff(expected_columns, names(global_ranks)), collapse = ", "))
  }
  allowed_methods <- c("MeanRank", "TopRank_integrated_scaled")
  if (!setequal(unique(global_ranks$Query_Name), global_queries) ||
      !setequal(unique(global_ranks$Integrated_Result_Type), allowed_methods)) {
    stop("The integrated KEA3 table does not describe exactly 12 FC1p30 queries / 2 methods.")
  }
  global_ranks$Kinase_symbol <- toupper(trimws(global_ranks$Kinase_symbol))
  global_ranks$Rank <- get_num(global_ranks$Rank)
  if (any(!is.finite(global_ranks$Rank)) ||
      anyNA(global_ranks$Kinase_symbol) ||
      any(!nzchar(global_ranks$Kinase_symbol))) {
    stop("Missing/invalid gene symbols or ranks in integrated KEA3 file")
  }
  # Original S9 uses top 50 per method in retained API table ORDER,
  # not 'numeric Rank <= 50', which differs for ties and occasional rank gaps.
  global_ranks <- global_ranks %>%
    group_by(Query_Name, Integrated_Result_Type) %>%
    mutate(Position = row_number()) %>% ungroup()
  global_table_audit <- global_ranks %>%
    group_by(Query_Name, Integrated_Result_Type) %>%
    summarise(N_records = n(), Min_rank = min(Rank),
              Top50_last_rank = Rank[50L], .groups = "drop")
  if (nrow(global_table_audit) != 24L || any(global_table_audit$N_records < 50L))
    stop("Some KEA3 global integrated query/method tables contain fewer than 50 rows")
  save_csv(global_table_audit, "Figure6b_FC1p30_global_KEA3_table_audit.csv")
  save_csv(global_ranks, "Figure6b_FC1p30_global_KEA3_integrated_ranks.csv")
  source_file_map <- data.frame(
    Source = c("All Expressions", "Global KEA3 integrated", "GO S9 priority",
               rep("AK-neighbourhood raw KEA3", length(ak_queries))),
    Query_Name = c("12 samples", "12 FC1p30 directional lists", "GO S9 analysis", ak_queries),
    File = c(expression_file, integrated_file, priority_file, unname(ak_json_files)),
    MD5 = unname(tools::md5sum(c(expression_file, integrated_file,
                                  priority_file, unname(ak_json_files)))),
    stringsAsFactors = FALSE)
  save_csv(source_file_map, "Figure6_FC1p30_GO_input_source_audit.csv")

  # 2 KEA3 summaries can identify the same kinase within one query. Do not
  # count it twice. The EXO-ISO component comprises the 3 contrasts with
  # EXOISO in the name, each with separate UP and DOWN directions (6 queries).
  global_hit_rows <- global_ranks %>%
    filter(Position <= 50L) %>% distinct(Kinase_symbol, Query_Name)
  global_counts <- global_hit_rows %>% group_by(Kinase_symbol) %>% summarise(
    N_queries_top50 = n_distinct(Query_Name),
    N_EXOISO_queries_top50 = n_distinct(Query_Name[grepl("^EXOISO_", Query_Name)]),
    Queries_EXOISO = paste(sort(Query_Name[grepl("^EXOISO_", Query_Name)]),
                           collapse = "; "),
    Queries_all = paste(sort(Query_Name), collapse = "; "),
    .groups = "drop")
  save_csv(global_counts, "Figure6b_FC1p30_global_recurrence.csv")

  # The S9 priority table is used only as a numerical consistency
  # check and to supply bubble sizes; Figure 6 candidates are not selected
  # from that table's tier labels or priority threshold.
  priority <- utils::read.csv(priority_file, check.names = FALSE)
  if (!all(c("Kinase_symbol", "N_EXOISO_queries_top50", "Priority_score") %in%
           names(priority))) stop("S9 priority table is missing required columns.")
  priority$Kinase_symbol <- toupper(trimws(as.character(priority$Kinase_symbol)))
  priority$N_EXOISO_queries_top50 <- get_num(priority$N_EXOISO_queries_top50)
  priority$Priority_score <- get_num(priority$Priority_score)
  priority_check <- priority %>%
    left_join(global_counts %>% select(Kinase_symbol, N_EXOISO_queries_top50),
              by = "Kinase_symbol", suffix = c("_S9", "_Figure6")) %>%
    mutate(N_EXOISO_queries_top50_Figure6 =
             blank_zero(N_EXOISO_queries_top50_Figure6))
  if (any(priority_check$N_EXOISO_queries_top50_S9 !=
          priority_check$N_EXOISO_queries_top50_Figure6)) {
    mismatch <- priority_check %>% filter(N_EXOISO_queries_top50_S9 !=
                                          N_EXOISO_queries_top50_Figure6)
    save_csv(mismatch, "Figure6_S9_count_discrepancies.csv")
    stop("Calculated global KEA3 hits disagree with S9. See Figure6_S9_count_discrepancies.csv")
  }

  # ---------- 3 independent converging evidence layers ----------------
  # Global support: kinase in >= 2 of the 6 EXO-ISO directional KEA3 lists.
  # AK support: kinase in top 50 in >= 2 nested neighbourhood sizes on
  #             either the positive or negative AK-associated side.
  # Transcript support: |Spearman rho| >= 0.60 and nominal p < 0.05.
  # These rules are independent of S9 tiering and the new linear FC cutoff.
  all_kinases <- data.frame(Kinase_symbol = sort(unique(c(
    ak_counts$Kinase_symbol, global_counts$Kinase_symbol))))

  # KEA3 sometimes uses older kinase aliases rather than approved gene
  # symbols. Match directly wherever possible; otherwise resolve the label
  # with org.Hs.eg.db. Ambiguous aliases are not assigned a transcript score.
  # This is the alias-resolution rule used in the original R analysis.
  mea <- rownames(expr_gene)
  non_direct <- setdiff(all_kinases$Kinase_symbol, mea)
  annot <- if (length(non_direct)) {
    tryCatch(
      suppressMessages(AnnotationDbi::select(org.Hs.eg.db::org.Hs.eg.db,
        keys = non_direct, keytype = "ALIAS", columns = "SYMBOL")),
      error = function(e) {
        warning("Alias lookup returned no results: ", conditionMessage(e))
        data.frame(ALIAS = character(), SYMBOL = character())
      }
    )
  } else data.frame(ALIAS = character(), SYMBOL = character())

  map_label <- function(label) {
    if (label %in% mea) {
      return(data.frame(Kinase_symbol = label, Expression_symbol = label,
                        Mapping_status = "Direct"))
    }
    candidates <- unique(toupper(na.omit(annot$SYMBOL[annot$ALIAS == label])))
    candidates <- intersect(candidates, mea)
    if (length(candidates) == 1L) {
      return(data.frame(Kinase_symbol = label, Expression_symbol = candidates,
                        Mapping_status = "Alias_to_expression"))
    }
    if (length(candidates) > 1L) {
      return(data.frame(Kinase_symbol = label, Expression_symbol = NA_character_,
                        Mapping_status = paste("Ambiguous", paste(candidates, collapse = ";"))))
    }
    data.frame(Kinase_symbol = label, Expression_symbol = NA_character_,
               Mapping_status = "Unresolved")
  }
  alias_map <- dplyr::bind_rows(lapply(all_kinases$Kinase_symbol, map_label))
  save_csv(alias_map, "Figure6_KEA3_expression_symbol_alias_map.csv")
  cor_link <- cor_table %>% rename(Expression_symbol = Kinase_symbol)

  # S9's GO-derived labels are metadata for the convergence table, not
  # candidate-selection thresholds or evidence of catalytic kinase activity.
  go_columns <- intersect(c("Biological_axis", "GO_MF_role", "GO_MF_ID", "GO_BP_ID"),
                          names(priority))
  priority_go <- priority %>%
    select(Kinase_symbol, all_of(go_columns)) %>%
    rename_with(~paste0("S9_", .x), all_of(go_columns))

  convergence <- all_kinases %>%
    left_join(global_counts, by = "Kinase_symbol") %>%
    left_join(ak_counts, by = "Kinase_symbol") %>%
    left_join(alias_map, by = "Kinase_symbol") %>%
    left_join(cor_link, by = "Expression_symbol") %>%
    left_join(priority %>% select(Kinase_symbol, Priority_score), by = "Kinase_symbol") %>%
    left_join(priority_go, by = "Kinase_symbol") %>%

                    height = height, units = "in", bg = "white")
