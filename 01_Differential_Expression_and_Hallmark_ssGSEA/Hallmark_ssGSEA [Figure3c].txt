# ==============================================================================
# HALLMARK ssGSEA AND GROUP HEATMAP
# ==============================================================================
# This script reproduces the Hallmark pathway analysis and the grouped heatmap
# in Figure 3c of the EXO-ISO study. The analysis uses normalized Clariom S
# Human microarray expression measurements from the same 4 HepG2 experimental groups:
# NC, FFA-treated Ctrl, Free ISO and EXO-ISO (3 biological samples each).
#
# The calculation begins with normalized gene expression, evaluates the 50
# human MSigDB Hallmark gene sets by single-sample GSEA and then averages the
# pathway scores across the 3 samples in each treatment group. To compare
# expression patterns on a common color scale, each Hallmark is standardized
# across the 4 group means. The plotted values are row Z-scores.
#
# HOW TO RUN IT?
# In R or RStudio run:
#     source(file.choose(), encoding = "UTF-8")
# Choose this TXT file. If no suitable input is found near the script or in
# the current working directory, R opens another file-selection window.
# Choose the following original input file:
#   All Expressions.txt               
# To apply the same analysis to another experiment, edit the sample names and
# group memberships in section 3. If beginning with a TAC expression export,
# also update raw_columns in section 4 to match its sample column headings.
# Different feature counts are accepted. The original study contained 21448
# transcript clusters and 19525 unique gene symbols after preparation.
# This script is configured for 4 groups and 50 Hallmark pathways. See
# section 3 for instructions on adapting the group labels or group number.
#
# For the full analysis, GSVA, msigdbr, ComplexHeatmap and circlize are needed.
# You can install missing packages from CRAN/Bioconductor when needed. The
# first run require internet access. 
#
# The script saves outputs beside the selected input folder, under
# Hallmark_ssGSEA_Group_Heatmap_Outputs. 
# Key generated files will be:
#   expression_matrix_for_GSVA.tsv                  (when TAC data are supplied)
#   ssGSEA_Hallmark_scores.csv                      (50 Hallmarks x 12 samples)
#   GSVA_Hallmark_scores.csv                        (same sample-level scores)
#   Hallmark_ssGSEA_grouped_group_means_Ctrl_descending.csv
#   Hallmark_ssGSEA_grouped_row_Zscores_Ctrl_descending.csv
#   Hallmark_ssGSEA_boxed_biological_groups_FINAL_Ctrl_descending.png
#   Hallmark_ssGSEA_boxed_biological_groups_FINAL_Ctrl_descending.pdf
#   sessionInfo_Hallmark_ssGSEA.txt
#
# The 50 Hallmark gene sets are organized into 8 biological-process
# categories following the original MSigDB Hallmark classification
# published by Liberzon et al. (2015).
# Reference:
# Liberzon A. et al. The Molecular Signatures Database (MSigDB)
# Hallmark Gene Set Collection. Cell Systems 1, 417-425 (2015).
# DOI: 10.1016/j.cels.2015.12.004
# I used these categories to solely organize the heatmap for a better reading
# and as you see from the code they do not represent any additional enrichment analyses.
# Within each category, Hallmarks are ordered by descending Ctrl
# Z-score for visualization. This ordering does not alter the calculated ssGSEA scores or their Z-scores.
# ==============================================================================

find_input_file <- function() {
  roots <- c(getwd(), file.path(getwd(), "Input_Source_Files"),
             file.path(getwd(), "data"))
  called_scripts <- vapply(sys.frames(), function(f) {
    if (is.null(f$ofile)) NA_character_ else as.character(f$ofile)[1L]
  }, character(1))
  called_scripts <- called_scripts[!is.na(called_scripts) & nzchar(called_scripts)]
  if (length(called_scripts)) {
    dirs <- unique(dirname(normalizePath(called_scripts, mustWork = FALSE)))
    roots <- c(roots, dirs, file.path(dirs, "Input_Source_Files"),
               file.path(dirs, "data"))
  }
  for (base in unique(roots)) {
    for (name in accepted_inputs) {
      candidate <- file.path(base, name)
      if (file.exists(candidate)) return(normalizePath(candidate, mustWork = TRUE))
    }
  }
  if (!interactive()) {
    stop("Input not found. Place a supported input file beside the code, ",
         "in Input_Source_Files/, or in data/.", call. = FALSE)
  }
  message("Select All Expressions.txt, expression_matrix_for_GSVA.tsv, ",
          "or the saved Hallmark-score CSV from your computer.")
  chosen <- file.choose()
  if (!(basename(chosen) %in% accepted_inputs)) {
    stop("Unexpected input filename: ", basename(chosen),
         ". Please select one of the documented inputs.", call. = FALSE)
  }
  normalizePath(chosen, mustWork = TRUE)
}

# 4. When starting from the TAC export, prepare the expression matrix that
# was used for Hallmark scoring. TAC contains transcript-cluster measurements.
# ssGSEA is applied to a gene-symbol expression matrix. Where an annotation
# contains multiple symbols, take its first symbol. If multiple clusters
# represent that same gene, use the one with the highest average expression
# across the supplied samples, retaining input order in the event of a tie.
# For the present study, this one-feature-per-symbol procedure yielded 19525
# gene symbols from the 21448 Clariom S transcript clusters. Other datasets may
# have different feature counts: these figures are not minimum requirements as I said before.
#
# If using your own TAC export, change raw_columns to the sample expression
# headers printed in your file. Their number and order must match sample_groups
# above. The first listed TAC expression column maps to the first sample name.
# If using a prepared TSV, use GeneSymbol as the first column, then the sample
# names given in sample_groups; raw_columns is not used for that route.
raw_columns <- c(
  "Ctr 1 Avg (log2)", "Ctr 2 Avg (log2)", "Ctr 3 Avg (log2)",
  "EXO ISO 1 Avg (log2)", "EXO ISO 2 Avg (log2)", "EXO ISO 3 Avg (log2)",
  "Free ISO 1 Avg (log2)", "Free ISO 2 Avg (log2)",
  "Free ISO 3 Avg (log2)",
  "NC 1 Avg (log2)", "NC 2 Avg (log2)", "NC 3 Avg (log2)"
)
# The raw TAC headers are needed only when reading the original TAC export.
# Any user starting with prepared expression or saved scores do not need to
# edit this list, even if their experiment contains another number of samples.
if (basename(input_file) == "All Expressions.txt" &&
    length(raw_columns) != length(sample_order)) {
  stop("raw_columns and sample_groups must specify the same number of samples ",
       "when reading a TAC export.", call. = FALSE)
}

read_original_tac <- function(file) {
  lines <- readLines(file, n = 15L, warn = FALSE, encoding = "UTF-8")
  header_line <- grep("^ID\\t", lines)[1L]
  if (is.na(header_line)) stop("Could not locate the ID column in: ", file)
  tab <- read.delim(file, skip = header_line - 1L,
                    header = TRUE, check.names = FALSE,
                    quote = "", comment.char = "", fileEncoding = "UTF-8-BOM")
  if (!all(c("ID", "Gene Symbol", raw_columns) %in% names(tab))) {
    stop("The TAC export is missing expected annotations or sample columns.")
  }
  if (anyNA(tab$ID) || anyDuplicated(tab$ID)) {
    stop("TAC transcript-cluster identifiers are missing or duplicated.")
  }
  tab
}

make_gene_expression_matrix <- function(tac) {
  # Feature counts vary across platforms and studies, so this keep valid identifiers,
  # rather than requiring the 21448 clusters of our manuscript dataset.
  message("TAC transcript clusters read: ", nrow(tac))
  if (nrow(tac) < 1L) stop("No transcript clusters found in the TAC input.")
  symbol <- toupper(trimws(sub(";.*$", "", as.character(tac[["Gene Symbol"]]))))
  value_frame <- tac[, raw_columns, drop = FALSE]
  expr <- as.matrix(value_frame)
  storage.mode(expr) <- "double"
  if (any(!is.finite(expr))) stop("TAC sample expression contains non-finite values.")
  selected <- which(!is.na(symbol) & nzchar(symbol))
  mean_expr <- rowMeans(expr)
  ranked <- selected[order(symbol[selected], -mean_expr[selected], selected)]
  keep <- ranked[!duplicated(symbol[ranked])]
  keep <- keep[order(symbol[keep])]
  mat <- expr[keep, , drop = FALSE]
  rownames(mat) <- symbol[keep]
  colnames(mat) <- sample_order
  if (nrow(mat) < 1L || anyDuplicated(rownames(mat))) {
    stop("Could not prepare a nonempty matrix of unique gene symbols.")
  }
  message("Unique gene symbols retained for ssGSEA: ", nrow(mat))
  mat
}

# 5. Read either the TAC file, the previously prepared expression matrix, or
# the saved sample-level Hallmark scores. Supplying saved scores reproduces
# the plotted figure without claiming that enrichment was recalculated.
if (!score_file_selected) {
  if (basename(input_file) == "All Expressions.txt") {
    tac <- read_original_tac(input_file)
    expression_matrix <- make_gene_expression_matrix(tac)
  } else {
    prepared <- read.delim(input_file, header = TRUE, check.names = FALSE)
    if (!"GeneSymbol" %in% names(prepared)) {
      stop("Prepared expression file must include a GeneSymbol column.")
    }
    if (!setequal(setdiff(names(prepared), "GeneSymbol"), sample_order)) {
      stop("Prepared expression sample columns do not match sample_groups. ",
           "Edit the sample names in section 3 to match this dataset.")
    }
    if (anyNA(prepared$GeneSymbol) || anyDuplicated(prepared$GeneSymbol)) {
      stop("Prepared expression file contains invalid gene symbols.")
    }
    prepared <- prepared[order(prepared$GeneSymbol), , drop = FALSE]
    expression_matrix <- as.matrix(prepared[, sample_order, drop = FALSE])
    storage.mode(expression_matrix) <- "double"
    rownames(expression_matrix) <- prepared$GeneSymbol
    if (nrow(expression_matrix) < 1L) {
      stop("The prepared gene expression matrix is empty.")
    }
    message("Unique gene symbols loaded: ", nrow(expression_matrix))
  }
  expression_matrix <- expression_matrix[, sample_order, drop = FALSE]
  if (any(!is.finite(expression_matrix))) {
    stop("Prepared expression matrix contains non-finite values.")
  }
  write.table(data.frame(GeneSymbol = rownames(expression_matrix),
                         expression_matrix, check.names = FALSE),
              file = file.path(output_dir, "expression_matrix_for_GSVA.tsv"),
              sep = "\t", row.names = FALSE, quote = FALSE)

  # The human Hallmark collection is obtained using the same msigdbr settings
  # as the analysis. Every score is calculated on each sample independently.
  # Package and MSigDB releases can affect the result, so the exact software
  # versions used by this R session are recorded with the outputs.
  hallmark_df <- msigdbr::msigdbr(species = "Homo sapiens", collection = "H")
  hallmark_sets <- split(hallmark_df$gene_symbol, hallmark_df$gs_name)
  if (length(hallmark_sets) != 50L) {
    stop("Expected 50 human Hallmark gene sets; received ",
         length(hallmark_sets), ". Review the MSigDB release.")
  }

  # The minSize/maxSize and normalization parameters are part of the original
  # ssGSEA procedure. 
  param <- GSVA::ssgseaParam(
    exprData = expression_matrix,
    geneSets = hallmark_sets,
    minSize = 10,
    maxSize = 500,
    normalize = TRUE
  )
  scores_mat <- as.matrix(GSVA::gsva(param))
  if (!setequal(colnames(scores_mat), sample_order)) {
    stop("The GSVA output does not retain the expected biological samples.")
  }
  scores_mat <- scores_mat[, sample_order, drop = FALSE]
  if (nrow(scores_mat) != 50L) {
    stop("The scoring step did not return exactly 50 Hallmark pathways.")
  }
  write.csv(scores_mat, file.path(output_dir, "ssGSEA_Hallmark_scores.csv"))
  write.csv(scores_mat, file.path(output_dir, "GSVA_Hallmark_scores.csv"))

  # When the reference score table is present, these lines report any differences
  # that could arise from the local GSVA or MSigDB package release.
  reference_score_file <- file.path(input_dir, "GSVA_Hallmark_scores.csv")
  if (file.exists(reference_score_file)) {
    reference <- as.matrix(read.csv(reference_score_file,
                                    row.names = 1, check.names = FALSE))
    storage.mode(reference) <- "double"
    if (setequal(rownames(reference), rownames(scores_mat)) &&
        setequal(colnames(reference), colnames(scores_mat))) {
      delta <- max(abs(reference[rownames(scores_mat), colnames(scores_mat)] -
                       scores_mat))
      message("Maximum difference against the supplied Hallmark scores: ",
              format(delta, digits = 6))
      if (delta > 1e-8) {
        warning("The recalculated ssGSEA scores differ from the supplied ",
                "reference. Check GSVA and MSigDB versions before ",
                "comparing figures.")
      }
    }
  }
} else {
  # This route is useful if you wishes to recreate the published
  # heatmap using the stored sample-level ssGSEA scores.
  scores_mat <- as.matrix(read.csv(input_file, row.names = 1,
                                   check.names = FALSE))
  storage.mode(scores_mat) <- "double"
}

stopifnot(nrow(scores_mat) == 50L,
          ncol(scores_mat) == length(sample_order),
          !anyDuplicated(rownames(scores_mat)),
          setequal(colnames(scores_mat), sample_order),
          all(is.finite(scores_mat)))
scores_mat <- scores_mat[, sample_order, drop = FALSE]

  write.csv(scores_mat, file.path(output_dir, "GSVA_Hallmark_scores.csv"))
  write.csv(scores_mat, file.path(output_dir, "ssGSEA_Hallmark_scores.csv"))
}

# 6. This calculate one mean score per Hallmark and experimental group. Group means
# are calculated across the biological-array scores assigned to each group,
# not by averaging technical wells or standardizing the individual arrays first.
# The original dataset had 3 biological samples per group, and other replicate
# counts are handled using the supplied sample_groups map.
means <- sapply(group_codes, function(group) {
  group_samples <- names(sample_groups)[sample_groups == group]
  rowMeans(scores_mat[, group_samples, drop = FALSE])
})
rownames(means) <- rownames(scores_mat)
colnames(means) <- group_order
stopifnot(identical(dim(means), c(50L, length(group_order))))

# 7. This standardize each Hallmark across the group means, and makes
# direction and relative group differences visible on a shared color scale.
# Standardization uses R's sample standard deviation (n - 1 in the divisor).
# Values shown in the figure are relative within each Hallmark and absolute
# colors are not directly comparable as enrichment magnitudes across sets.
z_mat <- t(scale(t(means)))
stopifnot(all(is.finite(z_mat)),
          identical(dim(z_mat), c(50L, length(group_order))))

# 9. This was used to retain the 8 biological groups and sort Hallmarks within each one
# by descending Z-score in the designated reference group (Ctrl by default).
# All sample-group columns move together, preserving the numerical scores.
# Neither the underlying pathway scores nor the row Z-scores are recalculated.
ordered_rows <- character()
for (group in category_order) {
  in_group <- rownames(z_mat)[hallmark_category[rownames(z_mat)] == group]
  sorted <- in_group[order(-z_mat[in_group, sorting_reference_group], in_group)]
  ordered_rows <- c(ordered_rows, sorted)

  }

  reference_path <- file.path(input_dir, ref_name)
  if (file.exists(reference_path)) {
    actual_file <- file.path(output_dir, ref_name)
    reference <- as.matrix(read.csv(reference_path, row.names = 1,
                                    check.names = FALSE))
    current <- as.matrix(read.csv(actual_file, row.names = 1,
                                  check.names = FALSE))
    storage.mode(reference) <- "double"
    storage.mode(current) <- "double"
    if (identical(rownames(reference), rownames(current)) &&
        identical(colnames(reference), colnames(current))) {
      difference <- max(abs(reference - current))
      message(ref_name, ": maximum difference = ",
              format(difference, digits = 6))
      if (difference > 1e-8) {
        warning("Generated values do not match the supplied reference: ",
                ref_name)
      }
    } else {
      warning("Reference file differs in row/column labels: ", ref_name)
    }
  }
}

# 11. Draw the 4-column group heatmap. Negative row Z-scores are blue,
# values near zero are pale yellow, and positive values are red. The original
# fixed color scale is used throughout (no automatic per-panel rescaling).
# The 8 biological groups are displayed as separate black-outlined boxes.

# The plot consists of 8 horizontal slices. This used to draw a black border around
# each slice, leaving visible gaps between the biological groups. It uses the
# published image dimensions for PNG and the published page size for PDF.
draw_grouped_heatmap <- function() {
  ComplexHeatmap::draw(heatmap, heatmap_legend_side = "right")
  for (i in seq_along(category_order)) {
    ComplexHeatmap::decorate_heatmap_body("Row Z-score", slice = i, {
      grid::grid.rect(gp = grid::gpar(fill = NA, col = "black", lwd = 1.2))
    })
  }
}
