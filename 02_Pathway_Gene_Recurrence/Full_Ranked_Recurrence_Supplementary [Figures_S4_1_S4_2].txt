# ========================================================================
# Full Ranked Recurrence 
# ========================================================================
#
# WHAT THIS ANALYSIS DOES?
# This script reproduces the full-ranked gene-recurrence maps shown in our Supplementary Figures S4.1 and S4.2.
#
# HOW TO RUN THIS CODE?
# It also uses R or RStudio using source(file.choose(), encoding = "UTF-8").
#
# Figure S4.1 shows the upregulated genes and S4.2 shows the downregulated genes. 
# Unlike the main Figure 4 panels, which display the top 50 genes, these
# supplementary maps include every gene overlapping at least one selected
# Hallmark, GO or KEGG term. Each direction is divided into 2 parts so that
# the individual gene labels and annotation columns remain legible.
#
# To run the analysis in RStudio, enter :
#   source(file.choose(), encoding = "UTF-8")
# Choose this script, then select the same input/All Expressions.txt when asked.
# Running the whole file through source() is important: a pasted console script
# can keep going after an error and leave incomplete output behind.
#
# REQUIRED INPUT FILES 
# The other TAC exports and annotation snapshots are found automatically when
# stored in their respective input/ and reference/ folders:
#   All Expressions.txt
#   All Expressions of Groups.txt
#   EXO-ISO Vs Ctrl ALL.txt
#   GO_2026_07_26_HUMAN_gene_term_membership.tsv.gz
#   MSIGDB_H_CACHE.tsv.gz
#   MSIGDB_KEGG_LEGACY_CACHE.tsv.gz
#
# We rebuild the feature evidence directly from these files, retain genes at
# nominal p < 0.05 and absolute linear fold change >= 1.30 and select GO/KEGG
# terms separately for UP and DOWN using the same rules as Figure 4. The 6
# Hallmark sets for each direction remain fixed. The code then calculates gene
# recurrence across the selected terms and generates the S4 panels.
#
# For continuity with the supplementary figures, each ranked list is
# split in half before Ward.D2 clustering of the genes within each part. The
# heatmaps show annotation hits, individual term memberships, 4-group
# expression Z-scores, and gene symbols. 
#
# These figures show all recurrence-positive genes, not all qualifying DEGs.
# Complete DEG lists are also exported separately so you can inspect
# genes with no membership in the selected terms. Every recurrence-positive
# gene must appear in exactly one part of the appropriate direction.
#
# This run writes its own intermediate tables and plots to Full_Ranked_Recurrence/.
# ---------------------------------------------------------------------------

# TAC exports begin with short metadata lines. Find the table header first,
# then read the data without altering cluster IDs or column names.
read_tac_export <- function(path) {
  lines <- readLines(path, n = 30L, warn = FALSE, encoding = "UTF-8")
  hdr <- which(startsWith(sub("^\ufeff", "", lines), "ID\t"))[1L]
  if (is.na(hdr)) stop("Could not find the TAC 'ID' header in ", basename(path))
  x <- read.delim(path, skip = hdr - 1L, header = TRUE,
                  check.names = FALSE, quote = "", comment.char = "",
                  stringsAsFactors = FALSE, fileEncoding = "UTF-8-BOM")
  if (!"ID" %in% names(x) || anyNA(x$ID) || anyDuplicated(x$ID)) {
    stop("TAC export has missing/duplicated ID values: ", basename(path))
  }
  x
}

as_tac_numeric <- function(x, nm) {
  y <- suppressWarnings(as.numeric(x))
  bad <- !is.na(x) & nzchar(trimws(as.character(x))) & is.na(y)
  if (any(bad)) stop("Non-numeric values in TAC column: ", nm)
  y
}

# If a cache is absent, msigdbr is used to retrieve the collection.
msig_get <- function(collection,subcollection=NULL) {
  f <- file.path(ref_dir,paste0(if(collection=="H")"MSIGDB_H_" else "MSIGDB_KEGG_LEGACY_", "CACHE.tsv.gz"))
  if (file.exists(f)) return(read.delim(gzfile(f),stringsAsFactors=FALSE))
  message("First run: loading MSigDB ",collection,"/",subcollection," via msigdbr")
  out <- tryCatch({
    if (is.null(subcollection)) msigdbr::msigdbr(species="Homo sapiens",collection=collection)
    else msigdbr::msigdbr(species="Homo sapiens",collection=collection,subcollection=subcollection)
  },error=function(ex) stop("Unable to load MSigDB membership. Internet/R package setup may be required: ",conditionMessage(ex)))
  if (!all(c("gs_name","gene_symbol") %in% names(out)) || nrow(out)<10)
    stop("MSigDB ",collection,"/",subcollection," absent. Cannot substitute partial ORA gene lists.")
  sel <- unique(out[,c("gs_name","gene_symbol")]); names(sel)<-c("Term","Gene")
  gz <- gzfile(f,"wt");write.table(sel,gz,sep="\t",row.names=FALSE,quote=TRUE);close(gz)
  sel
}

# Test each term against its full reference membership within the measured gene universe.
make_ora <- function(source,term_meta,groups,targets) {
  output <- vector("list",nrow(term_meta));i<-0L
  for (j in seq_len(nrow(term_meta))) {
    key <- term_meta$Key[j]
    tgenes <- intersect(unique(groups[[key]]),universe)
    K <- length(tgenes);n <- length(targets);g <- intersect(tgenes,targets);k <- length(g)
    if (K<3 || K>N || n<1) next
    i<-i+1L
    output[[i]] <- data.frame(Source=source,ID=term_meta$ID[j],Description=term_meta$Description[j],
      Count=k,Background_count=K,Foreground_count=n,pvalue=phyper(k-1,K,N-K,n,lower.tail=FALSE),
      geneID=paste(g,collapse="/"),stringsAsFactors=FALSE)
  }
  if (i == 0L) return(data.frame(Source = character(), ID = character(),
    Description = character(), Count = integer(), Background_count = integer(),
    Foreground_count = integer(), pvalue = numeric(), geneID = character(),
    p.adjust = numeric(), stringsAsFactors = FALSE))
  out <- do.call(rbind,output[seq_len(i)])
  out$p.adjust <- p.adjust(out$pvalue,method="BH")
  out[order(out$pvalue,-out$Count,out$Description),,drop=FALSE]
}

select_strict <- function(df,source) {
  if (nrow(df)<1) return(df)
  name <- tolower(df$Description)
  keep <- grepl(paste(include[[source]],collapse="|"),name) &
    !grepl(paste(exclude,collapse="|"),name)
  x <- df[keep & df$Count >= MIN_GENES_PER_SELECTED_TERM,,drop=FALSE]
  x <- x[order(x$pvalue,-x$Count,x$ID,x$Description),,drop=FALSE]
  head(x, TOP_TERMS_PER_COLLECTION)
}

# 5. Match the selected terms to their constituent genes
get_msig_genes <- function(groups, meta, query) {
  index <- match(normalize_name(query),normalize_name(meta$Description))
  if (is.na(index)) index <- match(normalize_name(query), normalize_name(meta$ID))
  if(is.na(index)) return(NULL)
  groups[[meta$Key[index]]]
}
make_annotation <- function(selected,genes) {
  rows <- list();i<-0L;miss<-character()
  for (j in seq_len(nrow(selected))) {
    src <- selected$Source[j]; t <- selected$Description[j]; id <- selected$ID[j]
    termgenes <- switch(src,
      "GO:BP"=GO_groups[[paste(src,id,sep="||")]],
      "GO:CC"=GO_groups[[paste(src,id,sep="||")]],
      "GO:MF"=GO_groups[[paste(src,id,sep="||")]],
      "KEGG"=get_msig_genes(K_groups,KM,t),
      "Hallmark"=get_msig_genes(H_groups,HM,t),
      NULL)
    if (is.null(termgenes)) {miss<-c(miss,paste(src,t,id,sep=" | "));next}
    gs <- intersect(intersect(unique(termgenes),genes),universe)
    if (length(gs)>0L) {
      i<-i+1L;rows[[i]]<-data.frame(Gene=gs,Source=src,Term=t,ID=id,
        Term_Label=paste(src,t,sep=": "),stringsAsFactors=FALSE)
    }
  }
  list(data=if(i)unique(do.call(rbind,rows)) else
    data.frame(Gene=character(),Source=character(),Term=character(),ID=character(),Term_Label=character()),
    unmapped=unique(miss))
}

# 6. Calculate full recurrence rankings for each DEG direction
run_direction <- function(dir) {
  genes <- representatives$Gene[representatives$Direction == dir]
  if (!length(genes)) stop("No ",dir," genes pass the DEG criteria.")
  prefix <- file.path(out_dir,paste0("FIG4_",dir))

  # Hallmark uses the defined 6 sets: GO and KEGG are tested against all
  # eligible background genes, separately for UP and DOWN.
  fixed_ids <- FIXED_HALLMARKS[[dir]]
  hm_selected <- HM[match(fixed_ids, HM$ID), , drop = FALSE]
  if (anyNA(hm_selected$ID)) stop("Predefined Hallmark ID missing for ", dir)
  ora <- list()
  for (src in c("Hallmark","GO:BP","KEGG","GO:MF","GO:CC")) {
    rr <- if (src == "Hallmark") {
      make_ora(src, hm_selected, H_groups, genes)
    } else if (src == "KEGG") {
      make_ora(src, KM, K_groups, genes)
    } else {
      make_ora(src, GO_meta[GO_meta$Source == src,,drop=FALSE], GO_groups, genes)
    }
    ora[[src]] <- rr
    write.csv(rr,paste0(prefix,"_",gsub(":","_",src),"_ALL_ORA.csv"),row.names=FALSE)
  }

  # Keep all 6 Hallmarks. For each GO/KEGG collection, choose up to 10
  # qualifying terms using the current direction's enrichment ranking.
  selection <- lapply(names(ora),function(src) {
    tab <- ora[[src]]
    if (src == "Hallmark") {
      tab <- tab[match(fixed_ids, tab$ID), , drop = FALSE]
      if (anyNA(tab$ID) || nrow(tab) != 6L) {
        stop("Unable to retain the six predefined Hallmarks for ", dir)
      }
      tab
    } else {
      select_strict(tab, src)
    }
  })
  names(selection) <- names(ora)
  if (!sum(vapply(selection,nrow,integer(1)))) {
    stop("No metabolism-related terms overlapped ",dir," DEGs.")
  }
  base_terms <- unique(do.call(rbind,lapply(selection,function(z) {
    z[,c("Source","ID","Description"),drop=FALSE]
  })))
  rownames(base_terms) <- NULL

  # Export the term-selection evidence so anyone can trace each chosen
  # annotation to its membership counts and nominal/adjusted p-values.
  selected_evidence <- do.call(rbind, lapply(names(selection), function(src) {
    z <- selection[[src]][,c("Source","ID","Description","Count",
                                "Background_count","Foreground_count",
                                "pvalue","p.adjust"),drop=FALSE]
    z$Selection_rule <- if (src == "Hallmark") {
      "Predefined direction-specific Hallmark set"
    } else {
      "Top metabolism-related ORA term; same procedure in UP and DOWN"
    }
    z
  }))
  write.csv(selected_evidence,paste0(prefix,"_Selected_Term_ORA_Evidence.csv"),row.names=FALSE)
  write.csv(data.frame(Direction=dir,Source=names(selection),
                       Selected_Terms=vapply(selection,nrow,integer(1))),
            paste0(prefix,"_Selected_Term_Counts_By_Collection.csv"),row.names=FALSE)
  write.csv(base_terms,paste0(prefix,"_Selected_Terms.csv"),row.names=FALSE)
  message(dir, " fixed Hallmarks: ", paste(fixed_ids, collapse = ", "))
  for (src in c("GO:BP", "GO:MF", "GO:CC", "KEGG")) {
    message(dir, " selected ", src, " metabolism terms: ", nrow(selection[[src]]))
  }
  # Consider every qualifying gene. 
  ann <- make_annotation(base_terms,genes)
  write.csv(data.frame(Unmapped_term=ann$unmapped),paste0(prefix,"_Unmapped_Terms.csv"),row.names=FALSE)
  if(!nrow(ann$data))stop("No annotations matched for direction: ",dir)
  write.csv(ann$data,paste0(prefix,"_Gene_Term_Membership_LONG.csv"),row.names=FALSE)
  ag <- ann$data
  # A hit is one unique gene-to-term membership.
  split_gene <- split(seq_len(nrow(ag)),ag$Gene)
  summarize <- function(g) {
    z<-ag[split_gene[[g]],,drop=FALSE]
    c(Total_Hits=length(unique(z$Term_Label)),
      Hallmark_Hits=length(unique(z$Term_Label[z$Source=="Hallmark"])),
      GO_BP_Hits=length(unique(z$Term_Label[z$Source=="GO:BP"])),
      KEGG_Hits=length(unique(z$Term_Label[z$Source=="KEGG"])),
      GO_MF_Hits=length(unique(z$Term_Label[z$Source=="GO:MF"])),
      GO_CC_Hits=length(unique(z$Term_Label[z$Source=="GO:CC"])))
  }
  genes_hit <- names(split_gene)
  cnt <- as.data.frame(t(vapply(genes_hit,summarize,numeric(6))),stringsAsFactors=FALSE)
  cnt$Gene<-genes_hit
  cnt<-cnt[order(-cnt$Total_Hits,-cnt$Hallmark_Hits,-cnt$KEGG_Hits,
                 -cnt$GO_BP_Hits,-cnt$GO_MF_Hits,-cnt$GO_CC_Hits,cnt$Gene),,drop=FALSE]
  # Retain the entire recurrence-positive list for S4, not just the first
  # 50 genes displayed in the main Figure 4 panels.
  if (nrow(cnt) < 2L) stop("Fewer than two recurrence-positive genes: ", dir)
  matched_rep <- match(cnt$Gene, representatives$Gene)
  if (anyNA(matched_rep)) stop("Missing representative TAC feature for ", dir)
  cnt$Direction <- dir
  cnt$Clariom_ID <- representatives$ID[matched_rep]
  cnt$log2FC <- representatives$log2FC[matched_rep]
  cnt$p <- representatives$p[matched_rep]
  write.csv(cnt, paste0(prefix,"_ALL_Ranked_Gene_Hits.csv"), row.names=FALSE)
  message(dir, ": ", nrow(cnt), " recurrence-positive genes retained for S4")
  return(cnt)
}

short_term <- function(x) {
  x <- sub("^Hallmark: *", "H: ", x)
  x <- sub("^GO:BP: *", "BP: ", x)
  x <- sub("^GO:MF: *", "MF: ", x)
  x <- sub("^GO:CC: *", "CC: ", x)
  x <- sub("^KEGG: *", "KEGG: ", x)
  gsub("_", " ", x, fixed = TRUE)
}

# Constant annotation columns carry no clustering information. Leave them
# in the plotted matrix while avoiding division by zero during scaling.
scale_columns_safe <- function(x) {
  x <- as.matrix(x)
  storage.mode(x) <- "double"
  avg <- colMeans(x)
  dev <- apply(x, 2, stats::sd)
  dev[!is.finite(dev) | dev == 0] <- 1
  sweep(sweep(x, 2, avg, "-"), 2, dev, "/")
}

# The supplementary layout combines a row dendrogram, 6 hit-count columns,
# the 5 annotation collections and the 4-group expression Z-scores.
# Gene labels sit on the right, as in the S4 figures.
make_s4_plot <- function(panel, direction, part, total_parts, n_total) {
  gene <- as.character(panel$genes$Gene)
  if (!length(gene) || anyNA(gene) || anyDuplicated(gene)) {
    stop("Missing or duplicated gene names in S4 plot input")
  }
  binary <- panel$term_mat
  z <- panel$z
  counts <- panel$counts
  n <- length(gene)
  rownames(counts) <- gene
  rownames(binary) <- gene
  rownames(z) <- gene
  if (CLUSTER_WITHIN_PANEL && n > 2L) {
    # Cluster within this rank-defined part using term membership and group
    # expression. This changes display order, not recurrence rank or part.
    features <- cbind(scale_columns_safe(binary), z)
    hc <- stats::hclust(stats::dist(features, method = "euclidean"),
                        method = "ward.D2")
    row_order <- hc$labels[hc$order]
  } else {
    hc <- FALSE
    row_order <- gene
  }

  stopifnot(setequal(row_order, gene), !anyDuplicated(row_order),
            nrow(binary) == length(gene), nrow(z) == length(gene),
            nrow(counts) == length(gene))
  panel$genes$Clustered_Within_Part_Position <- match(panel$genes$Gene, row_order)
  panel$genes$S4_Part <- part
  panel$genes$Original_Rank <- match(panel$genes$Gene, panel$all_ranked_genes)
  row_file <- file.path(OUTPUT, sprintf("S4_%s_part%d_gene_order.csv", direction, part))
  write.csv(panel$genes[order(panel$genes$Clustered_Within_Part_Position),],
            row_file, row.names = FALSE)

  top_mat <- as.matrix(counts[, count_cols, drop = FALSE])
  storage.mode(top_mat) <- "double"
  if (any(!is.finite(top_mat))) stop("Nonfinite S4 recurrence counts; check generated tables.")
  colnames(top_mat) <- c("Total Hits", "Hallmark Hits", "GO:BP Hits",
                         "KEGG Hits", "GO:MF Hits", "GO:CC Hits")
  # Show the overall recurrence count alongside the 5 collection counts.
  hit_col <- circlize::colorRamp2(c(0, max(1, max(top_mat, na.rm = TRUE))),
                                  c("white", "darkgreen"))
  z_col <- circlize::colorRamp2(c(-2, 0, 2),
                                c("#3E2B83", "#FFFFFF", "#F47766"))
  height_in <- min(MAX_PLOT_HEIGHT, max(MIN_PLOT_HEIGHT, 7 + 0.18 * n))
  heatmap_height <- grid::unit(max(5, min(height_in - 5.5, n * 0.14)), "in")
  base <- ComplexHeatmap::Heatmap(top_mat, name = "Hit counts", col = hit_col,
    cluster_rows = hc, cluster_columns = FALSE, show_row_dend = TRUE,
    row_dend_side = "left", show_row_names = FALSE, show_column_dend = FALSE,
    column_names_side = "bottom", column_names_rot = 90,
    column_names_gp = grid::gpar(fontsize = 7),
    height = heatmap_height, width = grid::unit(6 * 4.1, "mm"),
    show_heatmap_legend = TRUE, heatmap_legend_param = list(title = "Hit counts"),
    use_raster = FALSE)

  ht <- base
  for (src in sources) {
    src_cols <- which(startsWith(colnames(binary), paste0(src, ": ")))
    if (length(src_cols) == 0L) next
    m <- binary[, src_cols, drop = FALSE]
   # Use the same collection colors as Figure 4, white marks no membership
   # in the corresponding selected term.
    m2 <- matrix(ifelse(m > 0, "1", "0"), nrow = nrow(m), ncol = ncol(m),
                 dimnames = dimnames(m))
    colnames(m2) <- make.unique(short_term(colnames(m2)))
    ht <- ht + ComplexHeatmap::Heatmap(m2, name = src,
      col = c("0" = "white", "1" = unname(colors[[src]])),
      cluster_rows = FALSE, cluster_columns = FALSE,
      show_row_names = FALSE, show_column_names = TRUE,
      column_names_side = "bottom", column_names_rot = 90,
      column_names_gp = grid::gpar(fontsize = 7),
      column_names_max_height = grid::unit(100, "mm"),
      width = grid::unit(ncol(m2) * 4.0, "mm"),
      heatmap_legend_param = list(at = "1", labels = "Hit"),
      use_raster = FALSE)
  }
  ht <- ht + ComplexHeatmap::Heatmap(z, name = "Group Z", col = z_col,
    cluster_rows = FALSE, cluster_columns = FALSE,
    show_row_names = TRUE, row_names_side = "right",
    row_names_gp = grid::gpar(fontsize = if (n >= 150) 6.1 else 7),
    row_names_max_width = grid::unit(30, "mm"),
    column_names_side = "bottom", column_names_rot = 90,
    column_names_gp = grid::gpar(fontsize = 8),
    height = heatmap_height, width = grid::unit(4 * 5.0, "mm"),
    heatmap_legend_param = list(at = c(-2, -1, 0, 1, 2)),
    use_raster = FALSE)

  # Set the figure height from the number of rows. This keeps long gene
  # lists readable without changing the annotation or color conventions.
  prefix <- file.path(OUTPUT, sprintf("S4_%s_part%dof%d", direction, part, total_parts))
  render <- function() ComplexHeatmap::draw(ht, merge_legends = FALSE,
    heatmap_legend_side = "right", annotation_legend_side = "right",
    padding = grid::unit(c(5, 5, 5, 5), "mm"))

  grDevices::pdf(paste0(prefix, ".pdf"), width = PLOT_WIDTH, height = height_in,
                 useDingbats = FALSE, onefile = FALSE)
  tryCatch(render(), finally = grDevices::dev.off())
  grDevices::png(paste0(prefix, ".png"), width = PLOT_WIDTH * 190,
                 height = height_in * 190, res = 190, bg = "white")
  tryCatch(render(), finally = grDevices::dev.off())
  # Export TIFF separately for journal submission, alongside PDF and PNG.
  grDevices::tiff(paste0(prefix, ".tiff"), width = PLOT_WIDTH * 180,
                  height = height_in * 180, res = 180,
                  compression = "lzw", bg = "white")
  tryCatch(render(), finally = grDevices::dev.off())
  message(direction, " part ", part, "/", total_parts, ": ", n,
          " genes; saved ", basename(prefix), ".pdf/.png/.tiff")
  data.frame(Direction = direction, Part = part, Total_Parts = total_parts,
             First_Rank = min(panel$genes$Original_Rank),
             Last_Rank = max(panel$genes$Original_Rank),
             Genes = n, Selected_Terms = ncol(binary),
             Annotation_Hits = sum(binary),
             File = basename(prefix), stringsAsFactors = FALSE)
}

prepare_direction <- function(direction) {
  prefix <- file.path(RESULTS, paste0("FIG4_", direction))
  ranked <- read_csv(paste0(prefix, "_ALL_Ranked_Gene_Hits.csv"))
  selected <- read_csv(paste0(prefix, "_Selected_Terms.csv"))
  long <- read_csv(paste0(prefix, "_Gene_Term_Membership_LONG.csv"))
  if (!all(c("Gene", count_cols) %in% names(ranked))) {
    stop(direction, ": unexpected recurrence score columns.")
  }
  if (!all(c("Source", "Description", "ID") %in% names(selected))) {
    stop(direction, ": unexpected selected annotation term columns.")
  }
  if (!all(c("Gene", "Source", "Term", "ID", "Term_Label") %in% names(long))) {
    stop(direction, ": unexpected gene-to-term membership columns.")
  }
  if (anyDuplicated(ranked$Gene) || anyNA(ranked$Gene)) {
    stop(direction, ": gene ranks are not unique.")
  }
  if (!all(long$Gene %in% ranked$Gene) || !setequal(unique(long$Gene), ranked$Gene)) {
    stop(direction, ": ranked genes and membership genes do not match.")
  }
  if (any(ranked$Total_Hits <= 0)) stop(direction, ": ranked table contains zero-hit genes.")

  # Keep the selected terms in Figure 4 collection order for a consistent
  # display across both directions and across the split parts.
  selected <- selected[order(match(selected$Source, sources)), , drop = FALSE]
  term_labels <- paste(selected$Source, selected$Description, sep = ": ")
  if (anyDuplicated(term_labels)) stop(direction, ": duplicate display term labels.")
  if (!all(long$Term_Label %in% term_labels)) {
    stop(direction, ": long memberships contain terms absent from the selected term list.")
  }
  # Rebuild the binary matrix from every ranked gene.
  ranked$Original_Recurrence_Rank <- seq_len(nrow(ranked))
  all_genes <- ranked$Gene
  bin <- matrix(0L, nrow = nrow(ranked), ncol = length(term_labels),
                dimnames = list(all_genes, term_labels))
  ix <- cbind(match(long$Gene, all_genes), match(long$Term_Label, term_labels))
  bin[ix] <- 1L
  col_sums <- cbind(
    Total_Hits = rowSums(bin),
    Hallmark_Hits = rowSums(bin[, startsWith(colnames(bin), "Hallmark: "), drop = FALSE]),
    GO_BP_Hits = rowSums(bin[, startsWith(colnames(bin), "GO:BP: "), drop = FALSE]),
    KEGG_Hits = rowSums(bin[, startsWith(colnames(bin), "KEGG: "), drop = FALSE]),
    GO_MF_Hits = rowSums(bin[, startsWith(colnames(bin), "GO:MF: "), drop = FALSE]),
    GO_CC_Hits = rowSums(bin[, startsWith(colnames(bin), "GO:CC: "), drop = FALSE]))
  # Verify each gene's total and collection-level hit counts against the
  # membership matrix before producing any plot.
  for (nm in count_cols) {
    if (!identical(as.integer(col_sums[, nm]), as.integer(ranked[[nm]]))) {
      stop(direction, ": ", nm, " does not match Figure 4 membership evidence.")
    }
  }

  match_rep <- match(all_genes, reps$Gene)
  if (anyNA(match_rep)) stop(direction, ": missing representative gene features.")
  info <- reps[match_rep, c("Gene", "ID", "log2FC", "p", "Direction"), drop = FALSE]
  if (any(info$Direction != direction)) stop(direction, ": mixed DEG directions.")
  evidence_ix <- match(info$ID, evidence$ID)
  if (anyNA(evidence_ix)) stop(direction, ": missing TAC group expression evidence.")
  expr <- as.matrix(evidence[evidence_ix, means_cols, drop = FALSE])
  storage.mode(expr) <- "numeric"
  if (any(!is.finite(expr))) stop(direction, ": nonfinite group expression.")
  colnames(expr) <- c("NC", "Control", "FreeISO", "EXOISO")
  rownames(expr) <- all_genes
  z <- t(apply(expr, 1, function(x) {
    if (stats::sd(x) == 0) rep(0, length(x))
    else as.numeric((x - mean(x)) / stats::sd(x))
  }))
  dimnames(z) <- dimnames(expr)
  ranked$Clariom_ID <- info$ID
  ranked$log2FC <- info$log2FC
  ranked$nominal_p <- info$p
  ranked$Direction <- info$Direction
  write.csv(ranked, file.path(OUTPUT, paste0("S4_", direction, "_all_recurrence_positive_genes.csv")), row.names = FALSE)
  write.csv(selected, file.path(OUTPUT, paste0("S4_", direction, "_selected_terms.csv")), row.names = FALSE)
  write.csv(cbind(Gene = all_genes, as.data.frame(bin, check.names = FALSE)),
            file.path(OUTPUT, paste0("S4_", direction, "_full_binary_matrix.csv")), row.names = FALSE)
  write.csv(cbind(Gene = all_genes, as.data.frame(z, check.names = FALSE)),
            file.path(OUTPUT, paste0("S4_", direction, "_full_group_Z.csv")), row.names = FALSE)

  # Divide the ranked genes into nearly equal parts first, cluster gene
  # rows only within their assigned part, as in the S4 figures.
  n <- nrow(ranked)
  n_parts <- min(PANELS_PER_DIRECTION, n)
  groups <- split(seq_len(n), cut(seq_len(n), breaks = n_parts, labels = FALSE))
  if (sum(lengths(groups)) != n) stop(direction, ": failed to distribute all genes across parts.")
  list(ranked = ranked, binary = bin, z = z, groups = groups)
}

qc <- list()
for (direction in c("UP", "DOWN")) {
  message("Processing complete ", direction, " recurrence-positive gene list...")
  x <- prepare_direction(direction)
  included <- character()
  for (part in seq_along(x$groups)) {
    ix <- x$groups[[part]]
    part_genes <- x$ranked$Gene[ix]
    if (any(part_genes %in% included)) stop("A recurrence-positive gene was assigned twice.")
    included <- c(included, part_genes)
    sub <- list(genes = x$ranked[ix, , drop = FALSE],
                counts = x$ranked[ix, count_cols, drop = FALSE],
                term_mat = x$binary[ix, , drop = FALSE],
                z = x$z[ix, , drop = FALSE],
                all_ranked_genes = x$ranked$Gene)
    qc[[length(qc) + 1L]] <- make_s4_plot(sub, direction, part,
                                          length(x$groups), nrow(x$ranked))
  }
  if (anyDuplicated(included) || !identical(included, x$ranked$Gene)) {
    stop(direction, ": not every recurrence-positive gene was included exactly once.")
  }
}
