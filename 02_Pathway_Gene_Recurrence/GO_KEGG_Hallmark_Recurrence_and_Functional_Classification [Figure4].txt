# TAC places a few metadata lines before the tabular header. Skip those lines
# while retaining the original column names and transcript-cluster identifiers.
read_tac_export <- function(path) {
  lines <- readLines(path, n = 30L, warn = FALSE, encoding = "UTF-8")
  hdr <- which(startsWith(sub("^\ufeff", "", lines), "ID\t"))[1L]
  if (is.na(hdr)) stop("Could not find the TAC 'ID' header in ", basename(path))
  x <- read.delim(path, skip = hdr - 1L, header = TRUE,
                  check.names = FALSE, quote = "", comment.char = "",
                  stringsAsFactors = FALSE, fileEncoding = "UTF-8-BOM")
  if (!"ID" %in% names(x) || anyNA(x$ID) || anyDuplicated(x$ID)) {
    stop("TAC export has missing/duplicated ID values: ", basename(path))
  }
  x
}

  y <- suppressWarnings(as.numeric(x))
  bad <- !is.na(x) & nzchar(trimws(as.character(x))) & is.na(y)
  if (any(bad)) stop("Non-numeric values in TAC column: ", nm)
  y
}

msig_get <- function(collection,subcollection=NULL) {
  f <- file.path(ref_dir,paste0(if(collection=="H")"MSIGDB_H_" else "MSIGDB_KEGG_LEGACY_", "CACHE.tsv.gz"))
  if (file.exists(f)) return(read.delim(gzfile(f),stringsAsFactors=FALSE))
  message("First run: loading MSigDB ",collection,"/",subcollection," via msigdbr")
  out <- tryCatch({
    if (is.null(subcollection)) msigdbr::msigdbr(species="Homo sapiens",collection=collection)
    else msigdbr::msigdbr(species="Homo sapiens",collection=collection,subcollection=subcollection)
  },error=function(ex) stop("Unable to load MSigDB membership. Internet/R package setup may be required: ",conditionMessage(ex)))
  if (!all(c("gs_name","gene_symbol") %in% names(out)) || nrow(out)<10)
    stop("MSigDB ",collection,"/",subcollection," absent. Cannot substitute partial ORA gene lists.")
  sel <- unique(out[,c("gs_name","gene_symbol")]); names(sel)<-c("Term","Gene")
  gz <- gzfile(f,"wt");write.table(sel,gz,sep="\t",row.names=FALSE,quote=TRUE);close(gz)
  sel
}

# Evaluate each term against its full reference gene set.
make_ora <- function(source,term_meta,groups,targets) {
  output <- vector("list",nrow(term_meta));i<-0L
  for (j in seq_len(nrow(term_meta))) {
    key <- term_meta$Key[j]
    tgenes <- intersect(unique(groups[[key]]),universe)
    K <- length(tgenes);n <- length(targets);g <- intersect(tgenes,targets);k <- length(g)
    if (K<3 || K>N || n<1) next
    i<-i+1L
    output[[i]] <- data.frame(Source=source,ID=term_meta$ID[j],Description=term_meta$Description[j],
      Count=k,Background_count=K,Foreground_count=n,pvalue=phyper(k-1,K,N-K,n,lower.tail=FALSE),
      geneID=paste(g,collapse="/"),stringsAsFactors=FALSE)
  }
  if (i == 0L) return(data.frame(Source = character(), ID = character(),
    Description = character(), Count = integer(), Background_count = integer(),
    Foreground_count = integer(), pvalue = numeric(), geneID = character(),
    p.adjust = numeric(), stringsAsFactors = FALSE))
  out <- do.call(rbind,output[seq_len(i)])
  out$p.adjust <- p.adjust(out$pvalue,method="BH")
  out[order(out$pvalue,-out$Count,out$Description),,drop=FALSE]
}

select_strict <- function(df,source) {
  if (nrow(df)<1) return(df)
  name <- tolower(df$Description)
  keep <- grepl(paste(include[[source]],collapse="|"),name) &
    !grepl(paste(exclude,collapse="|"),name)
  x <- df[keep & df$Count >= MIN_GENES_PER_SELECTED_TERM,,drop=FALSE]
  x <- x[order(x$pvalue,-x$Count,x$ID,x$Description),,drop=FALSE]
  head(x, TOP_TERMS_PER_COLLECTION)
}

# 5. Build gene-to-term membership records
get_msig_genes <- function(groups, meta, query) {
  index <- match(normalize_name(query),normalize_name(meta$Description))
  if (is.na(index)) index <- match(normalize_name(query), normalize_name(meta$ID))
  if(is.na(index)) return(NULL)
  groups[[meta$Key[index]]]
}
make_annotation <- function(selected,genes) {
  rows <- list();i<-0L;miss<-character()
  for (j in seq_len(nrow(selected))) {
    src <- selected$Source[j]; t <- selected$Description[j]; id <- selected$ID[j]
    termgenes <- switch(src,
      "GO:BP"=GO_groups[[paste(src,id,sep="||")]],
      "GO:CC"=GO_groups[[paste(src,id,sep="||")]],
      "GO:MF"=GO_groups[[paste(src,id,sep="||")]],
      "KEGG"=get_msig_genes(K_groups,KM,t),
      "Hallmark"=get_msig_genes(H_groups,HM,t),
      NULL)
    if (is.null(termgenes)) {miss<-c(miss,paste(src,t,id,sep=" | "));next}
    gs <- intersect(intersect(unique(termgenes),genes),universe)
    if (length(gs)>0L) {
      i<-i+1L;rows[[i]]<-data.frame(Gene=gs,Source=src,Term=t,ID=id,
        Term_Label=paste(src,t,sep=": "),stringsAsFactors=FALSE)
    }
  }
  list(data=if(i)unique(do.call(rbind,rows)) else
    data.frame(Gene=character(),Source=character(),Term=character(),ID=character(),Term_Label=character()),
    unmapped=unique(miss))
}

# The 6 Hallmarks are kept for each direction. GO/KEGG terms are selected independently.

# Functional submodules: annotation-supported gene classification
#
# To make the recurrence map more informative, each displayed gene is assigned
# to its best-supported functional module. Classification considers the full
# frozen GO annotations and KEGG membership. More specific activities receive higher rule scores.
# Within each annotation source we use its strongest relevant match, so broad
# or inherited GO terms do not contribute repeatedly to the same module.
# The exported audit records supporting terms and competing classifications.
# Genes without adequate evidence stay in 'Other annotated features'.
#
# Module assignment controls only the descriptive labels and gene arrangement
# in the plot. DEG selection, GO/KEGG term selection, recurrence counts, and
# expression Z-scores are calculated separately.
# Researchers can adapt the documented rules for other biological questions.

MODULE_RULES <- data.frame(Module=character(), Source=character(),
                           Pattern=character(), Weight=numeric(),
                           stringsAsFactors=FALSE)
add_module_rule <- function(module, source, pattern, weight) {
  MODULE_RULES <<- rbind(MODULE_RULES,
    data.frame(Module=module, Source=source, Pattern=pattern,
               Weight=as.numeric(weight), stringsAsFactors=FALSE))
}

# Distinguish nucleotide kinases from general phosphate chemistry and from

# Score each gene using GO and KEGG term descriptions. Within a source
# (GO:BP, GO:MF, GO:CC or KEGG), only the strongest match contributes to
# a module score, related ontology terms should not inflate it.
assign_modules_automatically <- function(gene_vector, go_snapshot, kegg_membership) {
  gene_vector <- unique(toupper(as.character(gene_vector)))
  a <- unique(go_snapshot[go_snapshot$Gene %in% gene_vector,
                          c("Gene", "Source", "Term_ID", "Description")])
  kg <- unique(kegg_membership[
    toupper(kegg_membership$Gene) %in% gene_vector,
    c("Term", "Gene")])
  if (nrow(kg)) {
    a <- rbind(a, data.frame(
      Gene=toupper(kg$Gene), Source="KEGG", Term_ID=kg$Term,
      Description=kg$Term,
      stringsAsFactors=FALSE))
  }
  a$Gene <- toupper(as.character(a$Gene))
  a$Description <- as.character(a$Description)
  a <- unique(a[!is.na(a$Description) & nzchar(a$Description), , drop=FALSE])
  if (!nrow(a)) stop("No GO or KEGG annotations available for the module classifier.")

  hit_list <- vector("list", nrow(MODULE_RULES)); h <- 0L
  for (j in seq_len(nrow(MODULE_RULES))) {
    rule <- MODULE_RULES[j, , drop=FALSE]
    match <- a$Source == rule$Source &
      grepl(rule$Pattern, a$Description, ignore.case=TRUE, perl=TRUE)
    match[is.na(match)] <- FALSE
    if (!any(match)) next
    h <- h + 1L
    x <- a[match, , drop=FALSE]
    x$Module <- rule$Module
    # 'Regulation of X' supports an association with X, but does not mean the
    # gene itself performs X. Give these regulatory GO:BP terms less weight.
    regulatory <- rule$Source == "GO:BP" &
      grepl("^(positive |negative )?regulation of ",
            x$Description, ignore.case=TRUE)
    x$Weight <- ifelse(regulatory, pmin(rule$Weight, 2), rule$Weight)
    x$Rule_Index <- j
    hit_list[[h]] <- x
  }
  matched <- if(h) unique(do.call(rbind, hit_list[seq_len(h)])) else
    data.frame(Gene=character(), Source=character(), Term_ID=character(),
      Description=character(), Module=character(), Weight=numeric(),
      Rule_Index=integer(), stringsAsFactors=FALSE)

  # Add the strongest evidence across the 4 annotation sources.
  per_source <- if(nrow(matched))
    aggregate(Weight ~ Gene + Module + Source, matched, max) else
    data.frame(Gene=character(),Module=character(),Source=character(),Weight=numeric())
  per_module <- if(nrow(per_source))
    aggregate(Weight ~ Gene + Module, per_source, sum) else
    data.frame(Gene=character(),Module=character(),Weight=numeric())
  if(nrow(per_module)) {
    coverage <- aggregate(Source ~ Gene + Module, per_source,
                          function(x) length(unique(x)))
    names(coverage)[names(coverage)=="Source"] <- "Supporting_Sources"
    per_module <- merge(per_module, coverage, by=c("Gene", "Module"), sort=FALSE)
    per_module$Score <- per_module$Weight +
      pmax(0, per_module$Supporting_Sources-1) * 1.0
  }
  result <- vector("list", length(gene_vector))
  for(i in seq_along(gene_vector)) {
    gene <- gene_vector[i]
    z <- per_module[per_module$Gene==gene, , drop=FALSE]
    if(nrow(z)) {
      z$Tie_Order <- match(z$Module, MODULE_ORDER)
      z <- z[order(-z$Score, -z$Supporting_Sources, z$Tie_Order),,drop=FALSE]
      top <- z$Score[1]
      second <- if(nrow(z)>1) z$Score[2] else 0
      margin <- top-second
      # A specific molecular function can justify a module, because cellular location
      # alone is generally not strong enough for a functional assignment.
      chosen <- if(top >= 5) z$Module[1] else "Other annotated features"
      status <- if(chosen=="Other annotated features") "Insufficient specific evidence" else
        if(margin < 2) "Ambiguous: review alternative module" else
        if(z$Supporting_Sources[1]>=2 && top>=10) "Strong" else "Supported"
      winner_ev <- matched[matched$Gene==gene & matched$Module==chosen,,drop=FALSE]
      if(nrow(winner_ev)) {
        winner_ev <- winner_ev[order(-winner_ev$Weight,
                       winner_ev$Source,winner_ev$Description),,drop=FALSE]
        evidence <- paste(head(unique(paste0(winner_ev$Source, ": ",
                                winner_ev$Description)), 6), collapse=" | ")
      } else evidence <- ""
      result[[i]] <- data.frame(Gene=gene,Module=chosen,
        Module_Score=top,Next_Best_Module_Score=second,Score_Margin=margin,
        Module_Confidence=status,
        Module_Competing=if(nrow(z)>1) z$Module[2] else "",
        Evidence_Source_Count=z$Supporting_Sources[1],
        Module_Evidence=evidence,stringsAsFactors=FALSE)
    } else {
      result[[i]] <- data.frame(Gene=gene, Module="Other annotated features",
        Module_Score=0,Next_Best_Module_Score=0,Score_Margin=0,
        Module_Confidence="No matching functional module annotation",
        Module_Competing="",Evidence_Source_Count=0,
        Module_Evidence="",stringsAsFactors=FALSE)
    }
  }
  list(summary=do.call(rbind,result),
       evidence=matched, scores=per_module,
       rule_registry=MODULE_RULES)
}

# 6. Analyze UP and DOWN gene lists independently
run_direction <- function(dir) {
  genes <- representatives$Gene[representatives$Direction == dir]
  if (!length(genes)) stop("No ",dir," genes pass the DEG criteria.")
  prefix <- file.path(out_dir,paste0("FIG4_",dir))

  # Use the predefined Hallmarks for this direction. GO/KEGG ORA is run
  # against the full tested gene universe, with UP and DOWN analyzed separately.
  fixed_ids <- FIXED_HALLMARKS[[dir]]
  hm_selected <- HM[match(fixed_ids, HM$ID), , drop = FALSE]
  if (anyNA(hm_selected$ID)) stop("Predefined Hallmark ID missing for ", dir)
  ora <- list()
  for (src in c("Hallmark","GO:BP","KEGG","GO:MF","GO:CC")) {
    rr <- if (src == "Hallmark") {
      make_ora(src, hm_selected, H_groups, genes)
    } else if (src == "KEGG") {
      make_ora(src, KM, K_groups, genes)
    } else {
      make_ora(src, GO_meta[GO_meta$Source == src,,drop=FALSE], GO_groups, genes)
    }
    ora[[src]] <- rr
    write.csv(rr,paste0(prefix,"_",gsub(":","_",src),"_ALL_ORA.csv"),row.names=FALSE)
  }

  # Retain all 6 Hallmarks whether or not they overlap the DEG list.
  # For GO/KEGG, retain up to 10 eligible terms per collection by ORA rank.
  selection <- lapply(names(ora),function(src) {
    tab <- ora[[src]]
    if (src == "Hallmark") {
      tab <- tab[match(fixed_ids, tab$ID), , drop = FALSE]
      if (anyNA(tab$ID) || nrow(tab) != 6L) {
        stop("Unable to retain the six predefined Hallmarks for ", dir)
      }
      tab
    } else {
      select_strict(tab, src)
    }
  })
  names(selection) <- names(ora)
  if (!sum(vapply(selection,nrow,integer(1)))) {
    stop("No metabolism-related terms overlapped ",dir," DEGs.")
  }
  base_terms <- unique(do.call(rbind,lapply(selection,function(z) {
    z[,c("Source","ID","Description"),drop=FALSE]
  })))
  rownames(base_terms) <- NULL

  # Save the terms with overlap counts and both nominal and adjusted p-values.
  # This table lets readers check the ranking and selection independently.
  selected_evidence <- do.call(rbind, lapply(names(selection), function(src) {
    z <- selection[[src]][,c("Source","ID","Description","Count",
                                "Background_count","Foreground_count",
                                "pvalue","p.adjust"),drop=FALSE]
    z$Selection_rule <- if (src == "Hallmark") {
      "Predefined direction-specific Hallmark set"
    } else {
      "Top metabolism-related ORA term; same procedure in UP and DOWN"
    }
    z
  }))
  write.csv(selected_evidence,paste0(prefix,"_Selected_Term_ORA_Evidence.csv"),row.names=FALSE)
  write.csv(data.frame(Direction=dir,Source=names(selection),
                       Selected_Terms=vapply(selection,nrow,integer(1))),
            paste0(prefix,"_Selected_Term_Counts_By_Collection.csv"),row.names=FALSE)
  write.csv(base_terms,paste0(prefix,"_Selected_Terms.csv"),row.names=FALSE)
  message(dir, " fixed Hallmarks: ", paste(fixed_ids, collapse = ", "))
  for (src in c("GO:BP", "GO:MF", "GO:CC", "KEGG")) {
    message(dir, " selected ", src, " metabolism terms: ", nrow(selection[[src]]))
  }
  # All qualifying genes enter the membership step.
  ann <- make_annotation(base_terms,genes)
  write.csv(data.frame(Unmapped_term=ann$unmapped),paste0(prefix,"_Unmapped_Terms.csv"),row.names=FALSE)
  if(!nrow(ann$data))stop("No annotations matched for direction: ",dir)
  write.csv(ann$data,paste0(prefix,"_Gene_Term_Membership_LONG.csv"),row.names=FALSE)
  ag <- ann$data
  # Count unique source-term memberships for each gene. 
  split_gene <- split(seq_len(nrow(ag)),ag$Gene)
  summarize <- function(g) {
    z<-ag[split_gene[[g]],,drop=FALSE]
    c(Total_Hits=length(unique(z$Term_Label)),
      Hallmark_Hits=length(unique(z$Term_Label[z$Source=="Hallmark"])),
      GO_BP_Hits=length(unique(z$Term_Label[z$Source=="GO:BP"])),
      KEGG_Hits=length(unique(z$Term_Label[z$Source=="KEGG"])),
      GO_MF_Hits=length(unique(z$Term_Label[z$Source=="GO:MF"])),
      GO_CC_Hits=length(unique(z$Term_Label[z$Source=="GO:CC"])))
  }
  genes_hit <- names(split_gene)
  cnt <- as.data.frame(t(vapply(genes_hit,summarize,numeric(6))),stringsAsFactors=FALSE)
  cnt$Gene<-genes_hit
  cnt<-cnt[order(-cnt$Total_Hits,-cnt$Hallmark_Hits,-cnt$KEGG_Hits,
                 -cnt$GO_BP_Hits,-cnt$GO_MF_Hits,-cnt$GO_CC_Hits,cnt$Gene),,drop=FALSE]
  write.csv(cnt,paste0(prefix,"_ALL_Ranked_Gene_Hits.csv"),row.names=FALSE)
  cnt <- head(cnt,50)
  if (nrow(cnt)<2L) stop("Fewer than two annotated genes for ",dir,
                         ". Inspect selected terms and MSigDB downloads.")
  cnt$Direction <- dir
  idx<-match(cnt$Gene,representatives$Gene)
  cnt$Clariom_ID<-representatives$ID[idx]
  cnt$log2FC<-representatives$log2FC[idx]
  cnt$p<-representatives$p[idx]
  write.csv(cnt,paste0(prefix,"_TOP50_Gene_Table.csv"),row.names=FALSE)

  term_names <- paste(base_terms$Source,base_terms$Description,sep=": ")
  hit<-matrix(0L,nrow=nrow(cnt),ncol=length(term_names),dimnames=list(cnt$Gene,make.unique(term_names)))
  for(j in seq_len(nrow(ag))) {
    ri<-match(ag$Gene[j],cnt$Gene);ci<-match(ag$Term_Label[j],term_names)
    if(!is.na(ri)&&!is.na(ci))hit[ri,ci]<-1L
  }
  write.csv(cbind(Gene=rownames(hit),as.data.frame(hit,check.names=FALSE)),
            paste0(prefix,"_TOP50_Term_Binary_Matrix.csv"),row.names=FALSE)
  means<-as.matrix(e[match(cnt$Clariom_ID,e$ID),c("NC_Tac_mean","Ctrl_Tac_mean","Free_Tac_mean","EXO_Tac_mean")])
  storage.mode(means)<-"numeric"
  colnames(means)<-c("NC","Control","FreeISO","EXOISO")
  rownames(means)<-cnt$Gene
  z <- t(apply(means,1,function(x)if(any(!is.finite(x)) || sd(x)==0) rep(0,4) else (x-mean(x))/sd(x)))
  colnames(z)<-colnames(means);rownames(z)<-cnt$Gene
  write.csv(cbind(Gene=rownames(z),as.data.frame(z)),paste0(prefix,"_TOP50_Group_Z.csv"),row.names=FALSE)
  # Classify the top-ranked genes from the full GO/KEGG reference information.
  module_run <- assign_modules_automatically(cnt$Gene, go, kegg)
  ma <- module_run$summary[match(cnt$Gene, module_run$summary$Gene),,drop=FALSE]
  stopifnot(identical(as.character(ma$Gene),as.character(cnt$Gene)))
  cnt$Module <- ma$Module
  cnt$Module_Score <- ma$Module_Score
  cnt$Module_Confidence <- ma$Module_Confidence
  cnt$Module_Competing <- ma$Module_Competing
  cnt$Module_Evidence <- ma$Module_Evidence
  write.csv(module_run$summary,
            paste0(prefix,"_TOP50_Automatic_Module_Assignment_AUDIT.csv"),row.names=FALSE)
  write.csv(module_run$evidence,
            paste0(prefix,"_TOP50_Module_GO_KEGG_Matched_Evidence_LONG.csv"),row.names=FALSE)
  write.csv(module_run$scores,
            paste0(prefix,"_TOP50_All_Candidate_Module_Scores.csv"),row.names=FALSE)
  write.csv(module_run$rule_registry,
            paste0(prefix,"_Functional_Module_Rule_Registry.csv"),row.names=FALSE)
  write.csv(cnt,paste0(prefix,"_TOP50_Gene_Table_With_Modules.csv"),row.names=FALSE)
  message(dir, " automatically classified modules: ",
          paste(sort(unique(cnt$Module)), collapse="; "))
  message(dir, " assignment audit: ",
          paste0(prefix,"_TOP50_Automatic_Module_Assignment_AUDIT.csv"))

  # Plotting uses these same exported tables, so the displayed genes,
  # recurrence hits and 4-group Z-scores can all be checked separately.
  return(cnt)
}

# 8. Draw the 2 Figure 4 recurrence panels
# Read the tables calculated during this run. The layout combines gene names,
# total recurrence bars, hit counts by collection, term membership, functional
# modules and group-expression Z-scores. 

RESULTS <- out_dir
OUTPUT <- file.path(RESULTS,"plots")
dir.create(OUTPUT,recursive=TRUE,showWarnings=FALSE)
read_table <- function(path) {
  if (!file.exists(path)) stop("Missing generated result: ",path)
  read.csv(path,check.names=FALSE,stringsAsFactors=FALSE,fileEncoding="UTF-8-BOM")
}

COL_H <- "#7E57C2"        # Hallmark
COL_BP <- "#F4A261"       # GO biological process
COL_KEGG <- "#4CAF50"     # KEGG
COL_MF <- "#7F7F7F"       # GO molecular function
COL_CC <- "#2A9D8F"       # GO cellular component
COL_BAR <- "darkgreen"    # Total annotation hits per gene

# Both directions use the same functional categories. Their display order
# differs to make the main biological themes easier to follow.
DOWN_MODULE_ORDER <- MODULE_ORDER
UP_MODULE_ORDER <- c(
  "Xenobiotic / retinoid / drug metabolism",
  "Membrane transport / ion channels",
  "Glycan / proteoglycan / ECM remodeling",
  "Phosphoinositide metabolism / PI3K signaling",
  "Lipid / fatty acid / peroxisomal metabolism",
  "Nucleotide kinases", "Nucleotide synthesis / turnover",
  "Phosphate / phosphatase reactions",
  setdiff(MODULE_ORDER, c("Xenobiotic / retinoid / drug metabolism",
     "Membrane transport / ion channels", "Glycan / proteoglycan / ECM remodeling",
     "Phosphoinositide metabolism / PI3K signaling",
     "Lipid / fatty acid / peroxisomal metabolism",
     "Nucleotide kinases", "Nucleotide synthesis / turnover",
     "Phosphate / phosphatase reactions"))
)
standardize_module_label <- function(values, direction) {
  x <- trimws(as.character(values))
  x[is.na(x) | !nzchar(x)] <- "Other annotated features"
  # Keep the computed classifications; no gene-specific relabeling.
  x
}

ordered_modules <- function(values, direction) {
  preferred <- if (direction == "DOWN") DOWN_MODULE_ORDER else UP_MODULE_ORDER
  present <- unique(as.character(values))
  c(preferred[preferred %in% present], sort(setdiff(present, preferred)))
}

# Shorten the displayed term labels only. The full names and membership
# values remain available in the exported CSV files.
short_term_label <- function(label) {
  x <- as.character(label)
  x <- sub("^Hallmark: *", "H: ", x)
  x <- sub("^GO:BP: *", "BP: ", x)
  x <- sub("^GO:MF: *", "MF: ", x)
  x <- sub("^GO:CC: *", "CC: ", x)
  x <- sub("^KEGG: *", "KEGG: ", x)
  x <- gsub("_", " ", x, fixed = TRUE)
  x
}

plot_one_direction <- function(direction) {
  direction <- toupper(direction)
  if (!(direction %in% c("UP", "DOWN"))) stop("Use UP or DOWN")
  prefix <- file.path(RESULTS, paste0("FIG4_", direction))
  gene_file <- paste0(prefix, "_TOP50_Gene_Table_With_Modules.csv")
  matrix_file <- paste0(prefix, "_TOP50_Term_Binary_Matrix.csv")
  z_file <- paste0(prefix, "_TOP50_Group_Z.csv")
  genes <- read_table(gene_file)
  binary <- read_table(matrix_file)
  zdat <- read_table(z_file)

  # Show total recurrence in the bars and collection-specific counts in the
  # small shaded matrix, without duplicating the total there.
  hits_cols <- c("Hallmark_Hits", "GO_BP_Hits", "KEGG_Hits",
                 "GO_MF_Hits", "GO_CC_Hits")
  needed_genes <- c("Gene", "Module", "Total_Hits", hits_cols)
  needed_z <- c("Gene", "NC", "Control", "FreeISO", "EXOISO")
  for (name in setdiff(needed_genes, names(genes)))
    stop(direction, " gene table missing: ", name)
  for (name in setdiff(needed_z, names(zdat)))
    stop(direction, " expression Z-score table missing: ", name)
  if (!("Gene" %in% names(binary))) stop(direction, " term matrix missing Gene column")
  if (anyDuplicated(genes$Gene) || anyDuplicated(binary$Gene) || anyDuplicated(zdat$Gene))
    stop(direction, " has duplicated Gene entries")
  if (!all(genes$Gene %in% binary$Gene) || !all(genes$Gene %in% zdat$Gene))
    stop(direction, " gene IDs cannot be matched across this run's result tables")

  genes$Module <- standardize_module_label(genes$Module, direction)
  genes$Total_Hits <- as.numeric(genes$Total_Hits)
  module_levels <- ordered_modules(genes$Module, direction)
  genes$Module <- factor(genes$Module, levels = module_levels)

  # Within each module, order genes by recurrence and the stated tie-breakers.
  # The same rule is applied to every gene.
  ord <- order(as.integer(genes$Module), -genes$Total_Hits,
               -as.numeric(genes$Hallmark_Hits), -as.numeric(genes$KEGG_Hits),
               genes$Gene)
  genes <- genes[ord, , drop = FALSE]
  rownames(genes) <- NULL
  labels <- genes$Gene
  splits <- droplevels(genes$Module)
  n <- nrow(genes)
  if (n < 2) stop(direction, " has fewer than two genes")

  # A compact green scale summarizes hits within each annotation collection.
  count_mat <- as.matrix(genes[, hits_cols, drop = FALSE])
  storage.mode(count_mat) <- "numeric"
  if (any(!is.finite(count_mat))) stop(direction, ": nonnumeric hit count")
  rownames(count_mat) <- labels
  colnames(count_mat) <- c("Hallmark Hits", "GO:BP Hits", "KEGG Hits",
                           "GO:MF Hits", "GO:CC Hits")

  # The main matrix shows whether each gene belongs to each selected term.
  i_bin <- match(labels, binary$Gene)
  tcols <- setdiff(names(binary), "Gene")
  if (!length(tcols)) stop(direction, " has zero annotation columns")
  term_mat <- as.matrix(binary[i_bin, tcols, drop = FALSE])
  storage.mode(term_mat) <- "numeric"
  if (any(!is.finite(term_mat)) || any(!term_mat %in% c(0, 1)))
    stop(direction, ": unexpected values in the precomputed binary hit matrix")
  rownames(term_mat) <- labels
  colnames(term_mat) <- tcols

  # Add row-standardized TAC group means for NC, Ctrl, free ISO and EXO-ISO.
  i_z <- match(labels, zdat$Gene)
  z <- as.matrix(zdat[i_z, needed_z[-1], drop = FALSE])
  storage.mode(z) <- "numeric"
  rownames(z) <- labels
  if (any(!is.finite(z))) stop(direction, ": nonfinite Z-score")

  # Split the complete heatmap consistently by functional module, so all
  # annotation tracks remain aligned with the same gene rows.
  max_hits <- max(count_mat)
  hit_col <- circlize::colorRamp2(c(0, max(1, max_hits)),
                                  c("white", COL_BAR))
  z_col <- circlize::colorRamp2(c(-2, 0, 2),
                                c("#7735EF", "#FFFFFF", "#F47766"))

  # Gene names, total-hit bars and numeric counts on the left.
  left_anno <- ComplexHeatmap::rowAnnotation(
    Gene = ComplexHeatmap::anno_text(labels, which = "row",
      just = "right", location = grid::unit(1, "npc"),
      gp = grid::gpar(fontsize = 9), width = grid::unit(34, "mm")),
    Total_Hits_bar = ComplexHeatmap::anno_barplot(genes$Total_Hits,
      gp = grid::gpar(fill = COL_BAR, col = COL_BAR),
      baseline = 0, border = FALSE,
      width = grid::unit(26, "mm"),
      axis_param = list(side = "top", at = c(0, max(genes$Total_Hits)),
        labels = c("0", as.character(max(genes$Total_Hits))),
        gp = grid::gpar(fontsize = 7))),
    Total_Hits_n = ComplexHeatmap::anno_text(as.character(genes$Total_Hits),
      which = "row", just = "center", location = grid::unit(0.5, "npc"),
      gp = grid::gpar(fontsize = 8), width = grid::unit(8, "mm")),
    annotation_name_side = "top",
    annotation_name_gp = grid::gpar(fontsize = 9, fontface = "bold"),
    show_annotation_name = c(Gene = FALSE, Total_Hits_bar = TRUE,
                             Total_Hits_n = FALSE)
  )

  common_args <- list(
    cluster_rows = FALSE, cluster_columns = FALSE,
    show_row_dend = FALSE, show_column_dend = FALSE,
    show_row_names = FALSE, border = TRUE,
    row_title = NULL,          # CRITICAL: remove auto vertical module titles
    column_names_side = "bottom",
    column_names_rot = 90,
    column_names_gp = grid::gpar(fontsize = 7.5),
    column_names_max_height = grid::unit(98, "mm"),
    use_raster = FALSE
  )

  # 5-column summary of hits across the annotation collections.
  counts_ht <- do.call(ComplexHeatmap::Heatmap, c(
    list(matrix = count_mat, name = "Hit counts", col = hit_col,
      row_split = splits, cluster_row_slices = FALSE,
      row_gap = grid::unit(1.3, "mm"),
      width = grid::unit(5 * 4.8, "mm")), common_args))

  # Give each functional module a horizontal label beside its group of genes.
  module_anno <- ComplexHeatmap::rowAnnotation(
    Module = ComplexHeatmap::anno_block(
      gp = grid::gpar(fill = "white", col = NA),
      labels = levels(splits),
      labels_gp = grid::gpar(fontsize = 9.2, col = "black"),
      labels_rot = 0),
    width = grid::unit(89, "mm"),
    show_annotation_name = FALSE
  )

  # Plot each collection as a binary membership matrix in its own color.
  make_term_ht <- function(src, col) {
    cols <- which(startsWith(colnames(term_mat), paste0(src, ": ")))
    if (!length(cols)) return(NULL)
    m <- term_mat[, cols, drop = FALSE]
    m_chr <- matrix(ifelse(m > 0, "1", "0"), nrow = nrow(m),
                    ncol = ncol(m), dimnames = dimnames(m))
    colnames(m_chr) <- make.unique(short_term_label(colnames(m_chr)))
    do.call(ComplexHeatmap::Heatmap, c(
      list(matrix = m_chr, name = src, col = c("0" = "white", "1" = col),
        width = grid::unit(ncol(m_chr) * 4.35, "mm"),
        heatmap_legend_param = list(at = c("0", "1"), labels = c("0", "1"))),
      common_args))
  }

  hm_h <- make_term_ht("Hallmark", COL_H)
  hm_bp <- make_term_ht("GO:BP", COL_BP)
  hm_kegg <- make_term_ht("KEGG", COL_KEGG)
  hm_mf <- make_term_ht("GO:MF", COL_MF)
  hm_cc <- make_term_ht("GO:CC", COL_CC)

  # 4-group expression Z-scores on the right.
  expr_ht <- do.call(ComplexHeatmap::Heatmap, c(
    list(matrix = z, name = "Group Z", col = z_col,
      width = grid::unit(4 * 5.0, "mm"),
      heatmap_legend_param = list(at = c(-2, -1, 0, 1, 2))),
    modifyList(common_args, list(column_names_gp = grid::gpar(fontsize = 9)))))

  # Add the collections in a fixed order, skipping any without selected terms.
  ht <- left_anno + counts_ht + module_anno
  for (part in list(hm_h, hm_bp, hm_kegg, hm_mf, hm_cc, expr_ht)) {
    if (!is.null(part)) ht <- ht + part
  }

  draw_once <- function() {
    ComplexHeatmap::draw(ht,
      merge_legends = FALSE,
      heatmap_legend_side = "right",
      annotation_legend_side = "right",
      padding = grid::unit(c(4, 6, 4, 4), "mm"))
  }

  outfile <- file.path(OUTPUT, paste0("Fig4_", direction, "_AutoModules_FC1p30"))
  # Use a wide layout for readable term names. 
  if (.Platform$OS.type == "windows" && nchar(paste0(outfile, ".tiff")) >= 245L) {
    stop("The result path is too long for reliable Windows graphics export: ",
         outfile, "\nMove the extracted package to a shorter location (e.g. C:/EXOISO_F4).",
         call. = FALSE)
  }
  grDevices::pdf(paste0(outfile, ".pdf"), width = 28, height = 16,
                 onefile = FALSE, useDingbats = FALSE)
  tryCatch(draw_once(), finally = grDevices::dev.off())

  grDevices::png(paste0(outfile, ".png"), width = 7200, height = 4200,
                 res = 350, bg = "white")
  tryCatch(draw_once(), finally = grDevices::dev.off())

  grDevices::tiff(paste0(outfile, ".tiff"), width = 7200, height = 4200,
                  res = 300, compression = "lzw", bg = "white")
  tryCatch(draw_once(), finally = grDevices::dev.off())

  # Write a short record of the inputs and selection procedure for each panel.
  writeLines(c(
    paste("Panel:", direction),
    paste("Input gene table:", gene_file),
    paste("Input binary matrix:", matrix_file),
    paste("Input group Z:", z_file),
    "GO and KEGG terms selected independently by the same metabolic/ORA rules.",
    "Six original predefined Hallmark pathways retained in each direction.",
    "No gene-specific exceptions or preselected DOWN GO/KEGG terms used.",
    "Modules assigned from full frozen GO/KEGG annotations, not gene names.",
    "Collection annotations, recurrence bars and group Z values are drawn from this analysis."
  ), file.path(OUTPUT, paste0("FIG4_", direction, "_ANALYSIS_MANIFEST.txt")))
  message("SAVED: ", outfile, ".{pdf,png,tiff}")
  message("First module displayed: ", as.character(genes$Module[1]))
  message("First five genes displayed: ", paste(head(labels, 5), collapse = ", "))
  invisible(outfile)
}
