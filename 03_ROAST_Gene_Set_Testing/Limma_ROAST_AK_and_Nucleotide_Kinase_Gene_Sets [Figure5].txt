# WHAT THE ANALYSIS DOES?
# This script tests the AK and related nucleotide-kinase gene sets and generates the Figure 5 panels.
#
# HOW TO RUN THIS CODE?
# Run in R or RStudio using source(file.choose(), encoding = "UTF-8").
#
# REQUIRED INPUTS AND ANALYTICAL DETAILS
# ========================================================================
#
# Figure 5: adenylate-kinase activity and related nucleotide-kinase gene sets
#
# This script reproduces the microarray-based analyses shown in Figure 5,
# starting with the original Clariom S Human TAC expression export. The 3
# panels examine complementary aspects of the same expression dataset:
#
#   (a) A ROAST test of GO:0004017 across 4 treatment comparisons.
#   (b) The 8 measured adenylate-kinase genes, displayed as group-average
#       Z-scores after standardizing each gene across the 12 samples.
#   (c) A sensitivity analysis of 5 related GO molecular-function sets,
#       including GO:0004017, across the same 4 comparisons.
#
# The publication-style figure uses Ward.D2 clustering for panel (b) and
# horizontal, 2-line comparison labels for panel (c). Those presentation
# choices do not change the expression data, gene sets or statistical tests.
#
# 1 experimental input is required:
#   All Expressions.txt
#       Original normalized log2 measurements for the 12 biological samples.
#
# The 5 published GO:MF membership tables are reconstructed internally from
# fixed gene lists. The script downloads MSigDB 2025.1.Hs
# GO:MF collection. 
# The script reports the outcome of the external MSigDB verification, including
# cases where the public collection does not contain the original term.
# To run in RStudio, enter source(file.choose(), encoding = "UTF-8"), choose
# this script, and select All Expressions.txt if the input picker appears.
# The tests use 9999 rotations and set.statistic = "mean". Primary results
# use BH correction across 4 contrasts; the sensitivity panel uses BH
# correction across all 20 set-by-contrast tests. Rotation-based p-values can
# vary slightly if the original R/limma versions or random seeds differ.

# Look for the original inputs beside the script first. If they have been
# placed elsewhere, R will ask you to select them 
locate_input <- function(filename, preferred = NULL, dirs = character(), explanation = "") {
  candidates <- unique(c(preferred, file.path(dirs, filename)))
  hits <- candidates[!is.na(candidates) & file.exists(candidates)]
  if (length(hits)) {
    out <- normalizePath(hits[1L], winslash = "/", mustWork = TRUE)
    message("Found ", filename, ": ", out)
    return(out)
  }
  if (!interactive()) stop("Missing required input: ", filename, call. = FALSE)
  message("Select ", filename, ". ", explanation)
  out <- normalizePath(file.choose(), winslash = "/", mustWork = TRUE)
  if (basename(out) != filename)
    stop("Selected ", basename(out), "; expected ", filename, call. = FALSE)
  out
}

read_tac <- function(path) {
  lines <- readLines(path, n = 20L, warn = FALSE, encoding = "UTF-8")
  hdr <- which(startsWith(sub("^\ufeff", "", lines), "ID\t"))[1L]
  if (is.na(hdr)) stop("Could not find the TAC table header.")
  dat <- read.delim(path, skip = hdr - 1L, header = TRUE, check.names = FALSE,
                    quote = "", comment.char = "", stringsAsFactors = FALSE,
                    fileEncoding = "UTF-8-BOM")
  if (anyDuplicated(dat$ID) || anyNA(dat$ID)) stop("TAC transcript-cluster IDs are not unique.")
  dat
}

build_original_membership <- function() {
  if (!identical(names(GO_SET_MEMBERS), GENE_SETS$Set_Key) ||
      !identical(names(UNMEASURED_GENES), GENE_SETS$Set_Key))
    stop("The specified GO set names do not match the 5 prespecified tests.")
  ans <- lapply(seq_len(nrow(GENE_SETS)), function(i) {
    key <- GENE_SETS$Set_Key[i]
    genes <- GO_SET_MEMBERS[[key]]
    unmeasured <- UNMEASURED_GENES[[key]]
    if (anyDuplicated(genes) || !all(unmeasured %in% genes))
      stop("Invalid original gene membership in ", key)
    data.frame(Set_Key = key, GO_ID = GENE_SETS$GO_ID[i],
               Display_Name = GENE_SETS$Display_Name[i],
               Gene = genes, Present_on_array = !(genes %in% unmeasured),
               Source = "MSigDB exact GO ID (original Figure 5 reference)",
               stringsAsFactors = FALSE)
  })
  ans <- do.call(rbind, ans)
  rownames(ans) <- NULL
  expected <- c(AK_activity = 8L, Deoxynucleoside_P_kinase = 8L,
                dCMP_kinase = 6L, NDP_kinase = 16L, NMP_kinase = 19L)
  observed <- tapply(ans$Present_on_array, ans$Set_Key, sum)
  if (!identical(as.integer(observed[names(expected)]), unname(expected)))
    stop("Original array-coverage counts are inconsistent with Figure 5.")
  ans
}
check_msigdb_reference <- function(original, audit_dir) {
  file <- file.path(audit_dir, paste0("MSigDB_GO_MF_", MSIGDB_RELEASE, ".gmt"))
  if (!file.exists(file) || file.info(file)$size < 1000) {
    if (file.exists(file)) unlink(file)
    message("Checking official MSigDB ", MSIGDB_RELEASE, " online ...")
    ok <- tryCatch({
      old_timeout <- getOption("timeout")
      options(timeout = max(180L, old_timeout))
      on.exit(options(timeout = old_timeout), add = TRUE)
      suppressWarnings(utils::download.file(MSIGDB_GMT_URL, file,
                                            mode = "wb", quiet = TRUE))
      file.exists(file) && file.info(file)$size > 1000
    }, error = function(e) {
      message("MSigDB download: ", conditionMessage(e))
      FALSE
    })
  } else {
    ok <- TRUE
    message("Using previously downloaded MSigDB ", MSIGDB_RELEASE, " cache.")
  }
  if (!ok) {
    if (file.exists(file)) unlink(file)
    status <- data.frame(Set_Key = GENE_SETS$Set_Key,
                         GO_ID = GENE_SETS$GO_ID,
                         Public_Set_Name = unname(MSIGDB_MATCHED_SET_NAMES),
                         Online_Status = "Not checked: reference unavailable",
                         Membership_Changed = NA,
                         stringsAsFactors = FALSE)
    utils::write.csv(status, file.path(audit_dir,"MSigDB_Online_Membership_Audit.csv"),row.names=FALSE)
    writeLines(c(paste0("Reference: ",MSIGDB_GMT_URL),
                 "Online retrieval failed; the specified membership was used unchanged."),
               file.path(audit_dir,"MSigDB_Online_Check_Provenance.txt"))
    warning("Online MSigDB check unavailable. The Figure 5 memberships remain unchanged.")
    return(invisible(status))
  }
  lines <- readLines(file, warn = FALSE, encoding = "UTF-8")
  rows <- strsplit(lines, "\t", fixed = FALSE)
  nm <- vapply(rows, function(z) if (length(z)) z[1L] else "", character(1))
  if (length(lines) < 1000 || anyDuplicated(nm))
    stop("The downloaded official GO:MF file is incomplete or contains duplicate set names.")
  status <- lapply(seq_len(nrow(GENE_SETS)), function(i) {
    key <- GENE_SETS$Set_Key[i]
    target <- unname(MSIGDB_MATCHED_SET_NAMES[key])
    if (is.na(target)) {
      state <- "No verified matching set name for this specified GO ID"
      changed <- NA
    } else if (!(target %in% nm)) {
      state <- "Expected official set name absent in pinned release"
      changed <- NA
    } else {
      public <- unique(toupper(rows[[match(target, nm)]][-c(1L,2L)]))
      saved <- unique(toupper(original$Gene[original$Set_Key == key]))
      changed <- !setequal(public, saved)
      state <- if (changed) "DIFFERENT MEMBERSHIP: published set preserved" else "Exact agreement"
    }
    data.frame(Set_Key=key, GO_ID=GENE_SETS$GO_ID[i],
               Public_Set_Name=if (is.na(target)) "" else target,
               Online_Status=state, Membership_Changed=changed,
               stringsAsFactors=FALSE)
  })
  status <- do.call(rbind,status)
  utils::write.csv(status, file.path(audit_dir,"MSigDB_Online_Membership_Audit.csv"),row.names=FALSE)
  writeLines(c(paste0("MSigDB fixed release: ",MSIGDB_RELEASE),
               paste0("Source: ",MSIGDB_GMT_URL),
               paste0("MD5: ",unname(tools::md5sum(file))),
               "Gene sets used by ROAST are defined in this script.",
               "An online difference is flagged in the audit, never substituted silently."),
             file.path(audit_dir,"MSigDB_Online_Check_Provenance.txt"))
  if (any(status$Membership_Changed %in% TRUE,na.rm=TRUE))
    warning("Public MSigDB gene memberships differ from the Figure 5 reference; see online audit.")
  invisible(status)
}

# Connect the GO memberships to TAC transcript clusters. Prefer an exact
# gene-symbol match. Some TAC labels contain more than one symbol, in that
# situation a cluster is assigned only when the frozen reference identifies
# a single eligible set member. In particular, the shared NME1/NME2 label
# must not be counted as 2 independent expression measurements. Every
# choice is recorded in the mapping-audit table.
valid_genes <- unique(sets$Gene[sets$Present_on_array])
raw_labels <- toupper(trimws(as.character(tac[["Gene Symbol"]])))
resolve_symbol <- function(label) {
  if (is.na(label) || !nzchar(label)) return(c("", "Missing symbol"))
  if (!grepl(";", label, fixed = TRUE)) return(c(label, "Exact TAC symbol"))
  aliases <- unique(trimws(strsplit(label, ";", fixed = TRUE)[[1L]]))
  matches <- intersect(aliases, valid_genes)
  if (length(matches) == 1L) return(c(matches, "Shared TAC annotation: unique represented set gene"))
  c("", if (length(matches)) "Ambiguous multiple represented aliases" else "Ambiguous unmapped aliases")
}
resolved <- t(vapply(raw_labels, resolve_symbol, character(2)))

# Fit the 4 groups using the original biological samples as independent
# observations. ROAST tests the predefined GO memberships directly and genes
# are not removed because their individual fold changes or p-values are small.
meta_group <- factor(sample_info$Group, levels = c("NC", "Ctrl", "FreeISO", "EXOISO"))
design <- stats::model.matrix(~ 0 + meta_group)
colnames(design) <- levels(meta_group)
if (qr(design)$rank != ncol(design)) stop("Non-full-rank design.")
contrast_vector <- function(i) {
  v <- stats::setNames(rep(0, ncol(design)), colnames(design))
  v[CONTRASTS$Numerator[i]] <- 1
  v[CONTRASTS$Denominator[i]] <- -1
  v
}
write.csv(design, file.path(tables_dir, "06_Limma_Design_Matrix.csv"), row.names = TRUE)
write.csv(CONTRASTS, file.path(tables_dir, "07_Contrast_Definitions.csv"), row.names = FALSE)

# Figure 5a asks whether the measured adenylate-kinase family shows a
# coordinated response in each comparison. ROAST reports Up, Down, mixed,
# and 2-sided probabilities. The plotted p-value comes from the 2-sided
# UpOrDown test.
primary <- vector("list", nrow(CONTRASTS))
for (i in seq_len(nrow(CONTRASTS))) {
  set.seed(SEED_PRIMARY + i)
  rr <- limma::roast(expr, index = set_indices[["AK_activity"]],
                     design = design, contrast = contrast_vector(i),
                     nrot = N_ROTATIONS, set.statistic = SET_STATISTIC)
  tab <- rr$p.value
  if (is.null(tab) || !all(c("Down", "Up", "UpOrDown", "Mixed") %in% rownames(tab)) ||
      !all(c("Active.Prop", "P.Value") %in% names(tab)))
    stop("Unexpected ROAST output format; inspect package version.")
  direction <- if (tab["Down", "P.Value"] < tab["Up", "P.Value"]) "Down" else "Up"
  main_row <- if (direction == "Down") "Down" else "Up"
  primary[[i]] <- data.frame(
    Contrast = CONTRASTS$Contrast[i], Contrast_Label = CONTRASTS$Contrast_Label[i],
    NGenes = length(set_indices[["AK_activity"]]), Direction = direction,
    Active_Proportion = as.numeric(tab[main_row, "Active.Prop"]),
    P_Down = as.numeric(tab["Down", "P.Value"]),
    P_Up = as.numeric(tab["Up", "P.Value"]),
    P_TwoSided = as.numeric(tab["UpOrDown", "P.Value"]),
    P_Mixed = as.numeric(tab["Mixed", "P.Value"]), stringsAsFactors = FALSE)
  message("Primary ROAST ", CONTRASTS$Contrast_Label[i], ": p = ",
          signif(primary[[i]]$P_TwoSided, 5), " (", direction, ")")
}
primary <- do.call(rbind, primary)
primary$FDR_4contrasts <- p.adjust(primary$P_TwoSided, method = "BH")
primary$FDR_Mixed_4contrasts <- p.adjust(primary$P_Mixed, method = "BH")
primary$Signed_LogP <- ifelse(primary$Direction == "Down", -1, 1) *
  -log10(pmax(primary$P_TwoSided, .Machine$double.xmin))
primary$Significance <- ifelse(primary$FDR_4contrasts < PANEL_SIGNIFICANCE, "*", "")
write.csv(primary, file.path(tables_dir, "08_FIG5A_Primary_GO0004017_ROAST.csv"), row.names = FALSE)

# Figure 5c repeats the test for 5 nucleotide-kinase GO functions in
# each of the 4 comparisons. A separate seed makes each run reproducible.
# The significance stars use BH adjustment across all 20 tests, not the
# within-contrast adjusted values that mroast also reports.
sensitivity <- vector("list", nrow(CONTRASTS))
for (i in seq_len(nrow(CONTRASTS))) {
  set.seed(SEED_SENSITIVITY + i)
  rr <- limma::mroast(expr, index = set_indices, design = design,
                      contrast = contrast_vector(i), nrot = N_ROTATIONS,
                      set.statistic = SET_STATISTIC, adjust.method = "BH", midp = TRUE,
                      sort = "none")
  if (!all(GENE_SETS$Set_Key %in% rownames(rr))) stop("Missing sensitivity GO set in ROAST output.")
  rr <- rr[GENE_SETS$Set_Key, , drop = FALSE]
  req <- c("NGenes", "PropDown", "PropUp", "Direction", "PValue", "FDR", "PValue.Mixed", "FDR.Mixed")
  if (!all(req %in% names(rr))) stop("Unexpected mroast columns: ", paste(setdiff(req, names(rr)), collapse=", "))
  frame <- as.data.frame(rr[, req, drop = FALSE])
  frame$Set_Key <- rownames(rr)
  frame$Contrast <- CONTRASTS$Contrast[i]
  frame$GO_ID <- GENE_SETS$GO_ID[match(frame$Set_Key, GENE_SETS$Set_Key)]
  frame$Display_Name <- GENE_SETS$Display_Name[match(frame$Set_Key, GENE_SETS$Set_Key)]
  frame$Contrast_Label <- CONTRASTS$Contrast_Label[i]
  sensitivity[[i]] <- frame
  message("Sensitivity ROAST ", CONTRASTS$Contrast_Label[i], ": ", nrow(frame), " sets")
}
sensitivity <- do.call(rbind, sensitivity)
sensitivity$FDR_Global <- p.adjust(sensitivity$PValue, method = "BH")

# Cluster the 8 gene profiles using Euclidean distance and Ward.D2 linkage. 
hc <- hclust(dist(z, method = "euclidean"), method = "ward.D2")

# Branches of a hierarchical tree can be rotated without changing its
# underlying clusters. We used that freedom to display the same top-to-bottom
# orientation as the manuscript. Before plotting, verify that this ordering
# is genuinely compatible with the calculated tree, otherwise the code stop.
published_gene_order <- c("AK2", "AK3", "AK8", "AK5", "AK4", "AK7", "AK1", "AK9")
if (!setequal(published_gene_order, rownames(z)) ||
    anyDuplicated(published_gene_order))
  stop("The manuscript gene labels do not match the GO:0004017 matrix.")
branch_members <- vector("list", nrow(hc$merge))
for (i in seq_len(nrow(hc$merge))) {
  descendants <- lapply(hc$merge[i, ], function(child) {
    if (child < 0) rownames(z)[-child] else branch_members[[child]]
  })
  branch_members[[i]] <- unlist(descendants, use.names = FALSE)
  positions <- match(branch_members[[i]], published_gene_order)
  if (anyNA(positions) ||
      (max(positions) - min(positions) + 1L) != length(positions)) {
    stop("Published Figure 5b order is incompatible with the Ward.D2 tree. ",
         "Review the input data; do not force a misleading dendrogram.")
  }
}
gene_order <- published_gene_order
write.csv(data.frame(Position = seq_along(gene_order), Gene = gene_order,
                     Clustering = "Euclidean / Ward.D2",
                     Display_Orientation = "Manuscript Figure 5b"),
          file.path(tables_dir, "13_FIG5B_Clustered_Gene_Order.csv"), row.names = FALSE)
n <- length(gene_order)
# Draw the verified Ward.D2 dendrogram beside the 8 expression rows.
node_x <- vector("list", nrow(hc$merge))
node_h <- numeric(nrow(hc$merge))
leaf_y <- setNames(n:1, gene_order)
segments <- list()
for (i in seq_len(nrow(hc$merge))) {
  child <- hc$merge[i, ]
  child_y <- numeric(2)
  child_h <- numeric(2)
  for (k in 1:2) {
    if (child[k] < 0) {
      child_y[k] <- leaf_y[rownames(z)[-child[k]]]
      child_h[k] <- 0
    } else {
      child_y[k] <- node_x[[child[k]]]
      child_h[k] <- node_h[child[k]]
    }
  }
  yh <- mean(child_y)
  hh <- hc$height[i]
  segments[[length(segments)+1L]] <- data.frame(x = hh, xend = hh, y = min(child_y), yend = max(child_y))
  for (k in 1:2) segments[[length(segments)+1L]] <-
    data.frame(x = child_h[k], xend = hh, y = child_y[k], yend = child_y[k])
  node_x[[i]] <- yh
  node_h[i] <- hh
}
dendro_df <- do.call(rbind, segments)

                  width=width,height=height,units="in",device="pdf",bg="white",limitsize=FALSE)
  ggplot2::ggsave(file.path(plots_dir, paste0(file,".png")),plot=p,
                  width=width,height=height,units="in",dpi=300,bg="white",limitsize=FALSE)

                  width=width,height=height,units="in",dpi=300,
                  compression="lzw",bg="white",limitsize=FALSE)
}
