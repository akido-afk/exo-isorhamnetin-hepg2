    PROJECT

    / "results"

    / "AK1_ISO_pose_RMSD_matrix.csv"

)

df.index = df.index.astype(int)

df.columns = df.columns.astype(int)



dominant = [

    1001,

    2002,

    3003,

    4004,

    5005,

    7007,

    9009,

    10101

]



sub = df.loc[dominant, dominant]



means = {}



for seed in dominant:



    values = [

        sub.loc[seed, other]

        for other in dominant

        if other != seed

    ]



    means[seed] = sum(values) / len(values)





print()

print("=" * 60)

print("AK1 DOMINANT-CLUSTER REPRESENTATIVE POSE")

print("=" * 60)



for seed in sorted(means):

    print(

        f"Seed {seed}: "

        f"mean RMSD to dominant cluster = "

        f"{means[seed]:.3f} A"

    )



representative = min(

    means,

    key=means.get

)



print()

print("-------------------------------------------")

print(

    f"Representative dominant pose: seed {representative}"

)

print(

    f"Mean RMSD within dominant cluster: "

    f"{means[representative]:.3f} A"

)

fixer.findMissingAtoms()

    PDBFile.writeFile(

        fixer.topology,

        fixer.positions,

        keepIds=True

    )

pdb = r"..\ligand\AK1_native_AP5_chainA.pdb"



coords = []



with open(pdb) as f:

    for line in f:

        if line.startswith("HETATM"):

            x = float(line[30:38])

            y = float(line[38:46])

            z = float(line[46:54])

            coords.append((x, y, z))



xs = [x[0] for x in coords]

ys = [x[1] for x in coords]

zs = [x[2] for x in coords]



center_x = (min(xs) + max(xs)) / 2

center_y = (min(ys) + max(ys)) / 2

center_z = (min(zs) + max(zs)) / 2



# Add 5 Å of space on every side of the native ligand

size_x = (max(xs) - min(xs)) + 10

size_y = (max(ys) - min(ys)) + 10

size_z = (max(zs) - min(zs)) + 10



print()

print("AK1 docking box")
