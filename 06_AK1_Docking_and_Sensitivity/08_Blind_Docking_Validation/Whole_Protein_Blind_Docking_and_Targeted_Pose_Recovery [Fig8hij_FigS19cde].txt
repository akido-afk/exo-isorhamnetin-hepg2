# ELEMENT HANDLING

# ============================================================



def pdbqt_element(atom_type):



    atom_type = (

        atom_type

        .strip()

        .upper()

    )



    if atom_type.startswith("H"):

        return "H"



    if atom_type.startswith("O"):

        return "O"



    if atom_type.startswith("N"):

        return "N"



    if atom_type.startswith("S"):

        return "S"



    if atom_type.startswith("P"):

        return "P"



    if atom_type in (

        "C",

        "A"

    ):

        return "C"



    return atom_type[:1]

# READ MULTI-MODE PDBQT

# ============================================================



def read_pdbqt_models(path):



    path = Path(path)



    if not path.exists():



        raise FileNotFoundError(

            path

        )





    lines = path.read_text(

        errors="ignore"

    ).splitlines()





    models = {}

    scores = {}



    current_model = None

    atoms = []





    for line in lines:



        if line.startswith("MODEL"):



            current_model = int(

                line.split()[1]

            )



            atoms = []



            continue





        if line.startswith(

            "REMARK VINA RESULT:"

        ):



            if current_model is not None:



                scores[

                    current_model

                ] = float(

                    line.split()[3]

                )



            continue





        if line.startswith("ENDMDL"):



            if current_model is not None:



                models[

                    current_model

                ] = atoms



            current_model = None

            atoms = []



            continue





        if current_model is None:

            continue





        if not (

            line.startswith("ATOM")

            or

            line.startswith("HETATM")

        ):

            continue





        atom_name = (

            line[12:16]

            .strip()

        )



        atom_type = (

            line.split()[-1]

        )



        element = (

            pdbqt_element(

                atom_type

            )

        )





        if element == "H":

            continue





        x = float(

            line[30:38]

        )



        y = float(

            line[38:46]

        )



        z = float(

            line[46:54]

        )





        atoms.append({



            "atom_name":

                atom_name,



            "atom_type":

                atom_type,



            "element":

                element,



            "coord":

                np.array(

                    [x, y, z],

                    dtype=float

                )

        })





    if not models:



        raise RuntimeError(

            f"No docking models found in {path}"

        )





    return (

        models,

        scores

    )

# READ AP5 HEAVY ATOMS

# ============================================================



def read_ap5(path):



    coords = []





    with open(

        path,

        "r",

        encoding="utf-8",

        errors="ignore"

    ) as handle:



        for line in handle:



            if not (

                line.startswith("ATOM")

                or

                line.startswith("HETATM")

            ):

                continue





            element = (

                line[76:78]

                .strip()

                .upper()

                if len(line) >= 78

                else ""

            )





            if element == "H":

                continue





            try:



                x = float(

                    line[30:38]

                )



                y = float(

                    line[38:46]

                )



                z = float(

                    line[46:54]

                )



            except Exception:



                continue





            coords.append(

                [x, y, z]

            )





    if not coords:



        raise RuntimeError(

            "No AP5 heavy atoms found."

        )





    return np.array(

        coords,

        dtype=float

    )





# ============================================================

# POSITIONAL RMSD

#

# Same receptor coordinate frame.

# No ligand post-superposition.

# ============================================================



def positional_rmsd(

    coords_a,

    coords_b

):



    if coords_a.shape != coords_b.shape:



        raise RuntimeError(

            "Coordinate arrays have different shapes."

        )





    diff = (

        coords_a -

        coords_b

    )





    return math.sqrt(



        np.mean(



            np.sum(

                diff ** 2,

                axis=1

            )

        )

    )

# AP5 REGION METRICS

# ============================================================



def ap5_metrics(

    iso_coords,

    ap5_coords

):



    iso_centroid = np.mean(

        iso_coords,

        axis=0

    )



    ap5_centroid = np.mean(

        ap5_coords,

        axis=0

    )





    centroid_distance = float(

        np.linalg.norm(

            iso_centroid -

            ap5_centroid

        )

    )





    distance_matrix = np.linalg.norm(



        iso_coords[

            :,

            None,

            :

        ]

        -

        ap5_coords[

            None,

            :,

            :

        ],



        axis=2

    )





    minimum_distance = float(

        np.min(

            distance_matrix

        )

    )





    min_to_ap5 = np.min(

        distance_matrix,

        axis=1

    )





    percent_within_4 = float(

        100.0

        *

        np.mean(

            min_to_ap5 <= 4.0

        )

    )





    percent_within_6 = float(

        100.0

        *

        np.mean(

            min_to_ap5 <= 6.0

        )

    )





    return {



        "ISO_AP5_centroid_distance_A":

            centroid_distance,



        "Minimum_ISO_AP5_distance_A":

            minimum_distance,



        "ISO_atoms_within_4A_percent":

            percent_within_4,



        "ISO_atoms_within_6A_percent":

            percent_within_6

    }

def pdbqt_element(atom_type):



    atom_type = atom_type.strip().upper()



    if atom_type.startswith("H"):

        return "H"



    if atom_type.startswith("O"):

        return "O"



    if atom_type.startswith("N"):

        return "N"



    if atom_type.startswith("S"):

        return "S"



    if atom_type.startswith("P"):

        return "P"



    if atom_type in ("C", "A"):

        return "C"



    return atom_type[:1]

def read_pdbqt_models(path):



    lines = Path(path).read_text(

        errors="ignore"

    ).splitlines()



    models = {}

    scores = {}



    current_model = None

    atoms = []



    for line in lines:



        if line.startswith("MODEL"):



            current_model = int(

                line.split()[1]

            )



            atoms = []

            continue





        if line.startswith("REMARK VINA RESULT:"):



            if current_model is not None:



                scores[

                    current_model

                ] = float(

                    line.split()[3]

                )



            continue





        if line.startswith("ENDMDL"):



            if current_model is not None:



                models[

                    current_model

                ] = atoms



            current_model = None

            atoms = []

            continue





        if current_model is None:

            continue





        if not (

            line.startswith("ATOM")

            or

            line.startswith("HETATM")

        ):

            continue





        atom_name = (

            line[12:16]

            .strip()

        )



        atom_type = (

            line.split()[-1]

        )



        element = (

            pdbqt_element(

                atom_type

            )

        )



        if element == "H":

            continue





        x = float(

            line[30:38]

        )



        y = float(

            line[38:46]

        )



        z = float(

            line[46:54]

        )





        atoms.append({



            "atom_name":

                atom_name,



            "atom_type":

                atom_type,



            "element":

                element,



            "coord":

                np.array(

                    [x, y, z],

                    dtype=float

                )

        })





    return (

        models,

        scores

    )

# ============================================================

# RMSD

# ============================================================



def positional_rmsd(

    coords_a,

    coords_b

):



    if coords_a.shape != coords_b.shape:



        raise RuntimeError(

            "Coordinate arrays differ in shape."

        )



    diff = (

        coords_a -

        coords_b

    )



    return math.sqrt(

        np.mean(

            np.sum(

                diff ** 2,

                axis=1

            )

        )

    )





# ============================================================

# AP5 REGION METRICS

# ============================================================



def ap5_metrics(

    iso_coords,

    ap5_coords

):



    iso_centroid = np.mean(

        iso_coords,

        axis=0

    )



    ap5_centroid = np.mean(

        ap5_coords,

        axis=0

    )





    centroid_distance = float(

        np.linalg.norm(

            iso_centroid -

            ap5_centroid

        )

    )





    distance_matrix = np.linalg.norm(



        iso_coords[

            :,

            None,

            :

        ]

        -

        ap5_coords[

            None,

            :,

            :

        ],



        axis=2

    )





    min_to_ap5 = np.min(

        distance_matrix,

        axis=1

    )





    percent_within_4 = float(

        100.0

        *

        np.mean(

            min_to_ap5 <= 4.0

        )

    )





    percent_within_6 = float(

        100.0

        *

        np.mean(

            min_to_ap5 <= 6.0

        )

    )





    minimum_distance = float(

        np.min(

            distance_matrix

        )

    )





    return (

        centroid_distance,

        minimum_distance,

        percent_within_4,

        percent_within_6

    )

    f"{maximum[1]:.3f}, "

    f"{maximum[2]:.3f}"

)



print()



print(

    f"{protein_size[0]:.3f}, "

    f"{protein_size[1]:.3f}, "

    f"{protein_size[2]:.3f} A"

    / "AK1_blind_box.txt"
