    if not best_sdf.exists():



        writer = Chem.SDWriter(

    return mol

# ANALYZE ONE ISO POSE

# ============================================================



def analyze_pose(mol):



    conf = mol.GetConformer()



    # residue -> shortest heavy-atom distance

    any_contacts = {}



    # residue -> shortest polar-heavy-atom distance

    polar_contacts = {}



    for atom in mol.GetAtoms():



        element = atom.GetSymbol().upper()



        if element == "H":

            continue



        idx = atom.GetIdx()



        pos = conf.GetAtomPosition(idx)



        iso_coord = (

            float(pos.x),

            float(pos.y),

            float(pos.z)

        )



        for p in protein_atoms:



            d = distance(

                iso_coord,

                p["coord"]

            )



            residue = (

                f"{p['resname']}{p['resid']}"

            )



            # --------------------------------------------

            # ANY HEAVY-ATOM CONTACT

            # --------------------------------------------



            if d <= ANY_CONTACT_CUTOFF:



                if (

                    residue not in any_contacts

                    or d < any_contacts[residue]

                ):



                    any_contacts[residue] = d



            # --------------------------------------------

            # POLAR HEAVY-ATOM CONTACT

            # --------------------------------------------



            if (

                element in {"O", "N", "S"}

                and

                p["element"] in {"O", "N", "S"}

                and

                d <= POLAR_CONTACT_CUTOFF

            ):



                if (

                    residue not in polar_contacts

                    or d < polar_contacts[residue]

                ):



                    polar_contacts[residue] = d



    return any_contacts, polar_contacts

# ============================================================

# DISTANCE FUNCTION

# ============================================================



def distance(a, b):

    return sqrt(

        (a[0] - b[0]) ** 2

        + (a[1] - b[1]) ** 2

        + (a[2] - b[2]) ** 2

    )





# ============================================================

# READ PROTEIN ATOMS FROM PDB

# ============================================================



protein_atoms = []



with open(PROTEIN_FILE, "r") as f:



    for line in f:



        if not line.startswith("ATOM"):

            continue



        atom_name = line[12:16].strip()

        resname = line[17:20].strip()

        chain = line[21].strip()

        resid = line[22:26].strip()



        x = float(line[30:38])

        y = float(line[38:46])

        z = float(line[46:54])



        element = line[76:78].strip()



        if not element:

            element = atom_name[0]



        # Only polar heavy atoms

        if element.upper() not in {"O", "N", "S"}:

            continue



        protein_atoms.append({

            "atom": atom_name,

            "resname": resname,

            "resid": resid,

            "chain": chain,

            "element": element.upper(),

            "coord": (x, y, z)

        })





# ============================================================

# PROTEIN DONOR / ACCEPTOR RULES

# ============================================================



def protein_roles(resname, atom_name):



    roles = set()



    # -------------------------

    # Backbone

    # -------------------------



    if atom_name == "N" and resname != "PRO":

        roles.add("donor")



    if atom_name in {"O", "OXT"}:

        roles.add("acceptor")





    # -------------------------

    # Side-chain donors

    # -------------------------



    donor_atoms = {

        "ARG": {"NE", "NH1", "NH2"},

        "ASN": {"ND2"},

        "GLN": {"NE2"},

        "HIS": {"ND1", "NE2"},

        "LYS": {"NZ"},

        "SER": {"OG"},

        "THR": {"OG1"},

        "TRP": {"NE1"},

        "TYR": {"OH"},

        "CYS": {"SG"},

    }



    if atom_name in donor_atoms.get(resname, set()):

        roles.add("donor")





    # -------------------------

    # Side-chain acceptors

    # -------------------------



    acceptor_atoms = {

        "ASP": {"OD1", "OD2"},

        "GLU": {"OE1", "OE2"},

        "ASN": {"OD1"},

        "GLN": {"OE1"},

        "HIS": {"ND1", "NE2"},

        "SER": {"OG"},

        "THR": {"OG1"},

        "TYR": {"OH"},

        "CYS": {"SG"},

        "MET": {"SD"},

    }



    if atom_name in acceptor_atoms.get(resname, set()):

        roles.add("acceptor")



    return roles

# ============================================================



supplier = Chem.SDMolSupplier(

iso = None

            iso_roles[atom_id].add("acceptor")

# ============================================================



contacts = []



for iso_atom in iso.GetAtoms():



    iso_idx = iso_atom.GetIdx()

    iso_element = iso_atom.GetSymbol().upper()



    if iso_element not in {"O", "N", "S"}:

        continue



    pos = conf.GetAtomPosition(iso_idx)



    iso_coord = (

        float(pos.x),

        float(pos.y),

        float(pos.z)

    )



    ligand_roles = iso_roles.get(

        iso_idx,

    )



    for p in protein_atoms:

        if d > CUTOFF:

            continue



        p_roles = protein_roles(

            p["resname"],

        # ====================================================

        # ====================================================



        donor_acceptor_match = (

                and "acceptor" in p_roles

            (

                "acceptor" in ligand_roles

                and "donor" in p_roles

            )

        )

        contacts.append({

            "Residue": f"{p['resname']}{p['resid']}",

            "Chain": p["chain"],

            "Protein atom": p["atom"],

            "Protein role": "/".join(sorted(p_roles)) if p_roles else "polar atom",

            "ISO atom": iso_labels[iso_idx],

            "ISO atom index": iso_idx + 1,

            "ISO role": "/".join(sorted(ligand_roles)) if ligand_roles else "polar atom",

            "Distance (A)": d,

            "Contact type": contact_type

        })

clean_distance <- function(x) {

  x <- as.character(x)

  x[x == "-" | x == "" | is.na(x)] <- NA

}

# ============================================================

# CALCULATE OVERLAPS

# ============================================================



shared_any <- intersect(

  iso_any,

  ap5_any

)



iso_only_any <- setdiff(

  iso_any,

  ap5_any

)



ap5_only_any <- setdiff(

  ap5_any,

  iso_any

)





shared_polar <- intersect(

  iso_polar,

  ap5_polar

)



iso_only_polar <- setdiff(

  iso_polar,

  ap5_polar

)



ap5_only_polar <- setdiff(

  ap5_polar,

  iso_polar

)





# ============================================================

# SUMMARY METRICS

# ============================================================



any_iso_coverage <-

  length(shared_any) /

  length(iso_any) *

  100





polar_iso_coverage <-

  length(shared_polar) /

  length(iso_polar) *

  100





any_jaccard <-

  length(shared_any) /

  length(

    union(

      iso_any,

      ap5_any

    )

  )





polar_jaccard <-

  length(shared_polar) /

  length(

    union(

      iso_polar,

      ap5_polar

    )

  )

# ------------------------------------------------------------

# LONG FORMAT

# ------------------------------------------------------------



polar_matrix <- polar_class %>%



  mutate(



    ISO_present =

      Residue %in% iso_polar,



    AP5_present =

      Residue %in% ap5_polar

  ) %>%



  pivot_longer(



    cols = c(

      ISO_present,

      AP5_present

    ),



    names_to =

      "Ligand",



    values_to =

      "Present"

  ) %>%



  mutate(



    Ligand = recode(



      Ligand,



      "ISO_present" =

        "ISO dominant\ncluster",



      "AP5_present" =

        "Native AP5"

    ),



    Category = factor(



      Category,



      levels = c(

        "Shared",

        "ISO only",

        "AP5 only"

      )

    ),



    Residue = factor(



      Residue,



      levels = rev(

        polar_order

      )

    ),



    Ligand = factor(



      Ligand,



      levels = c(

        "ISO dominant\ncluster",

        "Native AP5"

      )

    ),



    Cell_type = case_when(



      Present &

        Ligand ==

        "ISO dominant\ncluster" ~

        "ISO",



      Present &

        Ligand ==

        "Native AP5" ~

        "AP5",



      TRUE ~

        "Absent"

    ),



    Cell_label = case_when(



      Present &

        Ligand ==

        "ISO dominant\ncluster" ~

        "8/8",



      Present &

        Ligand ==

        "Native AP5" ~

        "\u2713",



      TRUE ~

        ""

    )

  )

library(ggplot2)
library(dplyr)
library(tidyr)

# =========================================================
# INPUT / OUTPUT
# =========================================================
input_csv <- "../results/AK1_dominant8_contact_frequency/AK1_dominant8_contact_frequency.csv"
out_dir   <- "../results/AK1_panelG"

dir.create(out_dir, recursive = TRUE, showWarnings = FALSE)

# =========================================================
# HELPERS
# =========================================================
parse_ratio_numerator <- function(x) {
  x <- trimws(as.character(x))
  as.numeric(sub("/.*", "", x))
}

clean_distance <- function(x) {
  x <- as.character(x)
  x[x == "-" | x == "" | is.na(x)] <- NA
  suppressWarnings(as.numeric(gsub("[^0-9.]", "", x)))
}

# =========================================================
# READ TABLE
# Expected CSV columns:
# Residue, Any, Polar, Polar., Min.polar.dist.
# =========================================================
raw <- read.csv(input_csv, check.names = TRUE, stringsAsFactors = FALSE)

# Safety check for expected columns
print("Detected columns:")
print(colnames(raw))

# Accept the supplied recurrence table's explicit count/distance column names.
if (!("Any" %in% names(raw)) && "Any.contact.n" %in% names(raw)) raw$Any <- raw$Any.contact.n
if (!("Polar" %in% names(raw)) && "Polar.contact.n" %in% names(raw)) raw$Polar <- raw$Polar.contact.n
if (!("Polar.." %in% names(raw)) && "Polar.contact.." %in% names(raw)) raw[["Polar.."]] <- raw[["Polar.contact.."]]
if (!("Min.polar.dist." %in% names(raw)) && "Polar.shortest.distance" %in% names(raw)) raw[["Min.polar.dist."]] <- raw[["Polar.shortest.distance"]]
if (!all(c("Residue", "Any", "Polar") %in% names(raw))) stop("The recurrence CSV needs residue, broad-contact count and polar-contact count columns.")


freq <- raw %>%
  transmute(
    Residue = Residue,
    Any = Any,
    Polar = Polar,
    PolarPct = if ("Polar.." %in% colnames(raw)) `Polar..` else if ("Polar." %in% colnames(raw)) `Polar.` else NA,
    MinPolarDist = if ("Min.polar.dist." %in% colnames(raw)) `Min.polar.dist.` else if ("Min.polar.dist" %in% colnames(raw)) `Min.polar.dist` else NA,
    Any_n = parse_ratio_numerator(Any),
    Polar_n = parse_ratio_numerator(Polar)
  ) %>%
  mutate(
    MinPolarDist_num = clean_distance(MinPolarDist),
    RecurrentPolar = Polar_n == 8
  )

# =========================================================
# ORDER RESIDUES
# Main ordering:
# 1) highest polar recurrence
# 2) highest any-contact recurrence
# 3) shortest min polar distance
# =========================================================
freq <- freq %>%
  arrange(desc(Polar_n), desc(Any_n), MinPolarDist_num)

freq <- freq %>%
  mutate(
    ResidueLabel = ifelse(RecurrentPolar, paste0(Residue, " *"), Residue)
  )

residue_order <- freq$ResidueLabel

# =========================================================
# LONG FORMAT FOR TILE PLOT
# =========================================================
plot_df <- bind_rows(
  freq %>%
    transmute(
      ResidueLabel,
      ContactType = "Any contact",
      Count = Any_n
    ),
  freq %>%
    transmute(
      ResidueLabel,
      ContactType = "Polar contact",
      Count = Polar_n
    )
) %>%
  mutate(
    ContactType = factor(ContactType, levels = c("Any contact", "Polar contact")),
    ResidueLabel = factor(ResidueLabel, levels = rev(residue_order)),
    Label = paste0(Count, "/8")
  )

# recurrent polar residues for subtitle
recurrent_residues <- freq %>%
  filter(RecurrentPolar) %>%

subtitle_text <- paste0(

# =========================================================

p <- ggplot(plot_df, aes(x = ContactType, y = ResidueLabel, fill = Count)) +
  geom_tile(color = "white", linewidth = 1.0, width = 0.92, height = 0.92) +

    values = c(0, 0.5, 1),
    limits = c(0, 8),
    breaks = c(0, 2, 4, 6, 8),
    name = "Recurrence\nacross 8 poses"
  ) +

  labs(
    title = "AK1 dominant-cluster contact recurrence",
