def is_hydrogen_atom_type(atom_type):



    atom_type = atom_type.strip().upper()



    return atom_type.startswith("H")





def read_pdbqt_atoms(

    path,

    model_number=None

):



    path = Path(path)



    if not path.exists():

        raise FileNotFoundError(path)



    lines = path.read_text(

        errors="ignore"

    ).splitlines()



    has_models = any(

        line.startswith("MODEL")

        for line in lines

    )



    current_model = None



    atoms = []



    for line in lines:



        if line.startswith("MODEL"):



            parts = line.split()



            if len(parts) >= 2:



                try:

                    current_model = int(

                        parts[1]

                    )

                except Exception:

                    current_model = None



            continue



        if line.startswith("ENDMDL"):



            current_model = None

            continue



        if not (

            line.startswith("ATOM")

            or

            line.startswith("HETATM")

        ):

            continue



        if has_models:



            if model_number is None:

                continue



            if current_model != model_number:

                continue



        try:



            x = float(

                line[30:38]

            )



            y = float(

                line[38:46]

            )



            z = float(

                line[46:54]

            )



        except Exception as exc:



            raise RuntimeError(

                f"Could not parse coordinates in:\n"

                f"{path}\n"

                f"Line:\n{line}"

            ) from exc



        fields = line.split()



        atom_type = fields[-1]



        if is_hydrogen_atom_type(

            atom_type

        ):

            continue



        atom_name = (

            line[12:16].strip()

        )



        atoms.append({

            "atom_name":

                atom_name,



            "atom_type":

                atom_type,



            "x":

                x,



            "y":

                y,



            "z":

                z

        })



    if len(atoms) == 0:



        raise RuntimeError(

            f"No heavy atoms found in {path}"

        )



    return atoms





def atom_coordinates(atoms):



    return np.array(

        [

            [

                atom["x"],

                atom["y"],

                atom["z"]

            ]

            for atom in atoms

        ],

        dtype=float

    )

    raise RuntimeError(

def build_coordinate_mapping(

    reference_coords,

    target_coords

):



    distance_matrix = np.linalg.norm(

        reference_coords[:, None, :]

        -

        target_coords[None, :, :],

        axis=2

    )



    ref_indices, target_indices = (

        linear_sum_assignment(

            distance_matrix

        )

    )



    mapping = np.zeros(

        len(reference_coords),

        dtype=int

    )



    mapping[

        ref_indices

    ] = target_indices



    matched_distances = (

        distance_matrix[

            ref_indices,

            target_indices

        ]

    )



    return (

        mapping,

        matched_distances

    )

# RMSD FUNCTION

#

# IMPORTANT:

# No structural superposition.

# All ligands were docked into the same AK1 coordinate frame.

# ============================================================



def positional_rmsd(

    coords_a,

    coords_b

):



    difference = (

        coords_a -

        coords_b

    )



    squared_distance = np.sum(

        difference ** 2,

        axis=1

    )



    return math.sqrt(

        np.mean(

            squared_distance

        )

    )

# COMPLETE-LINKAGE CLUSTERING

# ============================================================



def cluster_pose_matrix(

    matrix,

    threshold

):



    n = matrix.shape[0]



    if n == 1:



        return np.array(

            [1],

            dtype=int

        )



    condensed = squareform(

        matrix,

        checks=False

    )



    hierarchy = linkage(

        condensed,

        method="complete"

    )



    labels = fcluster(

        hierarchy,

        t=threshold,

        criterion="distance"

    )



    return labels

def write_table_sheet(

    workbook,

    name,

    rows

):



    ws = workbook.add_worksheet(

        name

    )



    if not rows:

        return



    headers = list(

        rows[0].keys()

    )



    for col, header in enumerate(

        headers

    ):



        ws.write(

            0,

            col,

            header,

            header_fmt

        )



    for row_i, row in enumerate(

        rows,

        start=1

    ):



        for col, header in enumerate(

            headers

        ):



            value = row[header]



            if isinstance(

                value,

                (

                    float,

                    np.floating

                )

            ):



                ws.write_number(

                    row_i,

                    col,

                    float(value),

                    number_fmt

                )



            else:



                ws.write(

                    row_i,

                    col,

                    value,

                    cell_fmt

                )



    ws.freeze_panes(

        1,

        0

    )



    ws.set_column(

        0,

        len(headers) - 1,

        22

    )

# READ PDBQT HEAVY ATOMS

# ============================================================



def read_heavy_atoms(

    path,

    model_number=None

):



    path = Path(path)



    if not path.exists():

        raise FileNotFoundError(path)



    lines = path.read_text(

        errors="ignore"

    ).splitlines()



    has_models = any(

        line.startswith("MODEL")

        for line in lines

    )



    current_model = None

    atoms = []



    for line in lines:



        if line.startswith("MODEL"):



            parts = line.split()



            if len(parts) >= 2:



                try:

                    current_model = int(

                        parts[1]

                    )

                except Exception:

                    current_model = None



            continue



        if line.startswith("ENDMDL"):



            current_model = None

            continue



        if not (

            line.startswith("ATOM")

            or line.startswith("HETATM")

        ):

            continue



        if has_models:



            if current_model != model_number:

                continue



        fields = line.split()



        atom_type = fields[-1]



        if atom_type.upper().startswith("H"):

            continue



        x = float(line[30:38])

        y = float(line[38:46])

        z = float(line[46:54])



        atoms.append({

            "atom_type": atom_type,

            "coords": np.array(

                [x, y, z],

                dtype=float

            )

        })



    if len(atoms) == 0:



        raise RuntimeError(

            f"No heavy atoms found in {path}"

        )



    return atoms

def read_heavy_atoms(

    path,

    model_number=None

):



    path = Path(path)



    if not path.exists():

        raise FileNotFoundError(path)



    lines = path.read_text(

        errors="ignore"

    ).splitlines()



    has_models = any(

        line.startswith("MODEL")

        for line in lines

    )



    current_model = None

    atoms = []



    for line in lines:



        if line.startswith("MODEL"):



            parts = line.split()



            if len(parts) >= 2:



                try:

                    current_model = int(

                        parts[1]

                    )

                except Exception:

                    current_model = None



            continue



        if line.startswith("ENDMDL"):



            current_model = None

            continue



        if not (

            line.startswith("ATOM")

            or

            line.startswith("HETATM")

        ):

            continue



        if has_models:



            if current_model != model_number:

                continue



        fields = line.split()



        atom_type = fields[-1]



        if atom_type.upper().startswith("H"):

            continue



        x = float(

            line[30:38]

        )



        y = float(

            line[38:46]

        )



        z = float(

            line[46:54]

        )



        atoms.append({

            "atom_type":

                atom_type,



            "coords":

                np.array(

                    [x, y, z],

                    dtype=float

                )

        })



    if len(atoms) == 0:



        raise RuntimeError(

            f"No heavy atoms found in {path}"

        )



    return atoms

            fields = line.split()



            return float(

    raise RuntimeError(

        f"No Vina result found in {path}"

def load_molecule(path):



    supplier = Chem.SDMolSupplier(

        str(path),

        removeHs=False

    )



    mol = None



    for candidate in supplier:



        if candidate is not None:

            mol = candidate

            break



    if mol is None:



        raise RuntimeError(

            f"Could not read molecule:\n{path}"

        )



    # Remove explicit H for cleaner publication structure

    mol = Chem.RemoveHs(

        mol

    )



    # Generate clean 2D coordinates

    AllChem.Compute2DCoords(

        mol

    )



    return mol

# BASIC ELEMENT HELPERS

def infer_pdb_element(

    line

):



    element = (

        line[76:78]

        .strip()

        .upper()

        if len(line) >= 78

        else ""

    )



    if element:

        return element



    atom_name = (

        line[12:16]

        .strip()

        .upper()

    )



    letters = "".join(

        c

        for c in atom_name

        if c.isalpha()

    )



    if not letters:

        return ""



    # Standard protein atom naming:

    # CA = carbon alpha, not calcium.

    return letters[0]





def pdbqt_element(

    atom_type,

    atom_name

):



    atom_type = (

        atom_type

        .strip()

        .upper()

    )



    atom_name = (

        atom_name

        .strip()

        .upper()

    )



    if atom_type.startswith("H"):

        return "H"



    if atom_type.startswith("O"):

        return "O"



    if atom_type.startswith("N"):

        return "N"



    if atom_type.startswith("S"):

        return "S"



    if atom_type in (

        "C",

        "A"

    ):

        return "C"



    if atom_type.startswith("P"):

        return "P"



    letters = "".join(

        c

        for c in atom_name

        if c.isalpha()

    )



    if letters:

        return letters[0]



    return ""





def is_polar_element(

    element

    return element.upper() in {

def read_protein(

    path

):



    if not path.exists():



        raise FileNotFoundError(

            path

        )



    residues = defaultdict(

        list

    )



    with open(

        path,

        "r",

        encoding="utf-8",

        errors="ignore"

    ) as handle:



        for line in handle:



            if not line.startswith(

                "ATOM"

            ):

                continue



            chain = (

                line[21]

                .strip()

            )



            if chain not in (

                "",

            ):

                continue



            atom_name = (

                line[12:16]

                .strip()

            )



            residue_name = (

                line[17:20]

                .strip()

            )



            residue_number = (

                line[22:26]

                .strip()

            )



            insertion_code = (

                line[26]

                .strip()

            )



            element = (

                infer_pdb_element(

                    line

                )

            )



            if element == "H":

                continue



            try:



                x = float(

                    line[30:38]

                )



                y = float(

                    line[38:46]

                )



                z = float(

                    line[46:54]

                )



            except Exception:

                continue



            residue_id = (

                f"{residue_name}"

                f"{residue_number}"

            )



            if insertion_code:



                residue_id += (

                    insertion_code

            residues[

                residue_id

                "atom_name":

                    atom_name,



                "element":

                    element,



                "coord":

                    np.array(

                            x,

                            y,

                            z

                        ],

                        dtype=float

                    )

            })

        raise RuntimeError(

            "No protein residues were read."

# EXTRACT CONTACTS FOR ONE POSE

# ============================================================



def calculate_pose_contacts(

    protein_residues,

    ligand_atoms

):



    results = []



    ligand_polar = [

        atom

        for atom in ligand_atoms

        if is_polar_element(

            atom["element"]

        )

    ]



    for residue_id, protein_atoms in (

        protein_residues.items()

    ):



        min_any = math.inf

        min_any_pair = None



        min_polar = math.inf

        min_polar_pair = None





        # ----------------------------------------------------

        # ANY HEAVY-ATOM CONTACT

        # ----------------------------------------------------



        for protein_atom in protein_atoms:



            for ligand_atom in ligand_atoms:



                d = distance(

                    protein_atom["coord"],

                    ligand_atom["coord"]

                )



                if d < min_any:



                    min_any = d



                    min_any_pair = (

                        protein_atom,

                        ligand_atom

                    )





        # ----------------------------------------------------

        # POLAR HEAVY-ATOM PROXIMITY

        # ----------------------------------------------------



        protein_polar = [

            atom

            for atom in protein_atoms

            if is_polar_element(

                atom["element"]

            )

        ]



        for protein_atom in protein_polar:



            for ligand_atom in ligand_polar:



                d = distance(

                    protein_atom["coord"],

                    ligand_atom["coord"]

                )



                if d < min_polar:



                    min_polar = d



                    min_polar_pair = (

                        protein_atom,

                        ligand_atom

                    )





        any_contact = (

            min_any <=

            ANY_CONTACT_CUTOFF

        )



        polar_contact = (

            min_polar <=

            POLAR_CONTACT_CUTOFF

        )





        if (

            any_contact

            or

            polar_contact

        ):



            row = {



                "Residue":

                    residue_id,



                "Any_contact":

                    any_contact,



                "Any_min_distance_A":

                    (

                        min_any

                        if math.isfinite(

                            min_any

                        )

                        else None

                    ),



                "Polar_contact":

                    polar_contact,



                "Polar_min_distance_A":

                    (

                        min_polar

                        if math.isfinite(

                            min_polar

                        )

                        else None

                    ),



                "Any_protein_atom":

                    (

                        min_any_pair[0][

                            "atom_name"

                        ]

                        if min_any_pair

                        else ""

                    ),



                "Any_ligand_atom":

                    (

                        min_any_pair[1][

                            "atom_name"

                        ]

                        if min_any_pair

                        else ""

                    ),



                "Polar_protein_atom":

                    (

                        min_polar_pair[0][

                            "atom_name"

                        ]

                        if min_polar_pair

                        else ""

                    ),



                "Polar_ligand_atom":

                    (

                        min_polar_pair[1][

                            "atom_name"

                        ]

                        if min_polar_pair

                        else ""

                    )

            }



            results.append(

                row

            )



    return results

def write_sheet(

    if not rows:



        worksheet.write(

            0,

        return



    headers = list(

        rows[0].keys()

    for col, header in enumerate(

        headers

        worksheet.write(

            0,

            col,

            header,

            header_fmt

        )



    for row_i, row in enumerate(

        rows,

        start=1

        for col, header in enumerate(

            headers

        ):



            value = row.get(

                header,

                ""

            )



            if value is None:

                    row_i,

                    col,

                    None,

                    cell_fmt

                )



            elif isinstance(

                value,

                (

                    bool,

                    np.bool_

                )

            ):



                worksheet.write(

                    row_i,

                    col,

                    str(

                        bool(

                            value

                        )

                    ),

                    cell_fmt

                )



            elif isinstance(

                value,

                (

                    float,

                    np.floating

                )

            ):



                if (

                    "percent"

                    in header.lower()

                ):



                    worksheet.write_number(

                        row_i,

                        col,

                        float(value),

                        percent_fmt

                    )



                else:



                    worksheet.write_number(

                        row_i,

                        col,

                        float(value),

                        number_fmt

                    )



            elif isinstance(

                value,

                (

                    int,

                    np.integer

                )

            ):



                worksheet.write_number(

                    row_i,

                    col,

                    int(value),

                    cell_fmt

                )



            else:



                worksheet.write(

                    row_i,

                    col,

                    value,

                    cell_fmt

                )



    worksheet.freeze_panes(

        1,

        0

    )



    worksheet.autofilter(

        0,

        0,

        len(rows),

        len(headers) - 1

    )



    worksheet.set_column(

        0,

        len(headers) - 1,

        width

    )

# ============================================================

# FUNCTION TO DEPROTONATE SPECIFIC OXYGENS

# ============================================================



def make_deprotonated_state(

    original_mol,

    oxygen_indices

):



    rw = Chem.RWMol(

        original_mol

    )



    # Find explicit H indices BEFORE deleting atoms

    removals = []



    for oxygen_idx in oxygen_indices:



        h_idx = find_explicit_h_neighbor(

            original_mol,

            oxygen_idx

        )



        removals.append(

            (

                oxygen_idx,

                h_idx

            )

        )



    # Set oxygen formal charges first

    for oxygen_idx, h_idx in removals:



        oxygen = rw.GetAtomWithIdx(

            oxygen_idx

        )



        oxygen.SetFormalCharge(

            -1

        )



        oxygen.SetNoImplicit(

            True

        )



    # Remove hydrogens in descending atom index order

    # so earlier indices do not shift unexpectedly.

    h_indices = sorted(

        [

            h_idx

            for oxygen_idx, h_idx

            in removals

        ],

        reverse=True

    )



    for h_idx in h_indices:



        rw.RemoveAtom(

            h_idx

        )



    result = rw.GetMol()



    Chem.SanitizeMol(

        result

    )



    return result

# HELPER:

# FIND NEAREST MEMBER OF A REFERENCE FAMILY

#

# drop = FALSE is critical here.

# It prevents R from turning a one-column matrix into

# an unnamed scalar when only one candidate remains.

# ============================================================



get_nearest_reference <- function(

  query_seed,

  candidate_seeds

) {



  if (length(candidate_seeds) == 0) {



    stop(

      paste(

        "No reference candidates available for seed",

        query_seed

      )

    )

  }





  submatrix <- neutral_matrix[

    as.character(query_seed),

    as.character(candidate_seeds),

    drop = FALSE

  ]





  values <- as.numeric(

    submatrix[1, ]

  )





  names(values) <- colnames(

    submatrix

  )





  min_index <- which.min(

    values

  )





  list(



    seed = as.integer(

      names(values)[

        min_index

      ]

    ),



    rmsd = as.numeric(

      values[

        min_index

      ]

    )

  )

}

get_recurrent_set <- function(

  group_name,

  type = "polar",

  threshold = 80

) {



  group_data <- dat %>%

    filter(

      Group == group_name

    )



  if (type == "polar") {



    return(

      group_data %>%

        filter(

          Polar_contact_percent >= threshold

        ) %>%

        pull(

          Residue

        )

    )



  } else {



    return(

      group_data %>%

        filter(

          Any_contact_percent >= threshold

        ) %>%

        pull(

          Residue

        )

    )

  }

}

from pathlib import Path

from rdkit import Chem



PROJECT = Path(r"C:\Users\akidi\Desktop\ISO_AK_Docking")



SDF = PROJECT / "ligand" / "isorhamnetin_3D.sdf"



if not SDF.exists():

    raise FileNotFoundError(SDF)



supplier = Chem.SDMolSupplier(

    str(SDF),

    removeHs=False

)



mol = None



for m in supplier:

    if m is not None:

        mol = m

        break



if mol is None:

    raise RuntimeError("Could not read isorhamnetin SDF.")



print("=" * 75)

print("CURRENT ISORHAMNETIN LIGAND STATE")

print("=" * 75)



print()

print("Input:")

print(SDF)



print()

print("Molecular formula-like atom counts:")



counts = {}



for atom in mol.GetAtoms():

    symbol = atom.GetSymbol()

    counts[symbol] = counts.get(symbol, 0) + 1



for symbol in sorted(counts):

    print(f"  {symbol}: {counts[symbol]}")



formal_charge = sum(

    atom.GetFormalCharge()

    for atom in mol.GetAtoms()

)



print()

print("Total formal charge:", formal_charge)



print()

print("OXYGEN ATOMS")

print("-" * 75)



for atom in mol.GetAtoms():



    if atom.GetSymbol() != "O":

        continue



    idx = atom.GetIdx()



    neighbors = []



    for nbr in atom.GetNeighbors():



        neighbors.append(

            f"{nbr.GetSymbol()}{nbr.GetIdx()+1}"

        )



    print(

        f"O atom index {idx+1:>2} | "

        f"formal charge {atom.GetFormalCharge():>2} | "

        f"explicit H {atom.GetNumExplicitHs()} | "

        f"implicit H {atom.GetNumImplicitHs()} | "

        f"neighbors: {', '.join(neighbors)}"

    )



print()

print("Canonical SMILES:")

print(

    Chem.MolToSmiles(

        mol,

        canonical=True

    )

)



print()

print("=" * 75)
