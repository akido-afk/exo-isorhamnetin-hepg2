# PDBQT HEAVY-ATOM READER

# ============================================================



def pdbqt_element(

    atom_type

):



    atom_type = (

        atom_type

        .strip()

        .upper()

    )



    if atom_type.startswith("H"):

        return "H"



    if atom_type.startswith("O"):

        return "O"



    if atom_type.startswith("N"):

        return "N"



    if atom_type.startswith("S"):

        return "S"



    if atom_type.startswith("P"):

        return "P"



    if atom_type in (

        "C",

        "A"

    ):

        return "C"



    return atom_type[:1]





def read_pdbqt_atoms(

    path,

    model_number=None

):



    path = Path(path)



    if not path.exists():



        raise FileNotFoundError(

            path

        )



    lines = path.read_text(

        errors="ignore"

    ).splitlines()



    has_models = any(

        line.startswith("MODEL")

        for line in lines

    )



    current_model = None

    atoms = []



    for line in lines:



        if line.startswith("MODEL"):



            fields = line.split()



            try:



                current_model = int(

                    fields[1]

                )



            except Exception:



                current_model = None



            continue





        if line.startswith("ENDMDL"):



            current_model = None

            continue





        if not (

            line.startswith("ATOM")

            or

            line.startswith("HETATM")

        ):

            continue





        if has_models:



            if model_number is None:

                continue



            if current_model != model_number:

                continue





        atom_name = (

            line[12:16]

            .strip()

        )



        fields = line.split()



        atom_type = (

            fields[-1]

        )



        element = (

            pdbqt_element(

                atom_type

            )

        )





        if element == "H":

            continue





        x = float(

            line[30:38]

        )



        y = float(

            line[38:46]

        )



        z = float(

            line[46:54]

        )





        atoms.append({



            "atom_name":

                atom_name,



            "atom_type":

                atom_type,



            "element":

                element,



            "coord":

                np.array(

                    [x, y, z],

                    dtype=float

                )

        })





    if not atoms:



        raise RuntimeError(

            f"No heavy atoms found in {path}"

        )





    return atoms

def infer_pdb_element(

    line

):



    element = (

        line[76:78]

        .strip()

        .upper()

        if len(line) >= 78

        else ""

    )



    if element:



        return element





    atom_name = (

        line[12:16]

        .strip()

        .upper()

    )



    letters = "".join(

        c

        for c in atom_name

        if c.isalpha()

    )



    return (

        letters[0]

        if letters

        else ""

    )

    path = Path(path)



    if not path.exists():



        raise FileNotFoundError(

        )



    atoms = []





    with open(

        errors="ignore"

    ) as handle:



        for line in handle:



            if not (

                line.startswith("ATOM")

                line.startswith("HETATM")

            ):

                continue





            element = (

                infer_pdb_element(

            if element == "H":

            x = float(

                line[30:38]

            )



            y = float(

            z = float(

            atoms.append(

                    dtype=float

                )

            )





    if not atoms:



        raise RuntimeError(

            f"No heavy atoms found in {path}"

# POSITIONAL RMSD

#

# No post-docking structural superposition.

# All receptors were placed in the same 1Z83 coordinate frame.

# ============================================================



def positional_rmsd(

    coords_a,

    coords_b

):



    if coords_a.shape != coords_b.shape:



        raise ValueError(

            "Ligand coordinate arrays have different shapes."

        )





    diff = (

        coords_a -

        coords_b

    )





    return math.sqrt(



        np.mean(



            np.sum(

                diff ** 2,

                axis=1

            )

        )

    )

# CLUSTERING

# ============================================================



def cluster_matrix(

    matrix,

    threshold

):



    if matrix.shape[0] == 1:



        return np.array(

            [1],

            dtype=int

        )





    condensed = squareform(

        matrix,

        checks=False

    )





    hierarchy = linkage(

        condensed,

        method="complete"

    )





    return fcluster(

        hierarchy,

        t=threshold,

        criterion="distance"

    )

# ============================================================

# AP5 REGION METRICS

# ============================================================



def ap5_region_metrics(

    iso_coords

):



    iso_centroid = (

        np.mean(

            iso_coords,

            axis=0

        )

    )





    centroid_distance = float(

        np.linalg.norm(

            iso_centroid -

            ap5_centroid

        )

    )





    distance_matrix = np.linalg.norm(



        iso_coords[

            :,

            None,

            :

        ]

        -

        ap5_coords[

            None,

            :,

            :

        ],



        axis=2

    )





    minimum_distance = float(

        np.min(

            distance_matrix

        )

    )





    min_to_ap5_per_iso_atom = (

        np.min(

            distance_matrix,

            axis=1

        )

    )





    n_within_4 = int(

        np.sum(

            min_to_ap5_per_iso_atom <= 4.0

        )

    )





    n_within_6 = int(

        np.sum(

            min_to_ap5_per_iso_atom <= 6.0

        )

    )





    percent_within_4 = (

        100.0 *

        n_within_4 /

        len(

            iso_coords

        )

    )





    percent_within_6 = (

        100.0 *

        n_within_6 /

        len(

            iso_coords

        )

    )





    return {



        "ISO_AP5_centroid_distance_A":

            centroid_distance,



        "Minimum_ISO_AP5_heavy_atom_distance_A":

            minimum_distance,



        "ISO_atoms_within_4A_of_AP5_n":

            n_within_4,



        "ISO_atoms_within_4A_of_AP5_percent":

            percent_within_4,



        "ISO_atoms_within_6A_of_AP5_n":

            n_within_6,



        "ISO_atoms_within_6A_of_AP5_percent":

            percent_within_6

    }

def infer_pdb_element(line):



    element = (

        line[76:78]

        .strip()

        .upper()

        if len(line) >= 78

        else ""

    )



    if element:

        return element



    atom_name = (

        line[12:16]

        .strip()

        .upper()

    )



    letters = "".join(

        c

        for c in atom_name

        if c.isalpha()

    )



    if not letters:

        return ""



    # Protein CA = alpha carbon, not calcium.

    return letters[0]





def pdbqt_element(

    atom_type,

    atom_name

):



    atom_type = (

        atom_type

        .strip()

        .upper()

    )



    atom_name = (

        atom_name

        .strip()

        .upper()

    )



    if atom_type.startswith("H"):

        return "H"



    if atom_type.startswith("O"):

        return "O"



    if atom_type.startswith("N"):

        return "N"



    if atom_type.startswith("S"):

        return "S"



    if atom_type.startswith("P"):

        return "P"



    if atom_type in (

        "C",

        "A"

    ):

        return "C"



    letters = "".join(

        c

        for c in atom_name

        if c.isalpha()

    )



    if letters:

        return letters[0]



    return ""

# READ PROTEIN RECEPTOR

# ============================================================



def read_protein(path):



    if not path.exists():



        raise FileNotFoundError(

            path

        )



    residues = defaultdict(

        list

    )



    with open(

        path,

        "r",

        encoding="utf-8",

        errors="ignore"

    ) as handle:



        for line in handle:



            if not line.startswith(

                "ATOM"

            ):

                continue



            chain = (

                line[21]

                .strip()

            )



            atom_name = (

                line[12:16]

                .strip()

            )



            resname = (

                line[17:20]

                .strip()

            )



            resnum = int(

                line[22:26]

            )



            icode = (

                line[26]

                .strip()

            )



            element = (

                infer_pdb_element(

                    line

                )

            )



            if element == "H":

                continue



            try:



                x = float(

                    line[30:38]

                )



                y = float(

                    line[38:46]

                )



                z = float(

                    line[46:54]

                )



            except Exception:

                continue





            residue_key = (

                chain,

                resname,

                resnum,

                icode

            )





            residues[

                residue_key

            ].append({



                "atom_name":

                    atom_name,



                "element":

                    element,



                "coord":

                    np.array(

                        [x, y, z],

                        dtype=float

                    )

            })





    if not residues:



        raise RuntimeError(

            f"No protein residues read from {path}"

        )





    return residues

# READ BEST DOCKED ISO POSE

# ============================================================



def read_best_ligand_pose(path):



    if not path.exists():



        raise FileNotFoundError(

            path

        )



    atoms = []



    lines = path.read_text(

        errors="ignore"

    ).splitlines()



    has_models = any(

        line.startswith("MODEL")

        for line in lines

    )



    current_model = None





    for line in lines:



        if line.startswith("MODEL"):



            fields = line.split()



            try:



                current_model = int(

                    fields[1]

                )



            except Exception:



                current_model = None



            continue





        if line.startswith("ENDMDL"):



            current_model = None

            continue





        if not (

            line.startswith("ATOM")

            or

            line.startswith("HETATM")

        ):

            continue





        if has_models:



            if current_model != 1:

                continue





        atom_name = (

            line[12:16]

            .strip()

        )



        fields = line.split()



        atom_type = (

            fields[-1]

        )



        element = pdbqt_element(

            atom_type,

            atom_name

        )





        if element == "H":

            continue





        x = float(

            line[30:38]

        )



        y = float(

            line[38:46]

        )



        z = float(

            line[46:54]

        )





        atoms.append({



            "atom_name":

                atom_name,



            "atom_type":

                atom_type,



            "element":

                element,



            "coord":

                np.array(

                    [x, y, z],

                    dtype=float

                )

        })





    if not atoms:



        raise RuntimeError(

            f"No heavy ligand atoms read from {path}"

        )





    return atoms

# MAP PROTEIN RESIDUE TO 1Z83 NUMBERING

# ============================================================



def map_residue_to_reference(

    group_name,

    mapping_model,

    residue_key

):



    (

        chain,

        mobile_resname,

        mobile_resnum,

        icode

    ) = residue_key





    # --------------------------------------------------------

    # 1Z83 already uses reference numbering

    # --------------------------------------------------------



    if mapping_model is None:



        return {



            "Mapped":

                True,



            "Reference_resnum":

                mobile_resnum,



            "Reference_resname":

                mobile_resname,



            "Reference_residue":

                f"{mobile_resname}{mobile_resnum}",



            "Mobile_resnum":

                mobile_resnum,



            "Mobile_resname":

                mobile_resname,



            "Sequence_identical":

                True

        }





    # --------------------------------------------------------

    # 7X7S -> 1Z83 mapping

    # --------------------------------------------------------



    mapping = (

        mapping_by_model[

            mapping_model

        ]

    )





    if mobile_resnum not in mapping:



        return {



            "Mapped":

                False,



            "Reference_resnum":

                None,



            "Reference_resname":

                "",



            "Reference_residue":

                "",



            "Mobile_resnum":

                mobile_resnum,



            "Mobile_resname":

                mobile_resname,



            "Sequence_identical":

                False

        }





    item = mapping[

        mobile_resnum

    ]





    return {



        "Mapped":

            True,



        "Reference_resnum":

            item[

                "Reference_resnum"

            ],



        "Reference_resname":

            item[

                "Reference_resname"

            ],



        "Reference_residue":

            (

                f"{item['Reference_resname']}"

                f"{item['Reference_resnum']}"

            ),



        "Mobile_resnum":

            mobile_resnum,



        "Mobile_resname":

            mobile_resname,



        "Sequence_identical":

            item[

                "Identical"

            ]

    }

# DISTANCE

# ============================================================



def distance(

    coord_a,

    coord_b

):



    return float(

        np.linalg.norm(

            coord_a -

            coord_b

        )

    )

# CALCULATE CONTACTS FOR ONE POSE

# ============================================================



def calculate_pose_contacts(

    protein_residues,

    ligand_atoms,

    group_name,

    mapping_model

):



    rows = []





    ligand_polar_atoms = [



        atom



        for atom in ligand_atoms



        if is_polar_element(

            atom[

                "element"

            ]

        )

    ]





    for residue_key, protein_atoms in (

        protein_residues.items()

    ):



        mapping = (

            map_residue_to_reference(

                group_name,

                mapping_model,

                residue_key

            )

        )





        min_any = math.inf

        min_any_pair = None



        min_polar = math.inf

        min_polar_pair = None





        # ----------------------------------------------------

        # ANY CONTACT

        # ----------------------------------------------------



        for protein_atom in protein_atoms:



            for ligand_atom in ligand_atoms:



                d = distance(

                    protein_atom[

                        "coord"

                    ],

                    ligand_atom[

                        "coord"

                    ]

                )



                if d < min_any:



                    min_any = d



                    min_any_pair = (

                        protein_atom,

                        ligand_atom

                    )





        # ----------------------------------------------------

        # POLAR PROXIMITY

        # ----------------------------------------------------



        protein_polar_atoms = [



            atom



            for atom in protein_atoms



            if is_polar_element(

                atom[

                    "element"

                ]

            )

        ]





        for protein_atom in protein_polar_atoms:



            for ligand_atom in ligand_polar_atoms:



                d = distance(

                    protein_atom[

                        "coord"

                    ],

                    ligand_atom[

                        "coord"

                    ]

                )



                if d < min_polar:



                    min_polar = d



                    min_polar_pair = (

                        protein_atom,

                        ligand_atom

                    )





        any_contact = (

            min_any <=

            ANY_CONTACT_CUTOFF

        )



        polar_contact = (

            min_polar <=

            POLAR_CONTACT_CUTOFF

        )





        if not (

            any_contact

            or

            polar_contact

        ):

            continue





        rows.append({



            "Reference_residue":

                mapping[

                    "Reference_residue"

                ],



            "Reference_resnum":

                mapping[

                    "Reference_resnum"

                ],



            "Reference_resname":

                mapping[

                    "Reference_resname"

                ],



            "Mobile_resnum":

                mapping[

                    "Mobile_resnum"

                ],



            "Mobile_resname":

                mapping[

                    "Mobile_resname"

                ],



            "Mapped":

                mapping[

                    "Mapped"

                ],



            "Sequence_identical":

                mapping[

                    "Sequence_identical"

                ],



            "Any_contact":

                any_contact,



            "Any_min_distance_A":

                (

                    min_any

                    if math.isfinite(

                        min_any

                    )

                    else None

                ),



            "Any_protein_atom":

                (

                    min_any_pair[

                        0

                    ][

                        "atom_name"

                    ]

                    if min_any_pair

                    else ""

                ),



            "Any_ligand_atom":

                (

                    min_any_pair[

                        1

                    ][

                        "atom_name"

                    ]

                    if min_any_pair

                    else ""

                ),



            "Polar_contact":

                polar_contact,



            "Polar_min_distance_A":

                (

                    min_polar

                    if math.isfinite(

                        min_polar

                    )

                    else None

                ),



            "Polar_protein_atom":

                (

                    min_polar_pair[

                        0

                    ][

                        "atom_name"

                    ]

                    if min_polar_pair

                    else ""

                ),



            "Polar_ligand_atom":

                (

                    min_polar_pair[

                        1

                    ][

                        "atom_name"

                    ]

                    if min_polar_pair

                    else ""

                )

        })





    return rows

# READ MULTI-MODEL PDB

# Stores C-alpha coordinates for chain A.

def read_nmr_models(path):



    if not path.exists():

        raise FileNotFoundError(path)



    models = {}



    current_model = None



    with open(

        path,

        "r",

        encoding="utf-8",

        errors="ignore"

    ) as handle:



        for line in handle:



            if line.startswith("MODEL"):



                fields = line.split()



                if len(fields) >= 2:



                    current_model = int(

                        fields[1]

                    )



                    models[

                        current_model

                    ] = {}



                continue





            if line.startswith("ENDMDL"):



                current_model = None

                continue





            if current_model is None:

                continue





            if not line.startswith("ATOM"):

                continue





            chain = (

                line[21]

                .strip()

            )



            if chain != "A":

                continue





            atom_name = (

                line[12:16]

                .strip()

            )



            if atom_name != "CA":

                continue





            altloc = (

                line[16]

                .strip()

            )



            if altloc not in (

                "",

                "A"

            ):

                continue





            resname = (

                line[17:20]

                .strip()

            )



            resseq = int(

                line[22:26]

            )



            icode = (

                line[26]

                .strip()

            )





            x = float(

                line[30:38]

            )



            y = float(

                line[38:46]

            )



            z = float(

                line[46:54]

            )





            residue_key = (

                resseq,

                icode

            )





            models[

                current_model

            ][

                residue_key

            ] = {



                "resname":

                    resname,



                "coord":

                    np.array(

                        [

                            x,

                            y,

                            z

                        ],

                        dtype=float

                    )

            }





    if not models:



        raise RuntimeError(

            "No MODEL records were read from 7X7S."

        )





    return models

# ============================================================

# RESIDUE HELPER

# ============================================================



def find_residue_by_number(

    model,

    number

):



    matches = [



        value



        for (

            resseq,

            icode

        ), value in model.items()



        if resseq == number

    ]





    if len(matches) != 1:



        raise RuntimeError(

            f"Expected one residue numbered {number}, "

            f"found {len(matches)}."

        )





    return matches[0]





# ============================================================

# DISTANCE

# ============================================================



def ca_distance(

    model,

    residue_a,

    residue_b

):



    a = find_residue_by_number(

        model,

        residue_a

    )["coord"]



    b = find_residue_by_number(

        model,

        residue_b

    )["coord"]





    return float(

        np.linalg.norm(

            a - b

        )

    )





# ============================================================

# KABSCH RMSD

#

# Pairwise C-alpha RMSD after optimal rigid-body

# superposition of the residues common to all 20 models.

# ============================================================



def kabsch_rmsd(

    coords_a,

    coords_b

):



    if coords_a.shape != coords_b.shape:



        raise ValueError(

            "Coordinate arrays have different shapes."

        )





    a = (

        coords_a -

        np.mean(

            coords_a,

            axis=0

        )

    )



    b = (

        coords_b -

        np.mean(

            coords_b,

            axis=0

        )

    )





    covariance = (

        a.T @ b

    )





    u, s, vt = np.linalg.svd(

        covariance

    )





    rotation = (

        u @ vt

    )





    # Avoid improper reflection

    if np.linalg.det(

        rotation

    ) < 0:



        vt[-1, :] *= -1



        rotation = (

            u @ vt

        )





    aligned_a = (

        a @ rotation

    )





    difference = (

        aligned_a -

        b

    )





    return math.sqrt(



        np.mean(



            np.sum(

                difference ** 2,

                axis=1

            )

        )

    )

# HEATMAP FUNCTION

# ============================================================



make_contact_heatmap <- function(

  data,

  contact_type,

  cutoff_text,

  output_prefix

) {



  if (contact_type == "POLAR") {



    percent_col <- "Polar_contact_percent"

    count_col <- "Polar_contact_n"



  } else if (contact_type == "ANY") {



    percent_col <- "Any_contact_percent"

    count_col <- "Any_contact_n"



  } else {



    stop(

      "contact_type must be POLAR or ANY"

    )

  }





  # ----------------------------------------------------------

  # Keep residues recurrent >=80% in at least one state

  # ----------------------------------------------------------



  max_by_residue <- aggregate(

    data[[percent_col]],

    by = list(

      Reference_residue = data$Reference_residue

    ),

    FUN = max,

    na.rm = TRUE

  )



  names(max_by_residue)[2] <- "Max_percent"





  keep_residues <- max_by_residue$Reference_residue[

    max_by_residue$Max_percent >= 80

  ]





  plot_data <- data[

    data$Reference_residue %in% keep_residues,

    ,

    drop = FALSE

  ]





  # ----------------------------------------------------------

  # Order receptors

  # ----------------------------------------------------------



  plot_data$Group <- factor(

    plot_data$Group,

    levels = group_order,

    labels = group_labels

  )





  # ----------------------------------------------------------

  # Order residues numerically

  # ----------------------------------------------------------



  residue_table <- unique(

    data.frame(

      residue = plot_data$Reference_residue,

      number = get_resnum(

        plot_data$Reference_residue

      ),

      stringsAsFactors = FALSE

    )

  )





  residue_table <- residue_table[

    order(

      residue_table$number,

      residue_table$residue

    ),

    ,

    drop = FALSE

  ]





  residue_order <- residue_table$residue





  plot_data$Reference_residue <- factor(

    plot_data$Reference_residue,

    levels = rev(

      residue_order

    )

  )





  # ----------------------------------------------------------

  # Values

  # ----------------------------------------------------------



  plot_data$Percent <- plot_data[[percent_col]]



  plot_data$Count <- plot_data[[count_col]]





  plot_data$Label <- ifelse(

    plot_data$Count > 0,

    paste0(

      plot_data$Count,

      "/",

      plot_data$n_poses

    ),

    ""

  )





  # ----------------------------------------------------------

  # SAVE EXACT PLOTTED DATA

  # ----------------------------------------------------------



  write.csv(

    plot_data,

    file.path(

      figure_dir,

      paste0(

        output_prefix,

        "_plot_data.csv"

      )

    ),

    row.names = FALSE

  )





  # ----------------------------------------------------------

  # PLOT

  # ----------------------------------------------------------



  p <- ggplot(

    plot_data,

    aes(

      x = Group,

      y = Reference_residue,

      fill = Percent

    )

  ) +



    geom_tile(

      linewidth = 0.45,

      color = "white"

    ) +



    geom_text(

      aes(

        label = Label

      ),

      size = 3.5,

      fontface = "bold"

    ) +



    scale_fill_gradient(

      low = "white",

      high = "#4472C4",

      limits = c(

        0,

        100

      ),

      breaks = c(

        0,

        20,

        40,

        60,

        80,

        100

      ),

      name = "Recurrence\n(%)"

    ) +



    labs(

      x = NULL,

      y = "AK1 residue (1Z83 numbering)",

      title = ifelse(

        contact_type == "POLAR",

        "AK1 polar-contact recurrence across receptor conformations",

        "AK1 residue-contact recurrence across receptor conformations"

      ),

      subtitle = paste0(

        cutoff_text,

        "; cells show number of poses with contact / total poses"

      )

    ) +



    theme_classic(

      base_size = 12

    ) +



    theme(

      axis.text.x = element_text(

        size = 11,

        face = "bold",

        lineheight = 0.9

      ),



      axis.text.y = element_text(

        size = 10

      ),



      axis.title.y = element_text(

        size = 11

      ),



      plot.title = element_text(

        size = 15,

        face = "bold",

        hjust = 0

      ),



      plot.subtitle = element_text(

        size = 10,

        hjust = 0

      ),



      legend.title = element_text(

        size = 10,

        face = "bold"

      ),



      legend.text = element_text(

        size = 9

      ),



      panel.grid = element_blank(),



      plot.margin = margin(

        12,

        16,

        12,

        12

      )

    )





  # ----------------------------------------------------------

  # Dynamic figure height

  # ----------------------------------------------------------



  n_residues <- length(

    residue_order

  )



  fig_height <- max(

    5.5,

    2.5 + n_residues * 0.30

  )





  # ----------------------------------------------------------

  # PNG

  # ----------------------------------------------------------



  ggsave(

    filename = file.path(

      figure_dir,

      paste0(

        output_prefix,

        ".png"

      )

    ),

    plot = p,

    width = 8.2,

    height = fig_height,

    units = "in",

    dpi = 600,

    bg = "white"

  )





  # ----------------------------------------------------------

  # PDF

  # ----------------------------------------------------------



  ggsave(

    filename = file.path(

      figure_dir,

      paste0(

        output_prefix,

        ".pdf"

      )

    ),

    plot = p,

    width = 8.2,

    height = fig_height,

    units = "in",

    bg = "white"

  )





  return(

    invisible(

      p

    )

  )

}

def read_reference_structure(

    path

):



    atoms = []

    ca_ordered = []

    seen_ca = set()



    with open(

        path,

        "r",

        encoding="utf-8",

        errors="ignore"

    ) as handle:



        for line in handle:



            if not line.startswith(

                "ATOM"

            ):

                continue



            chain = (

                line[21]

                .strip()

            )



            if chain != "A":

                continue



            altloc = (

                line[16]

                .strip()

            )



            if altloc not in (

                "",

                "A"

            ):

                continue



            atom_name = (

                line[12:16]

                .strip()

            )



            resname = (

                line[17:20]

                .strip()

            )



            resnum = int(

                line[22:26]

            )



            icode = (

                line[26]

                .strip()

            )



            x = float(

                line[30:38]

            )



            y = float(

                line[38:46]

            )



            z = float(

                line[46:54]

            )



            atom = {

                "line":

                    line.rstrip("\n"),



                "atom_name":

                    atom_name,



                "resname":

                    resname,



                "resnum":

                    resnum,



                "icode":

                    icode,



                "coord":

                    np.array(

                        [x, y, z],

                        dtype=float

                    )

            }



            atoms.append(

                atom

            )



            key = (

                resnum,

                icode

            )



            if (

                atom_name == "CA"

                and

                key not in seen_ca

            ):



                seen_ca.add(

                    key

                )



                ca_ordered.append({



                    "resnum":

                        resnum,



                    "icode":

                        icode,



                    "resname":

                        resname,



                    "coord":

                        atom[

                            "coord"

                        ]

                })



    if not atoms:



        raise RuntimeError(

            "No chain-A atoms read from 1Z83."

        )



    return (

        atoms,

        ca_ordered

    )

# ============================================================

# GLOBAL NEEDLEMAN-WUNSCH ALIGNMENT

#

# Simple sequence alignment used only to establish

# structural residue correspondence.

# ============================================================



def global_align(

    seq_a,

    seq_b,

    match_score=2,

    mismatch_score=-1,

    gap_score=-2

):



    n = len(

        seq_a

    )



    m = len(

        seq_b

    )



    score = np.zeros(

        (

            n + 1,

            m + 1

        ),

        dtype=int

    )



    traceback = np.zeros(

        (

            n + 1,

            m + 1

        ),

        dtype=np.int8

    )





    # Directions:

    # 1 = diagonal

    # 2 = up

    # 3 = left



    for i in range(

        1,

        n + 1

    ):



        score[

            i,

            0

        ] = (

            i *

            gap_score

        )



        traceback[

            i,

            0

        ] = 2





    for j in range(

        1,

        m + 1

    ):



        score[

            0,

            j

        ] = (

            j *

            gap_score

        )



        traceback[

            0,

            j

        ] = 3





    for i in range(

        1,

        n + 1

    ):



        for j in range(

            1,

            m + 1

        ):



            diagonal = (

                score[

                    i - 1,

                    j - 1

                ]

                +

                (

                    match_score

                    if seq_a[

                        i - 1

                    ] == seq_b[

                        j - 1

                    ]

                    else mismatch_score

                )

            )



            up = (

                score[

                    i - 1,

                    j

                ]

                +

                gap_score

            )



            left = (

                score[

                    i,

                    j - 1

                ]

                +

                gap_score

            )



            best = max(

                diagonal,

                up,

                left

            )



            score[

                i,

                j

            ] = best



            if best == diagonal:



                traceback[

                    i,

                    j

                ] = 1



            elif best == up:



                traceback[

                    i,

                    j

                ] = 2



            else:



                traceback[

                    i,

                    j

                ] = 3





    aligned_a = []

    aligned_b = []



    index_pairs = []



    i = n

    j = m





    while (

        i > 0

        or

        j > 0

    ):



        direction = (

            traceback[

                i,

                j

            ]

        )



        if (

            i > 0

            and

            j > 0

            and

            direction == 1

        ):



            aligned_a.append(

                seq_a[

                    i - 1

                ]

            )



            aligned_b.append(

                seq_b[

                    j - 1

                ]

            )



            index_pairs.append(

                (

                    i - 1,

                    j - 1

                )

            )



            i -= 1

            j -= 1



        elif (

            i > 0

            and

            (

                j == 0

                or

                direction == 2

            )

        ):



            aligned_a.append(

                seq_a[

                    i - 1

                ]

            )



            aligned_b.append(

                "-"

            )



            index_pairs.append(

                (

                    i - 1,

                    None

                )

            )



            i -= 1



        else:



            aligned_a.append(

                "-"

            )



            aligned_b.append(

                seq_b[

                    j - 1

                ]

            )



            index_pairs.append(

                (

                    None,

                    j - 1

                )

            )



            j -= 1





    aligned_a.reverse()

    aligned_b.reverse()

    index_pairs.reverse()





    return (

        "".join(

            aligned_a

        ),

        "".join(

            aligned_b

        ),

        index_pairs,

        int(

            score[

                n,

                m

            ]

        )

    )

# BUILD RESIDUE CORRESPONDENCE

# ============================================================



def build_residue_mapping(

    reference_residues,

    mobile_residues

):



    ref_seq = (

        sequence_from_residues(

            reference_residues

        )

    )



    mob_seq = (

        sequence_from_residues(

            mobile_residues

        )

    )



    (

        aligned_ref,

        aligned_mob,

        index_pairs,

        alignment_score

    ) = global_align(

        ref_seq,

        mob_seq

    )





    mapping_rows = []



    aligned_pairs = []



    identical_count = 0

    paired_count = 0





    for alignment_position, (

        ref_index,

        mob_index

    ) in enumerate(

        index_pairs,

        start=1

    ):



        if (

            ref_index is None

            or

            mob_index is None

        ):

            continue



        ref_residue = (

            reference_residues[

                ref_index

            ]

        )



        mob_residue = (

            mobile_residues[

                mob_index

            ]

        )



        ref_letter = (

            ref_seq[

                ref_index

            ]

        )



        mob_letter = (

            mob_seq[

                mob_index

            ]

        )



        identical = (

            ref_letter ==

            mob_letter

        )



        paired_count += 1



        if identical:

            identical_count += 1





        row = {



            "Alignment_position":

                alignment_position,



            "Reference_index":

                ref_index + 1,



            "Reference_resnum":

                ref_residue[

                    "resnum"

                ],



            "Reference_resname":

                ref_residue[

                    "resname"

                ],



            "Reference_AA":

                ref_letter,



            "Mobile_index":

                mob_index + 1,



            "Mobile_resnum":

                mob_residue[

                    "resnum"

                ],



            "Mobile_resname":

                mob_residue[

                    "resname"

                ],



            "Mobile_AA":

                mob_letter,



            "Identical":

                identical

        }



        mapping_rows.append(

            row

        )



        aligned_pairs.append(

            (

                ref_residue,

                mob_residue,

                identical

            )

        )





    sequence_identity = (

        100.0 *

        identical_count /

        paired_count

        if paired_count > 0

        else 0.0

    )





    return {

        "reference_sequence":

            ref_seq,



        "mobile_sequence":

            mob_seq,



        "aligned_reference":

            aligned_ref,



        "aligned_mobile":

            aligned_mob,



        "alignment_score":

            alignment_score,



        "paired_count":

            paired_count,



        "identical_count":

            identical_count,



        "identity_percent":

            sequence_identity,



        "mapping_rows":

            mapping_rows,



        "aligned_pairs":

            aligned_pairs

    }





# ============================================================

# KABSCH TRANSFORMATION

# ============================================================



def calculate_transform(

    mobile_coords,

    reference_coords

):



    mobile_center = np.mean(

        mobile_coords,

        axis=0

    )



    reference_center = np.mean(

        reference_coords,

        axis=0

    )



    mobile_zero = (

        mobile_coords -

        mobile_center

    )



    reference_zero = (

        reference_coords -

        reference_center

    )



    covariance = (

        mobile_zero.T

        @

        reference_zero

    )



    u, s, vt = np.linalg.svd(

        covariance

    )



    rotation = (

        u @ vt

    )



    if np.linalg.det(

        rotation

    ) < 0:



        vt[

            -1,

            :

        ] *= -1



        rotation = (

            u @ vt

        )





    return (

        rotation,

        mobile_center,

        reference_center

    )





# ============================================================

# APPLY TRANSFORMATION

# ============================================================



def apply_transform(

    coord,

    rotation,

    mobile_center,

    reference_center

):



    return (

        (

            coord -

            mobile_center

        )

        @

        rotation

        +

        reference_center

    )





# ============================================================

# RMSD

# ============================================================



def rmsd(

    coordinates_a,

    coordinates_b

):



    difference = (

        coordinates_a -

        coordinates_b

    )



    return math.sqrt(



        np.mean(



            np.sum(

                difference ** 2,

                axis=1

            )

        )

    )

def write_sheet(

    name,

    rows,

    width=25

):



    ws = workbook.add_worksheet(

        name

    )



    if not rows:

        return



    headers = list(

        rows[

            0

        ].keys()

    )



    for col, header in enumerate(

        headers

    ):



        ws.write(

            0,

            col,

            header,

            header_fmt

        )



    for row_i, row in enumerate(

        rows,

        start=1

    ):



        for col, header in enumerate(

            headers

        ):



            value = row.get(

                header,

                ""

            )



            if isinstance(

                value,

                (

                    float,

                    np.floating

                )

            ):



                ws.write_number(

                    row_i,

                    col,

                    float(

                        value

                    ),

                    number_fmt

                )



            elif isinstance(

                value,

                (

                    bool,

                    np.bool_

                )

            ):



                ws.write(

                    row_i,

                    col,

                    str(

                        bool(

                            value

                        )

                    ),

                    cell_fmt

                )



            else:



                ws.write(

                    row_i,

                    col,

                    value,

                    cell_fmt

                )



    ws.freeze_panes(

        1,

        0

    )



    ws.autofilter(

        0,

        0,

        len(

            rows

        ),

        len(

            headers

        ) - 1

    )



    ws.set_column(

        0,

        len(

            headers

        ) - 1,

        width

    )
