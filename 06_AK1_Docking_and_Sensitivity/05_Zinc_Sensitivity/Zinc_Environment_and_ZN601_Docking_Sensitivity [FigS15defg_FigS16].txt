# READ BEST VINA SCORE

# ============================================================



def read_best_score(pdbqt):



    with open(

        pdbqt,

        "r",

        encoding="utf-8"

    ) as f:



        for line in f:



            if line.startswith(

                "REMARK VINA RESULT:"

                    line.split()[3]

                )



    raise RuntimeError(

        f"No score found in {pdbqt}"

    )





# ============================================================

# EXPORT BEST POSE

# ============================================================



def export_best_pose(

    pdbqt,

    output_dir,

    prefix

):



    all_sdf = (

        output_dir

        / f"{prefix}_allposes.sdf"

    )



    best_sdf = (

        output_dir

        / f"{prefix}_best.sdf"

    )





    if not all_sdf.exists():



        subprocess.run(

            [

                sys.executable,

                str(MEEKO_EXPORT),

                str(pdbqt),

                "-s",

                str(all_sdf)

            ],

            check=True

        )





    supplier = Chem.SDMolSupplier(

        str(all_sdf),

        removeHs=False

    )





    mol = None



    for candidate in supplier:



        if candidate is not None:



            mol = candidate

            break





    if mol is None:



        raise RuntimeError(

            f"Could not extract pose from {pdbqt}"

        )





    if not best_sdf.exists():



        writer = Chem.SDWriter(

            str(best_sdf)

        )



        writer.write(mol)

        writer.close()





    return mol, best_sdf





# ============================================================

# POSITIONAL RMSD

#

# No superposition:

# all poses are already in the common AK1 coordinate frame.

# ============================================================



def positional_rmsd(

    mol1,

    mol2

):



    if (

        mol1.GetNumAtoms()

        != mol2.GetNumAtoms()

    ):



        raise RuntimeError(

            "Different ligand atom counts."

        )





    c1 = mol1.GetConformer()

    c2 = mol2.GetConformer()



    values = []





    for i in range(

        mol1.GetNumAtoms()

    ):



        a1 = mol1.GetAtomWithIdx(i)

        a2 = mol2.GetAtomWithIdx(i)



        if (

            a1.GetAtomicNum()

            != a2.GetAtomicNum()

        ):



            raise RuntimeError(

                f"Atom-order mismatch at index {i}"

            )





        if a1.GetAtomicNum() == 1:

            continue





        p1 = c1.GetAtomPosition(i)

        p2 = c2.GetAtomPosition(i)





        values.append(

            (p1.x - p2.x) ** 2

            +

            (p1.y - p2.y) ** 2

            +

            (p1.z - p2.z) ** 2

        )





    return math.sqrt(

        sum(values)

        /

        len(values)

    )

# ZN601 -> ISO DISTANCES

# ============================================================



def zn_distances(mol):



    conf = mol.GetConformer()



    minimum_heavy = None

    heavy_atom = None



    minimum_polar = None

    polar_atom = None





    for atom in mol.GetAtoms():



        if atom.GetAtomicNum() == 1:

            continue





        i = atom.GetIdx()



        p = conf.GetAtomPosition(i)





        d = math.sqrt(

            (p.x - zn_coord[0]) ** 2

            +

            (p.y - zn_coord[1]) ** 2

            +

            (p.z - zn_coord[2]) ** 2

        )





        atom_name = (

            f"{atom.GetSymbol()}{i + 1}"

        )





        if (

            minimum_heavy is None

            or d < minimum_heavy

        ):



            minimum_heavy = d

            heavy_atom = atom_name





        if (

            atom.GetSymbol().upper()

            in {"O", "N", "S"}

        ):



            if (

                minimum_polar is None

                or d < minimum_polar

            ):



                minimum_polar = d

                polar_atom = atom_name





    return (

        minimum_heavy,

        heavy_atom,

        minimum_polar,

        polar_atom

    )

# CLUSTERING FUNCTION

# Complete-link hierarchical clustering

# ============================================================



def cluster_at_threshold(

    matrix,

    threshold

):



    condensed = squareform(

        matrix,

        checks=False

    )





    Z = linkage(

        condensed,

        method="complete"

    )





    raw_labels = fcluster(

        Z,

        t=threshold,

        criterion="distance"

    )





    # Remap labels by cluster size

    counts = Counter(

        raw_labels

    )





    ordered_old_labels = sorted(

        counts.keys(),

        key=lambda lab:

            (

                -counts[lab],

                lab

            )

    )





    mapping = {

        old:

        new_index + 1



        for new_index, old

        in enumerate(

            ordered_old_labels

        )

    }





    labels = [

        mapping[x]

        for x in raw_labels

    ]





    return labels

def positional_rmsd(mol1, mol2):



    if (

        mol1.GetNumAtoms()

        !=

        mol2.GetNumAtoms()

    ):



        raise RuntimeError(

            "Different atom counts."

        )



    conf1 = mol1.GetConformer()

    conf2 = mol2.GetConformer()



    values = []



    for i in range(

        mol1.GetNumAtoms()

    ):



        a1 = mol1.GetAtomWithIdx(i)

        a2 = mol2.GetAtomWithIdx(i)



        if (

            a1.GetAtomicNum()

            !=

            a2.GetAtomicNum()

        ):



            raise RuntimeError(

                f"Atom mismatch at {i}"

            )



        if a1.GetAtomicNum() == 1:

            continue



        p1 = conf1.GetAtomPosition(i)

        p2 = conf2.GetAtomPosition(i)



        values.append(

            (p1.x - p2.x) ** 2

            +

            (p1.y - p2.y) ** 2

            +

            (p1.z - p2.z) ** 2

        )



    return math.sqrt(

        sum(values)

        /

        len(values)

    )

def dist(a, b):



    return math.sqrt(

        (a[0] - b[0]) ** 2

        + (a[1] - b[1]) ** 2

        + (a[2] - b[2]) ** 2

    )





# ============================================================

# READ PDB

# ============================================================



atoms = []



with open(PDB, "r") as f:



    for line in f:



        if not (

            line.startswith("ATOM")

            or line.startswith("HETATM")

        ):

            continue



        record = line[0:6].strip()

        atom = line[12:16].strip()

        resn = line[17:20].strip()

        chain = line[21].strip()

        resi = line[22:26].strip()



        x = float(line[30:38])

        y = float(line[38:46])

        z = float(line[46:54])



        element = line[76:78].strip()



        if not element:

            element = atom[0]



        atoms.append({

            "record": record,

            "atom": atom,

            "resn": resn,

            "chain": chain,

            "resi": resi,

            "element": element.upper(),

            "coord": (x, y, z)

        })





# ============================================================

# FIND CHAIN-A Zn

# ============================================================



zn_atoms = [

    a for a in atoms

    if a["resn"] == "ZN"

    and a["chain"] == "A"

]



if not zn_atoms:

    raise RuntimeError(

        "No chain-A Zn atoms found."

    )





# ============================================================

# BUILD 5-A NEIGHBOR TABLE

# ============================================================



neighbors = []



for zn in zn_atoms:



    zn_id = f"ZN{zn['resi']}"



    for a in atoms:



        if a is zn:

            continue



        d = dist(

            zn["coord"],

            a["coord"]

        )



        if d > 5.0:

            continue



        if a["record"] == "ATOM":



            category = "PROTEIN"



        elif a["resn"] == "AP5":



            category = "AP5"



        elif a["resn"] in {

            "HOH",

            "WAT"

        }:



            category = "WATER"



        else:



            category = "OTHER"





        neighbors.append({

            "Zn": zn_id,

            "Zn residue": zn["resi"],

            "Zn X": zn["coord"][0],

            "Zn Y": zn["coord"][1],

            "Zn Z": zn["coord"][2],

            "Distance (A)": d,

        )





# Specifically summarize nearest AP5 / protein / water distances

def nearest(category_filter):



    hits = []



    for zn in zn_atoms:

        for a in atoms:



            if a is zn:

                continue



            if not category_filter(a):

                continue



            d = dist(

                zn["coord"],

                a["coord"]

            )



            hits.append(

                (d, a)

            )



    hits.sort(

        key=lambda x: x[0]

    )



    return hits[:10]

from pathlib import Path



PROJECT = Path(r"C:\Users\akidi\Desktop\ISO_AK_Docking")



FIXED_PROTEIN = PROJECT / "receptors" / "AK1_chainA_FIXED.pdb"

ORIGINAL_PDB  = PROJECT / "structures" / "AK1_1Z83.pdb"



OUTPUT = PROJECT / "receptors" / "AK1_chainA_FIXED_ZN601.pdb"



if not FIXED_PROTEIN.exists():

    raise FileNotFoundError(FIXED_PROTEIN)



if not ORIGINAL_PDB.exists():

    raise FileNotFoundError(ORIGINAL_PDB)



# ------------------------------------------------------------

# Read repaired AK1 protein

# ------------------------------------------------------------



protein_lines = []



with open(FIXED_PROTEIN, "r") as f:

    for line in f:



        if line.startswith("ATOM"):

            protein_lines.append(line)





# ------------------------------------------------------------

# Extract only chain-A ZN601 from original 1Z83

# ------------------------------------------------------------



zn_lines = []



with open(ORIGINAL_PDB, "r") as f:

    for line in f:



        if not line.startswith("HETATM"):

            continue



        resn = line[17:20].strip()

        chain = line[21].strip()

        resi = line[22:26].strip()



        if (

            resn == "ZN"

            and chain == "A"

            and resi == "601"

        ):

            zn_lines.append(line)





if len(zn_lines) != 1:

    raise RuntimeError(

        f"Expected exactly 1 ZN601 atom, found {len(zn_lines)}"

    )





# ------------------------------------------------------------

# Write combined receptor

# ------------------------------------------------------------



with open(OUTPUT, "w") as f:



    for line in protein_lines:

        f.write(line)



    f.write(zn_lines[0])



    f.write("TER\n")

    f.write("END\n")





print("=" * 70)

print("AK1 FIXED + ZN601 RECEPTOR CREATED")

print("=" * 70)



print("Protein ATOM records:", len(protein_lines))

print("ZN601 records:", len(zn_lines))

print("Output:")

print(OUTPUT)

END FILE: build_AK1_FIXED_ZN601.py
