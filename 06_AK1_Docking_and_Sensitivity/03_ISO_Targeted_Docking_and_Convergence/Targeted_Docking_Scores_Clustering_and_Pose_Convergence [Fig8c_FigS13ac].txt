from pathlib import Path

import pandas as pd

import numpy as np

from scipy.cluster.hierarchy import linkage, fcluster

from scipy.spatial.distance import squareform



PROJECT = Path(r"C:\Users\akidi\Desktop\ISO_AK_Docking")



matrix_file = (

    PROJECT

    / "results"

    / "AK1_ISO_pose_RMSD_matrix.csv"

)



df = pd.read_csv(matrix_file, index_col=0)



# Convert row/column labels to seed numbers

df.index = df.index.astype(int)

df.columns = df.columns.astype(int)



distance_matrix = df.values



# Convert full RMSD matrix into condensed format

condensed = squareform(

    distance_matrix,

    checks=False

)



# Average-linkage clustering

Z = linkage(

    condensed,

    method="average"

)



print()

print("=" * 60)

print("AK1 ISO POSE CLUSTERING")

print("=" * 60)



# Examine both 2 A and 3 A thresholds

for threshold in [2.0, 3.0]:



    labels = fcluster(

        Z,

        t=threshold,

        criterion="distance"

    )



    clusters = {}



    for seed, cluster in zip(df.index, labels):

        clusters.setdefault(cluster, []).append(seed)



    print()

    print(f"RMSD threshold = {threshold:.1f} A")

    print("-" * 40)



    ordered = sorted(

        clusters.items(),

        key=lambda x: len(x[1]),

        reverse=True

    )



    for number, (cluster, seeds) in enumerate(

        ordered,

        start=1

    ):



        print(

            f"Cluster {number}: "

            f"{len(seeds)} poses -> "

            f"seeds {seeds}"

        )



print()

from pathlib import Path

import re

import statistics



PROJECT = Path(r"C:\Users\akidi\Desktop\ISO_AK_Docking")

DOCKING = PROJECT / "docking"



dominant = {

    1001, 2002, 3003, 4004,

    5005, 7007, 9009, 10101

}



alternative = {

    6006, 8008

}





def get_best_score(path):



    with open(path, "r") as f:



        for line in f:



            if line.startswith("REMARK VINA RESULT:"):



                return float(

                    line.split()[3]

                )



    raise ValueError(

        f"No Vina result found in {path}"

    )





groups = {

    "Dominant cluster": dominant,

    "Alternative cluster": alternative

}





for name, seeds in groups.items():



    scores = []



    print()

    print("=" * 55)

    print(name)

    print("=" * 55)



    for seed in sorted(seeds):



        file = (

            DOCKING /

            f"AK1_ISO_seed{seed}.pdbqt"

        )



        score = get_best_score(file)



        scores.append(score)



        print(

            f"Seed {seed}: {score:.3f} kcal/mol"

        )



    print()

    print("n =", len(scores))

    print(

        "Mean =",

        f"{statistics.mean(scores):.3f}"

    )

    print(

        "Median =",

        f"{statistics.median(scores):.3f}"

    )



    if len(scores) > 1:

        print(

            "SD =",

            f"{statistics.stdev(scores):.3f}"

        )



    print(

        "Best =",

        f"{min(scores):.3f}"

    )



    print(

        "Worst =",

        f"{max(scores):.3f}"

    )

from pathlib import Path

import re

import math

import csv



PROJECT = Path(r"C:\Users\akidi\Desktop\ISO_AK_Docking")

DOCKING = PROJECT / "docking"

RESULTS = PROJECT / "results"



FILES = sorted(

    DOCKING.glob("AK1_ISO_seed*.pdbqt"),

    key=lambda p: int(re.search(r"seed(\d+)", p.name).group(1))

)





def read_best_pose_heavy_atoms(path):

    coords = []

    in_model_1 = False

    model_seen = False



    with open(path, "r") as f:

        for line in f:



            if line.startswith("MODEL"):

                model_number = int(line.split()[1])



                if model_number == 1:

                    in_model_1 = True

                    model_seen = True

                    continue



                elif in_model_1:

                    break



            if line.startswith("ENDMDL") and in_model_1:

                break



            # Some Vina files may not explicitly use MODEL

            if not model_seen and (

                line.startswith("ATOM") or line.startswith("HETATM")

            ):

                in_model_1 = True



            if in_model_1 and (

                line.startswith("ATOM") or line.startswith("HETATM")

            ):



                x = float(line[30:38])

                y = float(line[38:46])

                z = float(line[46:54])



                # AutoDock atom type is normally the last field

                atom_type = line.split()[-1].upper()



                # exclude hydrogens

                if atom_type.startswith("H"):

                    continue



                coords.append((x, y, z))



    return coords





def rmsd(coords1, coords2):



    if len(coords1) != len(coords2):

        raise ValueError(

            f"Different atom counts: {len(coords1)} vs {len(coords2)}"

        )



    squared = 0.0



    for a, b in zip(coords1, coords2):



        dx = a[0] - b[0]

        dy = a[1] - b[1]

        dz = a[2] - b[2]



        squared += dx*dx + dy*dy + dz*dz



    return math.sqrt(squared / len(coords1))





poses = {}



for file in FILES:



    seed = int(re.search(r"seed(\d+)", file.name).group(1))



    poses[seed] = read_best_pose_heavy_atoms(file)



    print(

        f"Seed {seed}: "

        f"{len(poses[seed])} heavy atoms"

    )





seeds = sorted(poses)



matrix = []



all_pairwise = []



for seed1 in seeds:



    row = []



    for seed2 in seeds:



        value = rmsd(

            poses[seed1],

            poses[seed2]

        )



        row.append(value)



        if seed2 > seed1:

            all_pairwise.append(value)



    matrix.append(row)





# ------------------------------------------------------------

# Average RMSD of each pose to all other poses

# ------------------------------------------------------------



average_to_others = {}



for i, seed in enumerate(seeds):



    values = [

        matrix[i][j]

        for j in range(len(seeds))

        if i != j

    ]



    average_to_others[seed] = sum(values) / len(values)





representative_seed = min(

    average_to_others,

    key=average_to_others.get

)





# ------------------------------------------------------------

# Output CSV matrix

# ------------------------------------------------------------



matrix_file = RESULTS / "AK1_ISO_pose_RMSD_matrix.csv"



with open(matrix_file, "w", newline="") as f:



    writer = csv.writer(f)



    writer.writerow(

        ["Seed"] + seeds

    )



    for seed, row in zip(seeds, matrix):



        writer.writerow(

            [seed] +

            [round(x, 3) for x in row]

        )





# ------------------------------------------------------------

# Summary

# ------------------------------------------------------------



mean_pairwise = sum(all_pairwise) / len(all_pairwise)



minimum_pairwise = min(all_pairwise)

maximum_pairwise = max(all_pairwise)





print()

print("=" * 60)

print("AK1 ISO POSE CONVERGENCE")

print("=" * 60)



print(

    "Number of best poses:",

    len(seeds)

)



print(

    "Pairwise comparisons:",

    len(all_pairwise)

)



print(

    f"Mean pairwise RMSD: {mean_pairwise:.3f} A"

)



print(

    f"Minimum pairwise RMSD: {minimum_pairwise:.3f} A"

)



print(

    f"Maximum pairwise RMSD: {maximum_pairwise:.3f} A"

)



print()



print(

    "Most representative pose:"

)



print(

    f"Seed {representative_seed}"

)



print(

    f"Mean RMSD to other poses: "

    f"{average_to_others[representative_seed]:.3f} A"

)



print()



print(

    "RMSD matrix saved to:"

)



print(matrix_file)

# HELPER: READ VINA PDBQT RESULTS

# ============================================================



def parse_vina_pdbqt(path):



    poses = []



    with open(path, encoding="utf-8") as f:



        for line in f:



            if line.startswith("REMARK VINA RESULT:"):



                parts = line.split()



                poses.append({

                    "mode": len(poses) + 1,

                    "score": float(parts[3]),

                    "rmsd_lb": float(parts[4]),

                    "rmsd_ub": float(parts[5])

                })



    return poses

    r"AK1_ISO_seed(\d+)\.pdbqt$",

    re.IGNORECASE

)



runs = []

all_poses = []



for file in DOCKING.glob("AK1_ISO_seed*.pdbqt"):



    match = pattern.match(file.name)



    if not match:

        continue



    seed = int(match.group(1))



    poses = parse_vina_pdbqt(file)



    if not poses:

        print("WARNING: no Vina results in", file.name)

        continue



    best = poses[0]



    runs.append({

        "seed": seed,

        "best_score": best["score"],

        "file": file.name

    })



    for pose in poses:



        all_poses.append({

            "seed": seed,

            "mode": pose["mode"],

            "score": pose["score"],

            "rmsd_lb": pose["rmsd_lb"],

            "rmsd_ub": pose["rmsd_ub"],

            "file": file.name

        })

# ============================================================

# HIERARCHICAL CLUSTERING

# ============================================================



condensed = squareform(

    rmsd_matrix,

    checks=False

)



Z = linkage(

    condensed,

    method="average"

)





def make_clusters(threshold):



    labels = fcluster(

        Z,

        t=threshold,

        criterion="distance"

    )



    clusters = {}



    for seed, cluster_id in zip(matrix_seeds, labels):



        clusters.setdefault(

            int(cluster_id),

            []

        ).append(seed)



    # Sort clusters largest first

    ordered = sorted(

        clusters.values(),

        key=lambda x: len(x),

        reverse=True

    )



    return ordered

def cluster_stats(seeds):



    cluster_scores = [

        score_by_seed[seed]

        for seed in seeds

    ]



    return {

        "n": len(cluster_scores),

        "mean": statistics.mean(cluster_scores),

        "median": statistics.median(cluster_scores),

        "sd": (

            statistics.stdev(cluster_scores)

            if len(cluster_scores) > 1

            else 0

        ),

        "best": min(cluster_scores),

        "worst": max(cluster_scores),

        "range": max(cluster_scores) - min(cluster_scores)

    }

# ============================================================

# READ VINA PDBQT FILES

# ============================================================



def parse_vina_pdbqt(path):



    results = []



    with open(path, encoding="utf-8") as f:



        for line in f:



            if line.startswith("REMARK VINA RESULT:"):



                parts = line.split()



                score = float(parts[3])

                rmsd_lb = float(parts[4])

                rmsd_ub = float(parts[5])



                results.append({

                    "mode": len(results) + 1,

                    "score": score,

                    "rmsd_lb": rmsd_lb,

                    "rmsd_ub": rmsd_ub

                })



    return results





# ============================================================

# FIND ALL ISO MULTI-SEED RESULTS AUTOMATICALLY

# Example:

# AK1_ISO_seed1001.pdbqt

# ============================================================



pattern = re.compile(

    r"(.+)_ISO_seed(\d+)\.pdbqt$",

    re.IGNORECASE

)



all_runs = []

all_modes = []



for file in sorted(DOCKING.glob("*_ISO_seed*.pdbqt")):



    match = pattern.match(file.name)



    if not match:

        continue



    target = match.group(1)

    seed = int(match.group(2))



    poses = parse_vina_pdbqt(file)



    if not poses:

        print("WARNING: no Vina result found in", file.name)

        continue



    best = poses[0]



    all_runs.append({

        "target": target,

        "seed": seed,

        "best_score": best["score"],

        "file": file.name

    })



    for pose in poses:



        all_modes.append({

            "target": target,

            "seed": seed,

            "mode": pose["mode"],

            "score": pose["score"],

            "rmsd_lb": pose["rmsd_lb"],

            "rmsd_ub": pose["rmsd_ub"],

            "file": file.name

        })





# ============================================================

# SUMMARY BY TARGET

# ============================================================



targets = sorted(set(x["target"] for x in all_runs))



summary = []



for target in targets:



    scores = [

        x["best_score"]

        for x in all_runs

        if x["target"] == target

    info = validation.get(target, {})

# ============================================================
# READ VINA PDBQT FILES
# ============================================================

def parse_vina_pdbqt(path):

    results = []

    with open(path, encoding="utf-8") as f:

        for line in f:

            if line.startswith("REMARK VINA RESULT:"):

                parts = line.split()

                score = float(parts[3])
                rmsd_lb = float(parts[4])
                rmsd_ub = float(parts[5])

                results.append({
                    "mode": len(results) + 1,
                    "score": score,
                    "rmsd_lb": rmsd_lb,
                    "rmsd_ub": rmsd_ub
                })

    return results


# ============================================================
# FIND ALL ISO MULTI-SEED RESULTS AUTOMATICALLY
# Example:
# AK1_ISO_seed1001.pdbqt
# ============================================================

pattern = re.compile(
    r"(.+)_ISO_seed(\d+)\.pdbqt$",
    re.IGNORECASE
)

all_runs = []
all_modes = []

for file in sorted(DOCKING.glob("AK1_ISO_seed*.pdbqt")):

    match = pattern.match(file.name)

    if not match:
        continue

    target = match.group(1)
    seed = int(match.group(2))

    poses = parse_vina_pdbqt(file)

    if not poses:
        print("WARNING: no Vina result found in", file.name)
        continue

    best = poses[0]

    all_runs.append({
        "target": target,
        "seed": seed,
        "best_score": best["score"],
        "file": file.name
    })

    for pose in poses:

        all_modes.append({
            "target": target,
            "seed": seed,
            "mode": pose["mode"],
            "score": pose["score"],
            "rmsd_lb": pose["rmsd_lb"],
            "rmsd_ub": pose["rmsd_ub"],
            "file": file.name
        })


# ============================================================
# SUMMARY BY TARGET
# ============================================================

targets = sorted(set(x["target"] for x in all_runs))

summary = []

for target in targets:

    scores = [
        x["best_score"]
        for x in all_runs
        if x["target"] == target
    ]

    info = validation.get(target, {})

    summary.append({
        "target": target,
        "structure": info.get("structure", ""),
        "n_runs": len(scores),
        "best_score": min(scores),
        "mean_score": statistics.mean(scores),
        "median_score": statistics.median(scores),
        "sd_score": statistics.stdev(scores) if len(scores) > 1 else None,
        "min_score": min(scores),
        "max_score": max(scores),
        "score_range": max(scores) - min(scores),
        "validation_ligand": info.get("validation_ligand", ""),
        "redocking_rmsd": info.get("redocking_rmsd_A", None)
    })


# More negative median score = stronger predicted docking
summary.sort(key=lambda x: x["median_score"])
